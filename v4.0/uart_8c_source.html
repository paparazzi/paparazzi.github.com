55 bool_t uart_check_free_space(struct uart_periph* p, uint8_t len) {
56  int16_t space = p->tx_extract_idx - p->tx_insert_idx;
57  if (space <= 0)
58  space += UART_TX_BUFFER_SIZE;
59  return (uint16_t)(space - 1) >= len;
