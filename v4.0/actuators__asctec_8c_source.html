77 void actuators_set(bool_t motors_on) {
78 #if defined ACTUATORS_START_DELAY && ! defined SITL
79  if (!actuators_delay_done) {

85  switch (actuators_asctec.i2c_trans.status) {
86  case I2CTransFailed:
87  actuators_asctec.nb_err++;

91  case I2CTransDone:
92  actuators_asctec.i2c_trans.status = I2CTransDone;

95  actuators_asctec.nb_err++;
96  return;
97  }
98 
99 #ifdef KILL_MOTORS
100  actuators_asctec.cmds[PITCH] = 0;
101  actuators_asctec.cmds[ROLL] = 0;
102  actuators_asctec.cmds[YAW] = 0;
103  actuators_asctec.cmds[THRUST] = 0;
104 #else /* ! KILL_MOTORS */
105  actuators_asctec.cmds[PITCH] = ((commands[COMMAND_PITCH] + SUPERVISION_TRIM_E) * ASCTEC_MAX_CMD) / MAX_PPRZ;
106  actuators_asctec.cmds[ROLL] = ((commands[COMMAND_ROLL] + SUPERVISION_TRIM_A) * ASCTEC_MAX_CMD) / MAX_PPRZ;
107  actuators_asctec.cmds[YAW] = ((commands[COMMAND_YAW] + SUPERVISION_TRIM_R) * ASCTEC_MAX_CMD) / MAX_PPRZ;
108  actuators_asctec.cmds[THRUST] = (commands[COMMAND_THRUST] * ASCTEC_MAX_THROTTLE) / MAX_PPRZ;
109  Bound(actuators_asctec.cmds[PITCH],ASCTEC_MIN_CMD, ASCTEC_MAX_CMD);
110  Bound(actuators_asctec.cmds[ROLL], ASCTEC_MIN_CMD, ASCTEC_MAX_CMD);
111  Bound(actuators_asctec.cmds[YAW], ASCTEC_MIN_CMD, ASCTEC_MAX_CMD);
112  if (motors_on) {
113  Bound(actuators_asctec.cmds[THRUST], ASCTEC_MIN_THROTTLE + 1, ASCTEC_MAX_THROTTLE);
114  }
115  else
116  actuators_asctec.cmds[THRUST] = 0;
117 #endif /* KILL_MOTORS */
118 
119  switch (actuators_asctec.cmd) {

122  actuators_asctec.i2c_trans.buf[1] = actuators_asctec.cur_addr;

130  actuators_asctec.i2c_trans.buf[3] = 234 + actuators_asctec.cur_addr;
131  break;
132  case SET_ADDR:
133  actuators_asctec.i2c_trans.buf[0] = 250;
134  actuators_asctec.i2c_trans.buf[1] = actuators_asctec.cur_addr;
135  actuators_asctec.i2c_trans.buf[2] = actuators_asctec.new_addr;
136  actuators_asctec.i2c_trans.buf[3] = 230 + actuators_asctec.cur_addr + actuators_asctec.new_addr;
137  actuators_asctec.cur_addr = actuators_asctec.new_addr;
138  break;
139  case NONE:
140  actuators_asctec.i2c_trans.buf[0] = 100 - actuators_asctec.cmds[PITCH];
141  actuators_asctec.i2c_trans.buf[1] = 100 + actuators_asctec.cmds[ROLL];
142  actuators_asctec.i2c_trans.buf[2] = 100 - actuators_asctec.cmds[YAW];
