192 static void attitude_run_ff(int32_t ff_commands[], struct Int32AttitudeGains *gains, struct Int32Rates *ref_accel)
193 {
194  /* Compute feedforward based on reference acceleration */
195 
196  ff_commands[COMMAND_ROLL] = GAIN_PRESCALER_FF * gains->dd.x * RATE_FLOAT_OF_BFP(ref_accel->p) / (1 << 7);
197  ff_commands[COMMAND_PITCH] = GAIN_PRESCALER_FF * gains->dd.y * RATE_FLOAT_OF_BFP(ref_accel->q) / (1 << 7);
198  ff_commands[COMMAND_YAW] = GAIN_PRESCALER_FF * gains->dd.z * RATE_FLOAT_OF_BFP(ref_accel->r) / (1 << 7);
199 }

201 static void attitude_run_fb(int32_t fb_commands[], struct Int32AttitudeGains *gains, struct Int32Quat *att_err,
202  struct Int32Rates *rate_err, struct Int32Quat *sum_err)
203 {
204  /* PID feedback */
205  fb_commands[COMMAND_ROLL] =
206  GAIN_PRESCALER_P * gains->p.x * QUAT1_FLOAT_OF_BFP(att_err->qx) / 4 +
207  GAIN_PRESCALER_D * gains->d.x * RATE_FLOAT_OF_BFP(rate_err->p) / 16 +
208  GAIN_PRESCALER_I * gains->i.x * QUAT1_FLOAT_OF_BFP(sum_err->qx) / 2;
209 
210  fb_commands[COMMAND_PITCH] =
211  GAIN_PRESCALER_P * gains->p.y * QUAT1_FLOAT_OF_BFP(att_err->qy) / 4 +
212  GAIN_PRESCALER_D * gains->d.y * RATE_FLOAT_OF_BFP(rate_err->q) / 16 +
213  GAIN_PRESCALER_I * gains->i.y * QUAT1_FLOAT_OF_BFP(sum_err->qy) / 2;
214 
215  fb_commands[COMMAND_YAW] =
216  GAIN_PRESCALER_P * gains->p.z * QUAT1_FLOAT_OF_BFP(att_err->qz) / 4 +
217  GAIN_PRESCALER_D * gains->d.z * RATE_FLOAT_OF_BFP(rate_err->r) / 16 +
218  GAIN_PRESCALER_I * gains->i.z * QUAT1_FLOAT_OF_BFP(sum_err->qz) / 2;

222 void stabilization_attitude_run(bool_t enable_integrator)
223 {
224 
225  /*
226  * Update reference
227  */
228  stabilization_attitude_ref_update();
229 
230  /*
231  * Compute errors for feedback
232  */
233 
234  /* attitude error */
235  struct Int32Quat att_err;
236  struct Int32Quat *att_quat = stateGetNedToBodyQuat_i();
237  INT32_QUAT_INV_COMP(att_err, *att_quat, stab_att_ref_quat);
238  /* wrap it in the shortest direction */
239  int32_quat_wrap_shortest(&att_err);
240  int32_quat_normalize(&att_err);
241 
242  /* rate error */
243  const struct Int32Rates rate_ref_scaled = {
244  OFFSET_AND_ROUND(stab_att_ref_rate.p, (REF_RATE_FRAC - INT32_RATE_FRAC)),
245  OFFSET_AND_ROUND(stab_att_ref_rate.q, (REF_RATE_FRAC - INT32_RATE_FRAC)),
246  OFFSET_AND_ROUND(stab_att_ref_rate.r, (REF_RATE_FRAC - INT32_RATE_FRAC))
247  };
248  struct Int32Rates rate_err;
249  struct Int32Rates *body_rate = stateGetBodyRates_i();
250  RATES_DIFF(rate_err, rate_ref_scaled, (*body_rate));
251 
252 #define INTEGRATOR_BOUND 100000
253  /* integrated error */
254  if (enable_integrator) {
255  stabilization_att_sum_err_quat.qx += att_err.qx / IERROR_SCALE;
256  stabilization_att_sum_err_quat.qy += att_err.qy / IERROR_SCALE;
257  stabilization_att_sum_err_quat.qz += att_err.qz / IERROR_SCALE;
258  Bound(stabilization_att_sum_err_quat.qx, -INTEGRATOR_BOUND, INTEGRATOR_BOUND);
259  Bound(stabilization_att_sum_err_quat.qy, -INTEGRATOR_BOUND, INTEGRATOR_BOUND);
260  Bound(stabilization_att_sum_err_quat.qz, -INTEGRATOR_BOUND, INTEGRATOR_BOUND);
261  } else {
262  /* reset accumulator */
263  int32_quat_identity(&stabilization_att_sum_err_quat);
264  }
265 
266  /* compute the feed forward command */
267  attitude_run_ff(stabilization_att_ff_cmd, &stabilization_gains, &stab_att_ref_accel);
268 
269  /* compute the feed back command */
270  attitude_run_fb(stabilization_att_fb_cmd, &stabilization_gains, &att_err, &rate_err, &stabilization_att_sum_err_quat);
271 
272  /* sum feedforward and feedback */
273  stabilization_cmd[COMMAND_ROLL] = stabilization_att_fb_cmd[COMMAND_ROLL] + stabilization_att_ff_cmd[COMMAND_ROLL];
274  stabilization_cmd[COMMAND_PITCH] = stabilization_att_fb_cmd[COMMAND_PITCH] + stabilization_att_ff_cmd[COMMAND_PITCH];
275  stabilization_cmd[COMMAND_YAW] = stabilization_att_fb_cmd[COMMAND_YAW] + stabilization_att_ff_cmd[COMMAND_YAW];
276 
277  /* bound the result */
278  BoundAbs(stabilization_cmd[COMMAND_ROLL], MAX_PPRZ);
279  BoundAbs(stabilization_cmd[COMMAND_PITCH], MAX_PPRZ);
280  BoundAbs(stabilization_cmd[COMMAND_YAW], MAX_PPRZ);
281 }
