71 void mpu60x0_i2c_read(struct Mpu60x0_I2c *mpu)
72 {
73  if (mpu->config.initialized && mpu->i2c_trans.status == I2CTransDone) {
74  /* set read bit and multiple byte bit, then address */
75  mpu->i2c_trans.buf[0] = MPU60X0_REG_INT_STATUS;
76  i2c_transceive(mpu->i2c_p, &(mpu->i2c_trans), mpu->i2c_trans.slave_addr, 1, mpu->config.nb_bytes);
77  }
78 }

82 void mpu60x0_i2c_event(struct Mpu60x0_I2c *mpu)
83 {
84  if (mpu->config.initialized) {
85  if (mpu->i2c_trans.status == I2CTransFailed) {
86  mpu->i2c_trans.status = I2CTransDone;
87  } else if (mpu->i2c_trans.status == I2CTransSuccess) {
88  // Successfull reading
89  if (bit_is_set(mpu->i2c_trans.buf[0], 0)) {
90  // new data
91  mpu->data_accel.vect.x = Int16FromBuf(mpu->i2c_trans.buf, 1);
92  mpu->data_accel.vect.y = Int16FromBuf(mpu->i2c_trans.buf, 3);
93  mpu->data_accel.vect.z = Int16FromBuf(mpu->i2c_trans.buf, 5);
94  mpu->data_rates.rates.p = Int16FromBuf(mpu->i2c_trans.buf, 9);
95  mpu->data_rates.rates.q = Int16FromBuf(mpu->i2c_trans.buf, 11);
96  mpu->data_rates.rates.r = Int16FromBuf(mpu->i2c_trans.buf, 13);
97 
98  // if we are reading slaves through the mpu, copy the ext_sens_data
99  if ((mpu->config.i2c_bypass == FALSE) && (mpu->config.nb_slaves > 0)) {
100  /* the buffer is volatile, since filled from ISR
101  * but we know it's ok to use it here so we silence the warning
102  */
103 #pragma GCC diagnostic push
104 #pragma GCC diagnostic ignored "-Wcast-qual"
105  memcpy(mpu->data_ext, (uint8_t *) & (mpu->i2c_trans.buf[15]), mpu->config.nb_bytes - 15);
106 #pragma GCC diagnostic pop
107  }
108 
109  mpu->data_available = TRUE;
110  }
111  mpu->i2c_trans.status = I2CTransDone;
112  }
113  } else if (mpu->config.init_status != MPU60X0_CONF_UNINIT) { // Configuring but not yet initialized
114  switch (mpu->i2c_trans.status) {
115  case I2CTransFailed:
116  mpu->config.init_status--; // Retry config (TODO max retry)
117  case I2CTransSuccess:
118  case I2CTransDone:
119  mpu60x0_send_config(mpu60x0_i2c_write_to_reg, (void *)mpu, &(mpu->config));
120  if (mpu->config.initialized) {
121  mpu->i2c_trans.status = I2CTransDone;
122  }
123  break;
124  default:
125  break;
126  }
127  }
128 }

131 bool_t mpu60x0_configure_i2c_slaves(Mpu60x0ConfigSet mpu_set, void *mpu)
132 {
133  struct Mpu60x0_I2c *mpu_i2c = (struct Mpu60x0_I2c *)(mpu);
134 
135  if (mpu_i2c->slave_init_status == MPU60X0_I2C_CONF_UNINIT) {
136  mpu_i2c->slave_init_status++;
137  }
138 
139  switch (mpu_i2c->slave_init_status) {
140  case MPU60X0_I2C_CONF_I2C_MST_DIS:
141  mpu_set(mpu, MPU60X0_REG_USER_CTRL, 0);
142  mpu_i2c->slave_init_status++;
143  break;
144  case MPU60X0_I2C_CONF_I2C_BYPASS_EN:
145  /* switch to I2C passthrough */
146  mpu_set(mpu, MPU60X0_REG_INT_PIN_CFG, (1 << 1));
147  mpu_i2c->slave_init_status++;
148  break;
149  case MPU60X0_I2C_CONF_SLAVES_CONFIGURE:
150  /* configure each slave. TODO: currently only one */
151  if (mpu_i2c->config.slaves[0].configure(mpu_set, mpu)) {
152  mpu_i2c->slave_init_status++;
153  }
154  break;
155  case MPU60X0_I2C_CONF_I2C_BYPASS_DIS:
156  if (mpu_i2c->config.i2c_bypass) {
157  /* if bypassing I2C skip MPU I2C master setup */
158  mpu_i2c->slave_init_status = MPU60X0_I2C_CONF_DONE;
159  } else {
160  /* disable I2C passthrough again */
161  mpu_set(mpu, MPU60X0_REG_INT_PIN_CFG, (0 << 1));
162  mpu_i2c->slave_init_status++;
163  }
164  break;
165  case MPU60X0_I2C_CONF_I2C_MST_CLK:
166  /* configure MPU I2C master clock and stop/start between slave reads */
167  mpu_set(mpu, MPU60X0_REG_I2C_MST_CTRL,
168  ((1 << 4) | mpu_i2c->config.i2c_mst_clk));
169  mpu_i2c->slave_init_status++;
170  break;
171  case MPU60X0_I2C_CONF_I2C_MST_DELAY:
172  /* Set I2C slaves delayed sample rate */
173  mpu_set(mpu, MPU60X0_REG_I2C_MST_DELAY, mpu_i2c->config.i2c_mst_delay);
174  mpu_i2c->slave_init_status++;
175  break;
176  case MPU60X0_I2C_CONF_I2C_SMPLRT:
177  /* I2C slave0 sample rate/2 = 100/2 = 50Hz */
178  mpu_set(mpu, MPU60X0_REG_I2C_SLV4_CTRL, 0);
179  mpu_i2c->slave_init_status++;
180  break;
181  case MPU60X0_I2C_CONF_I2C_MST_EN:
182  /* enable internal I2C master */
183  mpu_set(mpu, MPU60X0_REG_USER_CTRL, (1 << MPU60X0_I2C_MST_EN));
184  mpu_i2c->slave_init_status++;
185  break;
186  case MPU60X0_I2C_CONF_DONE:
187  return TRUE;
188  default:
189  break;
190  }
191  return FALSE;
192 }
