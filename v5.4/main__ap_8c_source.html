311 static inline uint8_t pprz_mode_update(void)
312 {
313  if ((pprz_mode != PPRZ_MODE_HOME &&
314  pprz_mode != PPRZ_MODE_GPS_OUT_OF_ORDER)
315 #ifdef UNLOCKED_HOME_MODE
316  || TRUE
317 #endif
318  ) {
319 #ifndef RADIO_AUTO_MODE
320  return ModeUpdate(pprz_mode, PPRZ_MODE_OF_PULSE(fbw_state->channels[RADIO_MODE]));
321 #else
322  INFO("Using RADIO_AUTO_MODE to switch between AUTO1 and AUTO2.")
323  /* If RADIO_AUTO_MODE is enabled mode swithing will be seperated between two switches/channels
324  * RADIO_MODE will switch between PPRZ_MODE_MANUAL and any PPRZ_MODE_AUTO mode selected by RADIO_AUTO_MODE.
325  *
326  * This is mainly a cludge for entry level radios with no three-way switch but two available two-way switches which can be used.
327  */
328  if (PPRZ_MODE_OF_PULSE(fbw_state->channels[RADIO_MODE]) == PPRZ_MODE_MANUAL) {
329  /* RADIO_MODE in MANUAL position */
330  return ModeUpdate(pprz_mode, PPRZ_MODE_MANUAL);
331  } else {
332  /* RADIO_MODE not in MANUAL position.
333  * Select AUTO mode bassed on RADIO_AUTO_MODE channel
334  */
335  return ModeUpdate(pprz_mode, (fbw_state->channels[RADIO_AUTO_MODE] > THRESHOLD2) ? PPRZ_MODE_AUTO2 : PPRZ_MODE_AUTO1);
336  }
337 #endif // RADIO_AUTO_MODE
338  } else {
339  return FALSE;
340  }
341 }

380 static inline void telecommand_task(void)
381 {
382  uint8_t mode_changed = FALSE;
383  copy_from_to_fbw();
384 
385  uint8_t really_lost = bit_is_set(fbw_state->status, STATUS_RADIO_REALLY_LOST) && (pprz_mode == PPRZ_MODE_AUTO1
386  || pprz_mode == PPRZ_MODE_MANUAL);
387  if (pprz_mode != PPRZ_MODE_HOME && pprz_mode != PPRZ_MODE_GPS_OUT_OF_ORDER && launch) {
388  if (too_far_from_home) {
389  pprz_mode = PPRZ_MODE_HOME;
390  mode_changed = TRUE;
391  }
392  if (really_lost) {
393  pprz_mode = RC_LOST_MODE;
394  mode_changed = TRUE;
395  }
396  }
397  if (bit_is_set(fbw_state->status, AVERAGED_CHANNELS_SENT)) {
398  bool_t pprz_mode_changed = pprz_mode_update();
399  mode_changed |= pprz_mode_changed;
400 #if defined RADIO_CALIB && defined RADIO_CONTROL_SETTINGS
401  bool_t calib_mode_changed = RcSettingsModeUpdate(fbw_state->channels);
402  rc_settings(calib_mode_changed || pprz_mode_changed);
403  mode_changed |= calib_mode_changed;
404 #endif
405  }
406  mode_changed |= mcu1_status_update();
407  if (mode_changed) { autopilot_send_mode(); }
408 
409 #if defined RADIO_CONTROL || defined RADIO_CONTROL_AUTO1
410 
413  if (pprz_mode == PPRZ_MODE_AUTO1) {
415  h_ctl_roll_setpoint = FLOAT_OF_PPRZ(fbw_state->channels[RADIO_ROLL], 0., AUTO1_MAX_ROLL);
416 
418  h_ctl_pitch_setpoint = FLOAT_OF_PPRZ(fbw_state->channels[RADIO_PITCH], 0., AUTO1_MAX_PITCH);
419  }
423  if (pprz_mode == PPRZ_MODE_MANUAL || pprz_mode == PPRZ_MODE_AUTO1) {
424  v_ctl_throttle_setpoint = fbw_state->channels[RADIO_THROTTLE];
425  }
428  mcu1_ppm_cpt = fbw_state->ppm_cpt;
429 #endif // RADIO_CONTROL
430 
431 
432  vsupply = fbw_state->vsupply;
433  current = fbw_state->current;
434  energy = fbw_state->energy;
435 
436 #ifdef RADIO_CONTROL
437  /* the SITL check is a hack to prevent "automatic" launch in NPS */
438 #ifndef SITL
439  if (!autopilot_flight_time) {
440  if (pprz_mode == PPRZ_MODE_AUTO2 && fbw_state->channels[RADIO_THROTTLE] > THROTTLE_THRESHOLD_TAKEOFF) {
441  launch = TRUE;
442  }
443  }
444 #endif
445 #endif
446 }

551 void attitude_loop(void)
552 {
553 
554 #if USE_INFRARED
555  ahrs_update_infrared();
556 #endif /* USE_INFRARED */
557 
558  if (pprz_mode >= PPRZ_MODE_AUTO2) {
559  if (v_ctl_mode == V_CTL_MODE_AUTO_THROTTLE) {
560  v_ctl_throttle_setpoint = nav_throttle_setpoint;
561  v_ctl_pitch_setpoint = nav_pitch;
562  } else if (v_ctl_mode >= V_CTL_MODE_AUTO_CLIMB) {
563  v_ctl_climb_loop();
564  }
565 
566 #if defined V_CTL_THROTTLE_IDLE
567  Bound(v_ctl_throttle_setpoint, TRIM_PPRZ(V_CTL_THROTTLE_IDLE * MAX_PPRZ), MAX_PPRZ);
568 #endif
569 
570 #ifdef V_CTL_POWER_CTL_BAT_NOMINAL
571  if (vsupply > 0.) {
572  v_ctl_throttle_setpoint *= 10. * V_CTL_POWER_CTL_BAT_NOMINAL / (float)vsupply;
573  v_ctl_throttle_setpoint = TRIM_UPPRZ(v_ctl_throttle_setpoint);
574  }
575 #endif
576 
577  h_ctl_pitch_setpoint = v_ctl_pitch_setpoint; // Copy the pitch setpoint from the guidance to the stabilization control
578  Bound(h_ctl_pitch_setpoint, H_CTL_PITCH_MIN_SETPOINT, H_CTL_PITCH_MAX_SETPOINT);
579  if (kill_throttle || (!autopilot_flight_time && !launch)) {
580  v_ctl_throttle_setpoint = 0;
581  }
582  }
583 
584  h_ctl_attitude_loop(); /* Set h_ctl_aileron_setpoint & h_ctl_elevator_setpoint */
585  v_ctl_throttle_slew();
586  ap_state->commands[COMMAND_THROTTLE] = v_ctl_throttle_slewed;
587  ap_state->commands[COMMAND_ROLL] = -h_ctl_aileron_setpoint;
588 
589  ap_state->commands[COMMAND_PITCH] = h_ctl_elevator_setpoint;
590 
591 #if defined MCU_SPI_LINK || defined MCU_UART_LINK || defined MCU_CAN_LINK
592  link_mcu_send();
593 #elif defined INTER_MCU && defined SINGLE_MCU
594 
595  inter_mcu_received_ap = TRUE;
596 #endif
597
