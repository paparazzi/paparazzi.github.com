68 void imu_impl_init(void)
69 {
70  imu_px4fmu.accel_valid = FALSE;
71  imu_px4fmu.gyro_valid = FALSE;
72  imu_px4fmu.mag_valid = FALSE;
73 
74  /* MPU is on spi1 and CS is SLAVE2 */
75  mpu60x0_spi_init(&imu_px4fmu.mpu, &spi1, SPI_SLAVE2);
76  // change the default configuration
77  imu_px4fmu.mpu.config.smplrt_div = PX4FMU_SMPLRT_DIV;
78  imu_px4fmu.mpu.config.dlpf_cfg = PX4FMU_LOWPASS_FILTER;
79  imu_px4fmu.mpu.config.gyro_range = PX4FMU_GYRO_RANGE;
80  imu_px4fmu.mpu.config.accel_range = PX4FMU_ACCEL_RANGE;
81 
82  /* initialize mag on i2c2 and set default options */
83  hmc58xx_init(&imu_px4fmu.hmc, &i2c2, HMC58XX_ADDR);
84 }

95 void imu_px4fmu_event(void)
96 {
97  mpu60x0_spi_event(&imu_px4fmu.mpu);
98  if (imu_px4fmu.mpu.data_available) {
99  RATES_ASSIGN(imu.gyro_unscaled,
100  imu_px4fmu.mpu.data_rates.rates.q,
101  imu_px4fmu.mpu.data_rates.rates.p,
102  -imu_px4fmu.mpu.data_rates.rates.r);
103  VECT3_ASSIGN(imu.accel_unscaled,
104  imu_px4fmu.mpu.data_accel.vect.y,
105  imu_px4fmu.mpu.data_accel.vect.x,
106  -imu_px4fmu.mpu.data_accel.vect.z);
107  imu_px4fmu.mpu.data_available = FALSE;
108  imu_px4fmu.gyro_valid = TRUE;
109  imu_px4fmu.accel_valid = TRUE;
110  }
111 
112  /* HMC58XX event task */
113  hmc58xx_event(&imu_px4fmu.hmc);
114  if (imu_px4fmu.hmc.data_available) {
115  imu.mag_unscaled.x = imu_px4fmu.hmc.data.vect.y;
116  imu.mag_unscaled.y = imu_px4fmu.hmc.data.vect.x;
117  imu.mag_unscaled.z = -imu_px4fmu.hmc.data.vect.z;
118  imu_px4fmu.hmc.data_available = FALSE;
119  imu_px4fmu.mag_valid = TRUE;
120  }
121 }
