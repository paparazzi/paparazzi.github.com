162 void ppm_arch_init ( void ) {
163 
164  /* timer clock enable */
165  rcc_peripheral_enable_clock(PPM_RCC, PPM_PERIPHERAL);
166 
167  /* GPIO clock enable */
168  gpio_enable_clock(PPM_GPIO_PORT);
169 
170  /* timer gpio configuration */
171  gpio_setup_pin_af(PPM_GPIO_PORT, PPM_GPIO_PIN, PPM_GPIO_AF, FALSE);
172 
173  /* Time Base configuration */
174  timer_reset(PPM_TIMER);
175  timer_set_mode(PPM_TIMER, TIM_CR1_CKD_CK_INT,
176  TIM_CR1_CMS_EDGE, TIM_CR1_DIR_UP);
177  timer_set_period(PPM_TIMER, 0xFFFF);
178  timer_set_prescaler(PPM_TIMER, (PPM_TIMER_CLK / (RC_PPM_TICKS_PER_USEC*ONE_MHZ_CLK)) - 1);
179 
180  /* TIM configuration: Input Capture mode ---------------------
181  The Rising edge is used as active edge
182  ------------------------------------------------------------ */
183 #if defined PPM_PULSE_TYPE && PPM_PULSE_TYPE == PPM_PULSE_TYPE_POSITIVE
184  timer_ic_set_polarity(PPM_TIMER, PPM_CHANNEL, TIM_IC_RISING);
185 #elif defined PPM_PULSE_TYPE && PPM_PULSE_TYPE == PPM_PULSE_TYPE_NEGATIVE
186  timer_ic_set_polarity(PPM_TIMER, PPM_CHANNEL, TIM_IC_FALLING);
187 #else
188 #error "Unknown PPM_PULSE_TYPE"
189 #endif
190  timer_ic_set_input(PPM_TIMER, PPM_CHANNEL, PPM_TIMER_INPUT);
191  timer_ic_set_prescaler(PPM_TIMER, PPM_CHANNEL, TIM_IC_PSC_OFF);
192  timer_ic_set_filter(PPM_TIMER, PPM_CHANNEL, TIM_IC_OFF);
193 
194  /* Enable timer Interrupt(s). */
195  nvic_set_priority(PPM_IRQ, 2);
196  nvic_enable_irq(PPM_IRQ);
197 
198 #ifdef PPM_IRQ2
199  nvic_set_priority(PPM_IRQ2, 2);
200  nvic_enable_irq(PPM_IRQ2);
201 #endif
202 
203  /* Enable the Capture/Compare and Update interrupt requests. */
204  timer_enable_irq(PPM_TIMER, (PPM_CC_IE | TIM_DIER_UIE));
205 
206  /* Enable capture channel. */
207  timer_ic_enable(PPM_TIMER, PPM_CHANNEL);
208 
209  /* TIM enable counter */
210  timer_enable_counter(PPM_TIMER);
211 
212  ppm_last_pulse_time = 0;
213  ppm_cur_pulse = RADIO_CONTROL_NB_CHANNEL;
214  timer_rollover_cnt = 0;
215 
216 }

220 void tim2_isr(void) {
221 
222  if((TIM2_SR & PPM_CC_IF) != 0) {
223  timer_clear_flag(TIM2, PPM_CC_IF);
224 
225  uint32_t now = timer_get_counter(TIM2) + timer_rollover_cnt;
226  DecodePpmFrame(now);
227  }
228  else if((TIM2_SR & TIM_SR_UIF) != 0) {
229  timer_rollover_cnt+=(1<<16);
230  timer_clear_flag(TIM2, TIM_SR_UIF);
231  }
232
