111 static inline void autopilot_check_in_flight_no_rc( bool_t motors_on ) {
112  if (autopilot_in_flight) {
113  if (autopilot_in_flight_counter > 0) {
114  if (stabilization_cmd[COMMAND_THRUST] == 0) {
115  autopilot_in_flight_counter--;
116  if (autopilot_in_flight_counter == 0) {
117  autopilot_in_flight = FALSE;
118  }
119  }
120  else { /* !THROTTLE_STICK_DOWN */
121  autopilot_in_flight_counter = AUTOPILOT_IN_FLIGHT_TIME;
122  }
123  }
124  }
125  else { /* not in flight */
126  if (autopilot_in_flight_counter < AUTOPILOT_IN_FLIGHT_TIME &&
127  motors_on) {
128  if (stabilization_cmd[COMMAND_THRUST] > 0) {
129  autopilot_in_flight_counter++;
130  if (autopilot_in_flight_counter == AUTOPILOT_IN_FLIGHT_TIME)
131  autopilot_in_flight = TRUE;
132  }
133  else { /* THROTTLE_STICK_DOWN */
134  autopilot_in_flight_counter = 0;
135  }
136  }
137  }

141 void autopilot_periodic(void) {
142 
143  RunOnceEvery(NAV_PRESCALER, nav_periodic_task());
144 #if FAILSAFE_GROUND_DETECT
145 INFO("Using FAILSAFE_GROUND_DETECT")
146  if (autopilot_mode == AP_MODE_FAILSAFE && autopilot_detect_ground) {
147  autopilot_set_mode(AP_MODE_KILL);
148  autopilot_detect_ground = FALSE;
149  }
150 #endif
151 
152  /* set failsafe commands, if in FAILSAFE or KILL mode */
153 #if !FAILSAFE_GROUND_DETECT
154  if (autopilot_mode == AP_MODE_KILL ||
155  autopilot_mode == AP_MODE_FAILSAFE) {
156 #else
157  if (autopilot_mode == AP_MODE_KILL) {
158 #endif
159  SetCommands(commands_failsafe);
160  }
161  else {
162  guidance_v_run( autopilot_in_flight );
163  guidance_h_run( autopilot_in_flight );
164  SetRotorcraftCommands(stabilization_cmd, autopilot_in_flight, autopilot_motors_on);
165  }
166 
167  // when we dont have RC, check in flight by looking at throttle
168  if (radio_control.status != RC_OK) {
169  autopilot_check_in_flight_no_rc(autopilot_motors_on);
170  }
171 }

174 void autopilot_set_mode(uint8_t new_autopilot_mode) {
175 
176  /* force kill mode as long as AHRS is not aligned */
177  if (!ahrs_is_aligned())
178  new_autopilot_mode = AP_MODE_KILL;
179 
180  if (new_autopilot_mode != autopilot_mode) {
181  /* horizontal mode */
182  switch (new_autopilot_mode) {
183  case AP_MODE_FAILSAFE:
184 #ifndef KILL_AS_FAILSAFE
185  stabilization_attitude_set_failsafe_setpoint();
186  guidance_h_mode_changed(GUIDANCE_H_MODE_ATTITUDE);
187  break;
188 #endif
189  case AP_MODE_KILL:
190  autopilot_set_motors_on(FALSE);
191  autopilot_in_flight = FALSE;
192  autopilot_in_flight_counter = 0;
193  guidance_h_mode_changed(GUIDANCE_H_MODE_KILL);
194  break;
195  case AP_MODE_RC_DIRECT:
196  guidance_h_mode_changed(GUIDANCE_H_MODE_RC_DIRECT);
197  break;
198  case AP_MODE_RATE_DIRECT:
199  case AP_MODE_RATE_Z_HOLD:
200  guidance_h_mode_changed(GUIDANCE_H_MODE_RATE);
201  break;
202  case AP_MODE_ATTITUDE_RC_CLIMB:
203  case AP_MODE_ATTITUDE_DIRECT:
204  case AP_MODE_ATTITUDE_CLIMB:
205  case AP_MODE_ATTITUDE_Z_HOLD:
206  guidance_h_mode_changed(GUIDANCE_H_MODE_ATTITUDE);
207  break;
208  case AP_MODE_FORWARD:
209  guidance_h_mode_changed(GUIDANCE_H_MODE_FORWARD);
210  break;
211  case AP_MODE_CARE_FREE_DIRECT:
212  guidance_h_mode_changed(GUIDANCE_H_MODE_CARE_FREE);
213  break;
214  case AP_MODE_HOVER_DIRECT:
215  case AP_MODE_HOVER_CLIMB:
216  case AP_MODE_HOVER_Z_HOLD:
217  guidance_h_mode_changed(GUIDANCE_H_MODE_HOVER);
218  break;
219  case AP_MODE_NAV:
220  guidance_h_mode_changed(GUIDANCE_H_MODE_NAV);
221  break;
222  default:
223  break;
224  }
225  /* vertical mode */
226  switch (new_autopilot_mode) {
227  case AP_MODE_FAILSAFE:
228 #ifndef KILL_AS_FAILSAFE
229  guidance_v_mode_changed(GUIDANCE_V_MODE_CLIMB);
230  guidance_v_zd_sp = SPEED_BFP_OF_REAL(0.5);
231  break;
232 #endif
233  case AP_MODE_KILL:
234  guidance_v_mode_changed(GUIDANCE_V_MODE_KILL);
235  break;
236  case AP_MODE_RC_DIRECT:
237  case AP_MODE_RATE_DIRECT:
238  case AP_MODE_ATTITUDE_DIRECT:
239  case AP_MODE_HOVER_DIRECT:
240  case AP_MODE_CARE_FREE_DIRECT:
241  case AP_MODE_FORWARD:
242  guidance_v_mode_changed(GUIDANCE_V_MODE_RC_DIRECT);
243  break;
244  case AP_MODE_RATE_RC_CLIMB:
245  case AP_MODE_ATTITUDE_RC_CLIMB:
246  guidance_v_mode_changed(GUIDANCE_V_MODE_RC_CLIMB);
247  break;
248  case AP_MODE_ATTITUDE_CLIMB:
249  case AP_MODE_HOVER_CLIMB:
250  guidance_v_mode_changed(GUIDANCE_V_MODE_CLIMB);
251  break;
252  case AP_MODE_RATE_Z_HOLD:
253  case AP_MODE_ATTITUDE_Z_HOLD:
254  case AP_MODE_HOVER_Z_HOLD:
255  guidance_v_mode_changed(GUIDANCE_V_MODE_HOVER);
256  break;
257  case AP_MODE_NAV:
258  guidance_v_mode_changed(GUIDANCE_V_MODE_NAV);
259  break;
260  default:
261  break;
262  }
263  autopilot_mode = new_autopilot_mode;
264  }

309 void autopilot_on_rc_frame(void) {
310 
311  if (kill_switch_is_on())
312  autopilot_set_mode(AP_MODE_KILL);
313  else {
314  uint8_t new_autopilot_mode = 0;
315  AP_MODE_OF_PPRZ(radio_control.values[RADIO_MODE], new_autopilot_mode);
316  /* don't enter NAV mode if GPS is lost (this also prevents mode oscillations) */
317  if (!(new_autopilot_mode == AP_MODE_NAV
318 #if USE_GPS
319  && GpsIsLost()
320 #endif
321  ))
322  autopilot_set_mode(new_autopilot_mode);
323  }
324 
325  /* if not in FAILSAFE mode check motor and in_flight status, read RC */
326  if (autopilot_mode > AP_MODE_FAILSAFE) {
327 
328  /* if there are some commands that should always be set from RC, do it */
329 #ifdef SetAutoCommandsFromRC
330  SetAutoCommandsFromRC(commands, radio_control.values);
331 #endif
332 
333  /* if not in NAV_MODE set commands from the rc */
334 #ifdef SetCommandsFromRC
335  if (autopilot_mode != AP_MODE_NAV) {
336  SetCommandsFromRC(commands, radio_control.values);
337  }
338 #endif
339 
340  /* an arming sequence is used to start/stop motors */
341  autopilot_arming_check_motors_on();
342  kill_throttle = ! autopilot_motors_on;
343 
344  autopilot_check_in_flight(autopilot_motors_on);
345 
346  guidance_v_read_rc();
347  guidance_h_read_rc(autopilot_in_flight);
348  }
