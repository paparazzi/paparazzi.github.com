32 void TRIG_ISR() {
33  static uint32_t last;
34  uint32_t delta_t0_temp;
35  trigger_t0 = TRIGGER_CR;
36  delta_t0_temp = trigger_t0 - last;
37  if (MSEC_OF_CPU_TICKS(delta_t0_temp) > 10) {
38  trigger_delta_t0 = delta_t0_temp;
39  last = trigger_t0;
40  trigger_ext_valid = TRUE;
41  }
42 }

48 #if defined TRIG_EXT_PULSE_TYPE && TRIG_EXT_PULSE_TYPE == TRIG_EXT_PULSE_TYPE_RISING

52 #else
53 #error "trig_ext_hw.h: Unknown PULSE_TYPE"
54 #endif
55  trigger_ext_valid = FALSE;
