64 static inline void main_periodic_task( void ) {
65  RunOnceEvery(10,
66  {
67  DOWNLINK_SEND_BOOT(DefaultChannel, DefaultDevice, &sys_time.nb_sec);
68  LED_PERIODIC();
69  });
70 
71  switch(ms2100_status) {
72  case MS2100_IDLE:
73  Ms2100SendReq();
74  break;
75  case MS2100_WAITING_EOC:
76  if (Ms2100HasEOC()) {
77  Ms2100ReadRes();
78  }
79  break;
80  }
81 }

83 static inline void main_event_task( void ) {
84  if (ms2100_status == MS2100_DATA_AVAILABLE) {
85  RunOnceEvery(10, {
86  DOWNLINK_SEND_IMU_MAG_RAW(DefaultChannel, DefaultDevice,
87  &ms2100_values[0],
88  &ms2100_values[1],
89  &ms2100_values[2]);
90  });
91  ms2100_status = MS2100_IDLE;
92  }
