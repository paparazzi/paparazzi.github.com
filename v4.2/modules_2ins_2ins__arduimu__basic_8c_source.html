107 void ArduIMU_periodicGPS( void ) {
108 
109  if (ardu_gps_trans.status != I2CTransDone) { return; }
110 
111 #if USE_HIGH_ACCEL_FLAG
112  // Test for high acceleration:
113  // - low speed
114  // - high thrust
115  if (estimator_hspeed_mod < HIGH_ACCEL_LOW_SPEED && ap_state->commands[COMMAND_THROTTLE] > HIGH_ACCEL_HIGH_THRUST && !high_accel_done) {
116  high_accel_flag = TRUE;
117  } else {
118  high_accel_flag = FALSE;
119  if (estimator_hspeed_mod > HIGH_ACCEL_LOW_SPEED && !high_accel_done) {
120  high_accel_done = TRUE; // After takeoff, don't use high accel before landing (GS small, Throttle small)
121  }
122  if (estimator_hspeed_mod < HIGH_ACCEL_HIGH_THRUST_RESUME && ap_state->commands[COMMAND_THROTTLE] < HIGH_ACCEL_HIGH_THRUST_RESUME) {
123  high_accel_done = FALSE; // Activate high accel after landing
124  }
125  }
126 #endif
127 
128  FillBufWith32bit(ardu_gps_trans.buf, 0, (int32_t)gps.speed_3d); // speed 3D
129  FillBufWith32bit(ardu_gps_trans.buf, 4, (int32_t)gps.gspeed); // ground speed
130  FillBufWith32bit(ardu_gps_trans.buf, 8, (int32_t)gps.course); // course
131  ardu_gps_trans.buf[12] = gps.fix; // status gps fix
132  ardu_gps_trans.buf[13] = (uint8_t)arduimu_calibrate_neutrals; // calibration flag
133  ardu_gps_trans.buf[14] = (uint8_t)high_accel_flag; // high acceleration flag (disable accelerometers in the arduimu filter)
134  I2CTransmit(ARDUIMU_I2C_DEV, ardu_gps_trans, ArduIMU_SLAVE_ADDR, 15);
135 
136  // Reset calibration flag
137  if (arduimu_calibrate_neutrals) arduimu_calibrate_neutrals = FALSE;
138 }

162 void ArduIMU_event( void ) {
163  // Handle INS I2C event
164  if (ardu_ins_trans.status == I2CTransSuccess) {
165  // received data from I2C transaction
166  recievedData[0] = (ardu_ins_trans.buf[1]<<8) | ardu_ins_trans.buf[0];
167  recievedData[1] = (ardu_ins_trans.buf[3]<<8) | ardu_ins_trans.buf[2];
168  recievedData[2] = (ardu_ins_trans.buf[5]<<8) | ardu_ins_trans.buf[4];
169  recievedData[3] = (ardu_ins_trans.buf[7]<<8) | ardu_ins_trans.buf[6];
170  recievedData[4] = (ardu_ins_trans.buf[9]<<8) | ardu_ins_trans.buf[8];
171  recievedData[5] = (ardu_ins_trans.buf[11]<<8) | ardu_ins_trans.buf[10];
172  recievedData[6] = (ardu_ins_trans.buf[13]<<8) | ardu_ins_trans.buf[12];
173  recievedData[7] = (ardu_ins_trans.buf[15]<<8) | ardu_ins_trans.buf[14];
174  recievedData[8] = (ardu_ins_trans.buf[17]<<8) | ardu_ins_trans.buf[16];
175 
176  // Update ArduIMU data
177  arduimu_eulers.phi = ANGLE_FLOAT_OF_BFP(recievedData[0]);
178  arduimu_eulers.theta = ANGLE_FLOAT_OF_BFP(recievedData[1]);
179  arduimu_eulers.psi = ANGLE_FLOAT_OF_BFP(recievedData[2]);
180  arduimu_rates.p = RATE_FLOAT_OF_BFP(recievedData[3]);
181  arduimu_rates.q = RATE_FLOAT_OF_BFP(recievedData[4]);
182  arduimu_rates.r = RATE_FLOAT_OF_BFP(recievedData[5]);
183  arduimu_accel.x = ACCEL_FLOAT_OF_BFP(recievedData[6]);
184  arduimu_accel.y = ACCEL_FLOAT_OF_BFP(recievedData[7]);
185  arduimu_accel.z = ACCEL_FLOAT_OF_BFP(recievedData[8]);
186 
187  // Update estimator
188  estimator_phi = arduimu_eulers.phi - ins_roll_neutral;
189  estimator_theta = arduimu_eulers.theta - ins_pitch_neutral;
190  estimator_p = arduimu_rates.p;
191  estimator_q = arduimu_rates.q;
192  estimator_r = arduimu_rates.r;
193  ardu_ins_trans.status = I2CTransDone;
194 
195 #ifdef ARDUIMU_SYNC_SEND
196  //RunOnceEvery(15, DOWNLINK_SEND_AHRS_EULER(DefaultChannel, DefaultDevice, &arduimu_eulers.phi, &arduimu_eulers.theta, &arduimu_eulers.psi));
197  RunOnceEvery(15, DOWNLINK_SEND_IMU_GYRO(DefaultChannel, DefaultDevice, &arduimu_rates.p, &arduimu_rates.q, &arduimu_rates.r));
198  RunOnceEvery(15, DOWNLINK_SEND_IMU_ACCEL(DefaultChannel, DefaultDevice, &arduimu_accel.x, &arduimu_accel.y, &arduimu_accel.z));
199 #endif
200  }
201  else if (ardu_ins_trans.status == I2CTransFailed) {
202  ardu_ins_trans.status = I2CTransDone;
203  }
204  // Handle GPS I2C event
205  if (ardu_gps_trans.status == I2CTransSuccess || ardu_gps_trans.status == I2CTransFailed) {
206  ardu_gps_trans.status = I2CTransDone;
207  }
208 }
