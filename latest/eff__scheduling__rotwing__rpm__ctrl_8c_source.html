171// for negative values, still should be low_lim < up_lim
172inline float bound_or_zero(float value, float low_lim, float up_lim) {
173 float output = value;
174 if (low_lim > 0.f) {
175 if (value < low_lim) {
176 output = 0.f;
177 } else if (value > up_lim) {
178 output = up_lim;
179 }
180 } else {
181 if (value > up_lim) {
182 output = 0.f;
183 } else if (value < low_lim) {
184 output = low_lim;
185 }
186 }
187 return output;
188}

280void eff_scheduling_rotwing_update_wing_angle(void)
281{
282 // Calculate sin and cosines of rotation
283 eff_sched_var.wing_rotation_deg = eff_sched_var.wing_rotation_rad / M_PI * 180.;
284
285 eff_sched_var.cosr = cosf(eff_sched_var.wing_rotation_rad);
286 eff_sched_var.sinr = sinf(eff_sched_var.wing_rotation_rad);
287
288 eff_sched_var.cosr2 = eff_sched_var.cosr * eff_sched_var.cosr;
289 eff_sched_var.sinr2 = eff_sched_var.sinr * eff_sched_var.sinr;
290
291 eff_sched_var.sinr3 = eff_sched_var.sinr2 * eff_sched_var.sinr;
292
293}

296{
297 eff_sched_var.Ixx = eff_sched_p.Ixx_body + eff_sched_var.cosr2 * eff_sched_p.Ixx_wing + eff_sched_var.sinr2 * eff_sched_p.Iyy_wing;
298 eff_sched_var.Iyy = eff_sched_p.Iyy_body + eff_sched_var.sinr2 * eff_sched_p.Ixx_wing + eff_sched_var.cosr2 * eff_sched_p.Iyy_wing;
299
300 // Bound inertia
301 Bound(eff_sched_var.Ixx, 0.01, 100.);
302 Bound(eff_sched_var.Iyy, 0.01, 100.);
303}

322void eff_scheduling_rotwing_update_hover_motor_effectiveness(void)
323{
324 // Scaling so CAN_cmd squared fits into MAX_PPRZ
325 float scaling = eff_sched_p.hover_max_rpm_squared / MAX_PPRZ;
326
327 // Calculate the change in moment per change in RPM squared
328 float dMdu_pitch = ROTWING_EFF_SCHED_PITCH_ARM * eff_sched_p.CT_hover * scaling;
329 float dMdu_roll = ROTWING_EFF_SCHED_ROLL_ARM * eff_sched_p.CT_hover * scaling;
330
331 float front_motor_pitch_eff = dMdu_pitch / eff_sched_var.Iyy;
332 float back_motor_pitch_eff = -dMdu_pitch / eff_sched_var.Iyy;
333
334 // pitch motor has roll effectiveness due to z offset from c.g. and tilt
335 static const float roll_eff_pitch_motor_scaling = ROTWING_EFF_SCHED_ROLL_ARM_PITCH_MOTORS / ROTWING_EFF_SCHED_ROLL_ARM * sinf(RadOfDeg(ROTWING_PITCH_MOTOR_TILT_ANGLE_DEG));
336
337 // Update front motor roll and pitch effectiveness
338 g1g2[RW_aq][COMMAND_MOTOR_FRONT] = front_motor_pitch_eff; // pitch effectiveness front motor
339 g1g2[RW_ap][COMMAND_MOTOR_FRONT] = dMdu_pitch * roll_eff_pitch_motor_scaling / eff_sched_var.Ixx; // Roll effectiveness front motor
340
341 // Update back motor roll and pitch effectiveness
342 g1g2[RW_aq][COMMAND_MOTOR_BACK] = back_motor_pitch_eff; // pitch effectiveness back motor
343 g1g2[RW_ap][COMMAND_MOTOR_BACK] = -dMdu_pitch * roll_eff_pitch_motor_scaling / eff_sched_var.Ixx; // Roll effectiveness back motor
344
345 // Update right motor roll and pitch effectiveness
346 float right_motor_roll_eff = -(dMdu_roll * eff_sched_var.cosr) / eff_sched_var.Ixx;
347 right_motor_roll_eff = bound_or_zero(right_motor_roll_eff, -1.f, -0.00001f);
348
349 float right_motor_pitch_eff = (dMdu_roll * eff_sched_var.sinr) / eff_sched_var.Iyy;
350 right_motor_pitch_eff = bound_or_zero(right_motor_pitch_eff, 0.00001f, front_motor_pitch_eff);
351
352 g1g2[RW_ap][COMMAND_MOTOR_RIGHT] = right_motor_roll_eff; // roll effectiveness right motor
353 g1g2[RW_aq][COMMAND_MOTOR_RIGHT] = right_motor_pitch_eff; // pitch effectiveness right motor
354
355 // Update left motor roll and pitch effectiveness
356 float left_motor_roll_eff = (dMdu_roll * eff_sched_var.cosr) / eff_sched_var.Ixx;
357 left_motor_roll_eff = bound_or_zero(left_motor_roll_eff, 0.00001f, 1.f);
358
359 float left_motor_pitch_eff = -(dMdu_roll * eff_sched_var.sinr) / eff_sched_var.Iyy;
360 left_motor_pitch_eff = bound_or_zero(left_motor_pitch_eff, back_motor_pitch_eff, -0.00001f);
361
362 g1g2[RW_ap][COMMAND_MOTOR_LEFT] = left_motor_roll_eff; // roll effectiveness left motor
363 g1g2[RW_aq][COMMAND_MOTOR_LEFT] = left_motor_pitch_eff; // pitch effectiveness left motor
364}

366void eff_scheduling_rotwing_update_elevator_effectiveness(void)
367{
368 float de = eff_sched_p.k_elevator_deflection[0] + eff_sched_p.k_elevator_deflection[1] * eff_sched_var.cmd_elevator;
369
370 float dMyde = (eff_sched_p.k_elevator[0] * de * eff_sched_var.airspeed2 +
371 eff_sched_p.k_elevator[1] * eff_sched_var.cmd_pusher_scaled * eff_sched_var.cmd_pusher_scaled * eff_sched_var.airspeed +
372 eff_sched_p.k_elevator[2] * eff_sched_var.airspeed2) / 10000.;
373
374 // scale the effectiveness of the elevator down if it has a large deflection to encourage it to become flat quickly (pragmatic, not physically inpsired)
375 float elevator_ineffectiveness_scaling = (50-de)/40;
376 Bound(elevator_ineffectiveness_scaling, 0.5, 1.0);
377
378 float dMydpprz = dMyde * eff_sched_p.k_elevator_deflection[1];
379
380 // Convert moment to effectiveness
381 float eff_y_elev = dMydpprz / eff_sched_var.Iyy;
382
383 Bound(eff_y_elev, 0.00001, 0.1);
384
385 g1g2[RW_aq][COMMAND_ELEVATOR] = eff_y_elev;
386}

388void eff_scheduling_rotwing_update_rudder_effectiveness(void)
389{
390 float dMzdr = (eff_sched_p.k_rudder[0] * eff_sched_var.cmd_pusher_scaled * eff_sched_var.cmd_T_mean_scaled +
391 eff_sched_p.k_rudder[1] * eff_sched_var.cmd_T_mean_scaled * eff_sched_var.airspeed2 * eff_sched_var.cosr +
392 eff_sched_p.k_rudder[2] * eff_sched_var.airspeed2) / 10000.;
393
394 // Convert moment to effectiveness
395
396 float dMzdpprz = dMzdr * eff_sched_p.d_rudder_d_pprz;
397
398 // Convert moment to effectiveness
399 float eff_z_rudder = dMzdpprz / eff_sched_p.Izz;
400
401 Bound(eff_z_rudder, 0.000001, 0.1);
402
403 g1g2[RW_ar][COMMAND_RUDDER] = eff_z_rudder;
404}

440void eff_scheduling_rotwing_schedule_liftd(void)
441{
442 float lift_d_wing = (eff_sched_p.k_lift_wing[0] + eff_sched_p.k_lift_wing[1] * eff_sched_var.sinr2) * eff_sched_var.airspeed2 / eff_sched_p.m;
443 float lift_d_fuselage = eff_sched_p.k_lift_fuselage * eff_sched_var.airspeed2 / eff_sched_p.m;
444 float lift_d_tail = eff_sched_p.k_lift_tail * eff_sched_var.airspeed2 / eff_sched_p.m;
445
446 float lift_d = lift_d_wing + lift_d_fuselage + lift_d_tail;
447 if (eff_sched_var.wing_rotation_deg < 60.) {
448 lift_d = 0.0;
449 }
450 Bound(lift_d, -130., 0.);
451 eff_scheduling_rotwing_lift_d = lift_d;
452}

459void stabilization_indi_set_wls_settings(void)
460{
461 int32_t flap_saturation_limit;
462
463 // Calculate the min and max increments
464 for (uint8_t i = 0; i < INDI_NUM_ACT; i++) {
465 wls_stab_p.u_min[i] = -MAX_PPRZ * act_is_servo[i];
466 wls_stab_p.u_max[i] = MAX_PPRZ;
467 wls_stab_p.u_pref[i] = act_pref[i];
468
469 switch(i) {
470 // With RPM control we need to limit the max command to the hover motors based on the battery voltage
471 case COMMAND_MOTOR_FRONT:
472 wls_stab_p.u_max[i] = eff_sched_p.hover_rpm_lim_coef[0] + electrical.vsupply * eff_sched_p.hover_rpm_lim_coef[1];
473 wls_stab_p.u_min[i] = (int16_t)((float)SERVO_MOTOR_FRONT_NEUTRAL * ((float)SERVO_MOTOR_FRONT_NEUTRAL * ((float)MAX_PPRZ / (float)SERVO_MOTOR_FRONT_MAX) / (float)SERVO_MOTOR_FRONT_MAX));
474 break;
475 case COMMAND_MOTOR_RIGHT:
476 wls_stab_p.u_max[i] = eff_sched_p.hover_rpm_lim_coef[0] + electrical.vsupply * eff_sched_p.hover_rpm_lim_coef[1];
477 wls_stab_p.u_min[i] = (int16_t)((float)SERVO_MOTOR_RIGHT_NEUTRAL * ((float)SERVO_MOTOR_RIGHT_NEUTRAL * ((float)MAX_PPRZ / (float)SERVO_MOTOR_RIGHT_MAX) / (float)SERVO_MOTOR_RIGHT_MAX));
478 break;
479 case COMMAND_MOTOR_BACK:
480 wls_stab_p.u_max[i] = eff_sched_p.hover_rpm_lim_coef[0] + electrical.vsupply * eff_sched_p.hover_rpm_lim_coef[1];
481 wls_stab_p.u_min[i] = (int16_t)((float)SERVO_MOTOR_BACK_NEUTRAL * ((float)SERVO_MOTOR_BACK_NEUTRAL * ((float)MAX_PPRZ / (float)SERVO_MOTOR_BACK_MAX) / (float)SERVO_MOTOR_BACK_MAX));
482 break;
483 case COMMAND_MOTOR_LEFT:
484 wls_stab_p.u_max[i] = eff_sched_p.hover_rpm_lim_coef[0] + electrical.vsupply * eff_sched_p.hover_rpm_lim_coef[1];
485 wls_stab_p.u_min[i] = (int16_t)((float)SERVO_MOTOR_LEFT_NEUTRAL * ((float)SERVO_MOTOR_LEFT_NEUTRAL * ((float)MAX_PPRZ / (float)SERVO_MOTOR_LEFT_MAX) / (float)SERVO_MOTOR_LEFT_MAX));
486 break;
487 case COMMAND_ELEVATOR:
488 wls_stab_p.u_pref[i] = actuator_state_filt_vect[i]; // Set change in prefered state to 0 for elevator
489 wls_stab_p.u_min[i] = 0; // cmd 0 is lowest position for elevator
490 break;
491 case COMMAND_FLAPS:
492 // If an offset is used, limit the max differential command to prevent unilateral saturation.
493 flap_saturation_limit = MAX_PPRZ - abs(rw_flap_offset);
494 wls_stab_p.u_min[i] = -flap_saturation_limit;
495 wls_stab_p.u_max[i] = flap_saturation_limit;
496 break;
497 case COMMAND_THRUST_X:
498 // dt (min to max) MAX_PPRZ / (dt * f) dt_min == 0.002
499 Bound(eff_sched_pusher_time, 0.002, 5.);
500 float max_increment = MAX_PPRZ / (eff_sched_pusher_time * 500);
501 wls_stab_p.u_min[i] = actuators_pprz[i] - max_increment;
502 wls_stab_p.u_max[i] = actuators_pprz[i] + max_increment;
503
504 // With RPM control we need to limit the max command to the pusher motor based on the battery voltage
505 wls_stab_p.u_min[i] = Max((int16_t)((float)SERVO_MOTOR_PUSH_NEUTRAL * ((float)SERVO_MOTOR_PUSH_NEUTRAL * ((float)MAX_PPRZ / (float)SERVO_MOTOR_PUSH_MAX) / (float)SERVO_MOTOR_PUSH_MAX)), wls_stab_p.u_min[i]);
506 wls_stab_p.u_max[i] = Min(eff_sched_p.pusher_rpm_lim_coef[0] + electrical.vsupply * eff_sched_p.pusher_rpm_lim_coef[1], wls_stab_p.u_max[i]);
507 break;
508 default:
509 break;
510 }
511
512 Bound(wls_stab_p.u_min[i], -MAX_PPRZ*act_is_servo[i], MAX_PPRZ);
513 Bound(wls_stab_p.u_max[i], -MAX_PPRZ*act_is_servo[i], MAX_PPRZ);
514 }

517void guidance_indi_hybrid_set_wls_settings(float body_v[3], float roll_angle, float pitch_angle)
518{
519 // adjust weights
520 float fixed_wing_percentile = (rotwing_state_hover_motors_idling())? 1:0; // TODO: when hover props go below 40%, ...
521 Bound(fixed_wing_percentile, 0, 1);
522#define AIRSPEED_IMPORTANCE_IN_FORWARD_WEIGHT 16
523
524 float Wv_original[GUIDANCE_INDI_HYBRID_V] = GUIDANCE_INDI_WLS_PRIORITIES;
525
526 // Increase importance of forward acceleration in forward flight

528 AIRSPEED_IMPORTANCE_IN_FORWARD_WEIGHT); // stall n low hover motor_off (weight 16x more important than vertical weight)
529
530 struct FloatEulers eulers_zxy;
531 float_eulers_of_quat_zxy(&eulers_zxy, stateGetNedToBodyQuat_f());

537 Bound(du_min_thrust_z, -50., 0.);
538 float du_max_thrust_z = -(actuator_state_filt_vect[COMMAND_MOTOR_FRONT] * g1g2[RW_aZ][COMMAND_MOTOR_FRONT]
539 + actuator_state_filt_vect[COMMAND_MOTOR_RIGHT] * g1g2[RW_aZ][COMMAND_MOTOR_RIGHT]
540 + actuator_state_filt_vect[COMMAND_MOTOR_BACK] * g1g2[RW_aZ][COMMAND_MOTOR_BACK]
541 + actuator_state_filt_vect[COMMAND_MOTOR_LEFT] * g1g2[RW_aZ][COMMAND_MOTOR_LEFT]);
542 Bound(du_max_thrust_z, 0., 50.);
543
544 float roll_limit_rad = guidance_indi_max_bank;
545 float max_pitch_limit_rad = RadOfDeg(GUIDANCE_INDI_MAX_PITCH);
546 float min_pitch_limit_rad = RadOfDeg(GUIDANCE_INDI_MIN_PITCH);
547
548 float fwd_pitch_limit_rad = RadOfDeg(GUIDANCE_INDI_MAX_PITCH);
549 float quad_pitch_limit_rad = RadOfDeg(5.0);
550
551 float airspeed = stateGetAirspeed_f();
552
553 float scheduled_pitch_angle = 0.f;
554 float pitch_angle_range = 3.;
555 float meas_skew_angle = rotwing_state.meas_skew_angle_deg;
556 Bound(meas_skew_angle, 0, 90); // Bound to prevent errors
557 if (meas_skew_angle < ROTWING_SKEW_ANGLE_STEP) {
558 scheduled_pitch_angle = ROTWING_QUAD_PREF_PITCH;
559 wls_guid_p.Wu[1] = Wu_gih_original[1];
560 max_pitch_limit_rad = quad_pitch_limit_rad;
561 } else {
562 float pitch_progression = (airspeed - rotwing_state.fw_min_airspeed) / 2.f;
563 Bound(pitch_progression, 0.f, 1.f);
564 scheduled_pitch_angle = pitch_angle_range * pitch_progression + ROTWING_QUAD_PREF_PITCH*(1.f-pitch_progression);
565 wls_guid_p.Wu[1] = Wu_gih_original[1] * (1.f - pitch_progression*0.99);
566 max_pitch_limit_rad = quad_pitch_limit_rad + (fwd_pitch_limit_rad - quad_pitch_limit_rad) * pitch_progression;
567 }
568 if (!rotwing_state_hover_motors_running()) {
569 scheduled_pitch_angle = 8.;
570 max_pitch_limit_rad = fwd_pitch_limit_rad;
571 }
572 Bound(scheduled_pitch_angle, -5., 8.);
573 guidance_indi_pitch_pref_deg = scheduled_pitch_angle;
574
575 float pitch_pref_rad = RadOfDeg(guidance_indi_pitch_pref_deg);
576
577 // Set lower limits
578 wls_guid_p.u_min[0] = -roll_limit_rad - roll_angle; //roll
579 wls_guid_p.u_min[1] = min_pitch_limit_rad - pitch_angle; // pitch
580
581 // Set upper limits limits
582 wls_guid_p.u_max[0] = roll_limit_rad - roll_angle; //roll
583 wls_guid_p.u_max[1] = max_pitch_limit_rad - pitch_angle; // pitch
584
585 if(rotwing_state_hover_motors_running()) {
586 wls_guid_p.u_min[2] = du_min_thrust_z;
587 wls_guid_p.u_max[2] = du_max_thrust_z;
588 } else {
589 wls_guid_p.u_min[2] = 0.;
590 wls_guid_p.u_max[2] = 0.;
591 }
592
593 if(rotwing_state_pusher_motor_running()) {
594 wls_guid_p.u_min[3] = (-actuator_state_filt_vect[COMMAND_THRUST_X] * g1g2[RW_aX][COMMAND_THRUST_X]);
595 wls_guid_p.u_max[3] = 9.0; // Hacky value to prevent drone from pitching down in transition
596 } else {
597 wls_guid_p.u_min[3] = 0.;
598 wls_guid_p.u_max[3] = 0.;
599 }
600
601 // Set prefered states
602 wls_guid_p.u_pref[0] = 0; // prefered delta roll angle
603 wls_guid_p.u_pref[1] = -pitch_angle + pitch_pref_rad;// prefered delta pitch angle
604 wls_guid_p.u_pref[2] = wls_guid_p.u_max[2]; // Low thrust better for efficiency
605 wls_guid_p.u_pref[3] = body_v[0]; // solve the body acceleration
606}

616static float u_squared_to_pprz_cmd(float u_squared, float max, float neutral, float travel_up)
617{
618 float u = (u_squared > 0 ? 1 : (u_squared < 0 ? -1 : 0)) * sqrtf(fabsf(u_squared) / (float)MAX_PPRZ * max * max);
619 float u_unbounded = (u - neutral) / travel_up;
620 return (u_unbounded < 0) ? 0 : u_unbounded;
621}
622
631static float pprz_cmd_to_u_squared(float pprz_cmd, float max, float neutral, float travel_up)
632{
633 float u_can_cmd = pprz_cmd * travel_up + neutral;
634 return (u_can_cmd * u_can_cmd) / (max * max) * (float)MAX_PPRZ;
635}
