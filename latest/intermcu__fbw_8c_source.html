89void intermcu_periodic(void)
90{
91 /* Check for interMCU loss */
92 if (intermcu.time_since_last_frame >= INTERMCU_LOST_CNT) {

99void intermcu_event(void)
100{
101 uint8_t i, c;
102
103 // Check if there are messages in the device
104 if (intermcu.device->char_available(intermcu.device->periph)) {
105 while (intermcu.device->char_available(intermcu.device->periph) && !intermcu.transport.trans_rx.msg_received) {
106 c = intermcu.device->get_byte(intermcu.device->periph);
107 parse_pprz(&intermcu.transport, c);
108 }
109
110 // If we have a message copy it
111 if (intermcu.transport.trans_rx.msg_received) {
112 for (i = 0; i < intermcu.transport.trans_rx.payload_len; i++) {
113 imcu_msg_buf[i] = intermcu.transport.trans_rx.payload[i];
114 }
115
116 intermcu.msg_available = true;
117 intermcu.transport.trans_rx.msg_received = false;
118 }
119 }
120
121 if (intermcu.msg_available) {
122 uint8_t class_id = pprzlink_get_msg_class_id(imcu_msg_buf);
123 if (class_id == DL_intermcu_CLASS_ID) {
124 // parse intermcu messages and call callbacks
125 dl_parse_msg(intermcu.device, &intermcu.transport.trans_tx, imcu_msg_buf);
126#if TELEMETRY_INTERMCU
127 } else {
128 // forward all other messages if needed
129 intermcu_dl_on_msg(imcu_msg_buf, intermcu.transport.trans_rx.payload_len);
130#endif
131 }
132 intermcu.msg_available = false;
133 }
134}

151void intermcu_parse_IMCU_COMMANDS(uint8_t *buf)
152{
153 uint8_t size = DL_IMCU_COMMANDS_values_length(buf);
154 int16_t *new_commands = DL_IMCU_COMMANDS_values(buf);
155 intermcu.cmd_status |= DL_IMCU_COMMANDS_status(buf);
156
157 // Read the autopilot status and then clear it
158 intermcu_ap_motors_on = INTERMCU_GET_CMD_STATUS(INTERMCU_CMD_MOTORS_ON);
159 INTERMCU_CLR_CMD_STATUS(INTERMCU_CMD_MOTORS_ON);
160
161 for (int i = 0; i < size; i++) {
162 intermcu_commands[i] = new_commands[i];
163 }
164
165 intermcu.status = INTERMCU_OK;
166 intermcu.time_since_last_frame = 0;
167}
