66void thermal_ctl_thermal_ctl_handler(float value) {
67 int val = (int)value;
68 uint8_t tab[2];
69 uint8_t dst_id = 0;
70 if(val == THERMAL_CTL_PWR_OFF) {
71 tab[0] = 's';
72 tab[1] = 0;
73 } else if(val == THERMAL_CTL_REC_ON) {
74 tab[0] = 'r';
75 tab[1] = 1;
76 } else if(val == THERMAL_CTL_REC_OFF) {
77 tab[0] = 'r';
78 tab[1] = 0;
79 } else {
80 return;
81 }
82
83 DOWNLINK_SEND_PAYLOAD_COMMAND(extra_pprz_tp, EXTRA_DOWNLINK_DEVICE, &dst_id, 2, tab);
84
85}

95void thermal_clusters_cb(uint8_t* buf)
96{
97 // PAYLOAD_FLOAT: cluster_key,east,north,up,latitude,longitude,median_temperature,member_count
98
99 int nb = pprzlink_get_PAYLOAD_FLOAT_values_length(buf);

106 memcpy(data, b, 8*sizeof(float));

110 .cluster_idx = data[0] + 1, // first is id 1, 0 is reserved.
111 .lat = data[4] * 1e7,
112 .lon = data[5] * 1e7,
113 .temp = data[6],
114 .count = data[7],
115 };

117 update_hottests(&spot);
118 thermal_send_draw(&spot);

121static void update_hottests(struct thermal_spot* spot) {
122 int coldest_idx = 0;
123 for (int i=0; i<NB_SPOTS; i++) {
124 struct thermal_spot *hp = &hottest_spots[i];
125
126 // cluster found in hottests, just update it.
127 if (hp->cluster_idx == spot->cluster_idx) {
128 memcpy(hp, spot, sizeof(struct thermal_spot));
129 return;
130 }
131
132 // find the coldest/hottest amongst the array.
133 if ((hp->temp < hottest_spots[coldest_idx].temp && thermal_threshold > 0) ||
134 (hp->temp > hottest_spots[coldest_idx].temp && thermal_threshold < 0)) {
135 coldest_idx = i;
136 }
137 }
138
139 // the new spot it hotter than the coldest of the array, save it in place of the coldest.
140 // or colder than the hottest.
141 if ((spot->temp > hottest_spots[coldest_idx].temp && thermal_threshold > 0) ||
142 (spot->temp < hottest_spots[coldest_idx].temp && thermal_threshold < 0)) {
143 memcpy(&hottest_spots[coldest_idx], spot, sizeof(struct thermal_spot));
144 }
145}
