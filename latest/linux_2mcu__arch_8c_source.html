45static uint8_t uid_from_file(const char *path, uint8_t *uid, uint8_t max_len)
46{
47 char line[64];
48 if (path[0] == '\0') {
49 return 0;
50 }
51 FILE *f = fopen(path, "r");
52 if (f == NULL) {
53 return 0;
54 }
55 uint8_t len = 0;
56 if (fgets(line, sizeof(line), f) != NULL) {
57 line[strcspn(line, "\r\n")] = '\0';
58 len = Min(strlen(line), max_len);
59 memcpy(uid, line, len);
60 }
61 fclose(f);
62 return len;
63}

68static uint8_t uid_from_mac(uint8_t *uid, uint8_t max_len)
69{
70 DIR *dir = opendir("/sys/class/net");
71 if (dir == NULL) {
72 return 0;
73 }
74 char best_name[64] = "";
75 uint8_t best_mac[6];
76 struct dirent *ent;
77 while ((ent = readdir(dir)) != NULL) {
78 char path[300];
79 unsigned int m[6];
80 if (ent->d_name[0] == '.' || strcmp(ent->d_name, "lo") == 0 ||
81 strlen(ent->d_name) >= sizeof(best_name) ||
82 (best_name[0] != '\0' && strcmp(ent->d_name, best_name) >= 0)) {
83 continue;
84 }
85 snprintf(path, sizeof(path), "/sys/class/net/%s/address", ent->d_name);
86 FILE *f = fopen(path, "r");
87 if (f == NULL) {
88 continue;
89 }
90 int n = fscanf(f, "%x:%x:%x:%x:%x:%x", &m[0], &m[1], &m[2], &m[3], &m[4], &m[5]);
91 fclose(f);
92 if (n != 6 || (m[0] | m[1] | m[2] | m[3] | m[4] | m[5]) == 0) {
93 continue;
94 }
95 strcpy(best_name, ent->d_name);
96 for (int i = 0; i < 6; i++) {
97 best_mac[i] = m[i];
98 }
99 }
100 closedir(dir);
101 if (best_name[0] == '\0') {
102 return 0;
103 }
104 uint8_t len = Min(6, max_len);
105 memcpy(uid, best_mac, len);
106 return len;
107}

114uint8_t mcu_get_uid(uint8_t *uid, uint8_t max_len)
115{
116 uint8_t len = uid_from_file(LINUX_UID_FILE, uid, max_len);
117 if (len == 0) {
118 len = uid_from_mac(uid, max_len);
119 }
120 return len;
121}
