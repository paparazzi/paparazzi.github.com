214static void send_accel_raw(struct transport_tx *trans, struct link_device *dev)
215{
216 if(imu.accel_abi_send_id == ABI_DISABLE)
217 return;
218 static uint8_t id = 0;
219 pprz_msg_send_IMU_ACCEL_RAW(trans, dev, AC_ID, &imu.accels[id].abi_id, &imu.accels[id].temperature,
220 &imu.accels[id].unscaled.x, &imu.accels[id].unscaled.y, &imu.accels[id].unscaled.z);
221 if(imu.accel_abi_send_id != ABI_BROADCAST && imu.accels[id].abi_id == imu.accel_abi_send_id)
222 return;
223 id++;
224 if(id >= IMU_MAX_SENSORS || imu.accels[id].abi_id == ABI_DISABLE)
225 id = 0;
226}

400static void cmd_imu(shell_stream_t *sh, int argc, const char *const argv[])
401{
402 (void) argv;
403 int i = 0;
404 if (argc > 0) {
405 chprintf(sh, "Usage: imu\r\n");
406 return;
407 }
408 chprintf(sh, "GYRO IMU data\r\n");
409 for (i = 0; i < IMU_MAX_SENSORS; i++) {
410 if (imu.gyros[i].abi_id != 0) {
411#if IMU_INTEGRATION
412 chprintf(sh, " Gyro id: %u, time %lu, measured rate %.2f Hz (0: not measured yet)\r\n", imu.gyros[i].abi_id, imu.gyros[i].last_stamp, imu.gyros[i].rate_est.rate);
413#else
414 chprintf(sh, " Gyro id: %u, time %lu\r\n", imu.gyros[i].abi_id, imu.gyros[i].last_stamp);
415#endif
416 show_calibrated(sh, &imu.gyros[i].calibrated);
417 show_rates(sh, "neutral", &imu.gyros[i].neutral);
418 show_ratesf(sh, "scale", &imu.gyros[i].scale_f);
419 show_matrix(sh, "body_to_sensor", &imu.gyros[i].body_to_sensor);
420 show_rates(sh, "unscaled", &imu.gyros[i].unscaled);
421 show_rates(sh, "scaled", &imu.gyros[i].scaled);
422 struct FloatRates gf;
423 RATES_FLOAT_OF_BFP(gf, imu.gyros[i].scaled);
424 chprintf(sh, " -> gyro (rad/s): %.3f, %.3f, %.3f\r\n", gf.p, gf.q, gf.r);
425 }
426 }
427 chprintf(sh, "ACCEL IMU data\r\n");
428 for (i = 0; i < IMU_MAX_SENSORS; i++) {
429 if (imu.accels[i].abi_id != 0) {
430#if IMU_INTEGRATION
431 chprintf(sh, " Accel id: %u, time %lu, measured rate %.2f Hz (0: not measured yet)\r\n", imu.accels[i].abi_id, imu.accels[i].last_stamp, imu.accels[i].rate_est.rate);
432#else
433 chprintf(sh, " Accel id: %u, time %lu\r\n", imu.accels[i].abi_id, imu.accels[i].last_stamp);
434#endif
435 show_calibrated(sh, &imu.accels[i].calibrated);
436 show_vect3(sh, "neutral", &imu.accels[i].neutral);
437 show_vect3f(sh, "scale", &imu.accels[i].scale_f);
438 show_matrix(sh, "body_to_sensor", &imu.accels[i].body_to_sensor);
439 show_vect3(sh, "unscaled", &imu.accels[i].unscaled);
440 show_vect3(sh, "scaled", &imu.accels[i].scaled);
441 struct FloatVect3 af;
442 ACCELS_FLOAT_OF_BFP(af, imu.accels[i].scaled);
443 chprintf(sh, " -> accel (m/s2): %.3f, %.3f, %.3f\r\n", af.x, af.y, af.z);
444 }
445 }
446 chprintf(sh, "MAG IMU data\r\n");
447 for (i = 0; i < IMU_MAX_SENSORS; i++) {
448 if (imu.mags[i].abi_id != 0) {
449 chprintf(sh, " Mag id: %u\r\n", imu.mags[i].abi_id);
450 show_calibrated(sh, &imu.mags[i].calibrated);
451 show_vect3(sh, "neutral", &imu.mags[i].neutral);
452 show_vect3f(sh, "scale", &imu.mags[i].scale_f);
453 show_matrix(sh, "body_to_sensor", &imu.mags[i].body_to_sensor);
454 show_vect3(sh, "unscaled", &imu.mags[i].unscaled);
455 show_vect3(sh, "scaled", &imu.mags[i].scaled);

480static bool imu_filter_init(Butterworth2LowPass filter[3], float freq, float sample_freq, const float init[3])
481{
482 if (!(freq > 0.f) || !(sample_freq > 0.f) || !isfinite(sample_freq)) {

486 float sample_time = 1.0 / sample_freq;
487 for (uint8_t j = 0; j < 3; j++) {
488 init_butterworth_2_low_pass(&filter[j], tau, sample_time, init[j]);
489 }
490 return true;
491}

543static float imu_rate_est_update(struct imu_rate_est_t *est, const char *name, uint8_t id, float ref,
544 uint32_t stamp, uint32_t last_stamp, uint8_t samples)
545{
546 if(!isfinite(ref) || !(ref > 0.f)) {
547 return ref;
548 }
549
550 // No previous timestamp or the timer overflowed
551 if(last_stamp == 0 || stamp <= last_stamp) {
552 est->start = stamp;
553 est->samples = 0;
554 return 0.f;
555 }
556
557 const float bound = IMU_RATE_EST_BOUND * ref;
558 const float max_interval = samples * (1e6f / ref) * (1.f + IMU_RATE_EST_BOUND) + IMU_RATE_EST_JITTER;
559 if((float)(stamp - last_stamp) > max_interval) {
560 // Start a new window from this callback (its samples are from before the timestamp)
561 est->start = stamp;
562 est->samples = 0;
563 } else {
564 est->samples += samples;
565 const uint32_t window = stamp - est->start;
566 if(window >= IMU_RATE_EST_WINDOW) {
567 const float window_rate = (float)est->samples * 1e6f / (float)window;
568 if(fabsf(window_rate - ref) <= bound) {
569 est->rate = (est->rate > 0.f) ? est->rate + IMU_RATE_EST_ALPHA * (window_rate - est->rate) : window_rate;
570 if(est->windows < IMU_RATE_EST_REPORT && ++est->windows == IMU_RATE_EST_REPORT) {
571 imu_rate_est_info(name, id, est->rate, ref, false);
572 }
573 } else if(!est->rejected) {
574 est->rejected = true;
575 imu_rate_est_info(name, id, window_rate, ref, true);
576 }
577 est->start = stamp;
578 est->samples = 0;
579 }
580 }
581
582 return (est->rate > 0.f && fabsf(est->rate - ref) <= bound) ? est->rate : ref;
583}

596static inline void imu_integrate_sample(struct FloatVect3 *alpha, struct FloatVect3 *beta, struct FloatVect3 *last_sample,
597 struct FloatVect3 *last_delta, const struct FloatVect3 *sample, float dt)
598{
599 struct FloatVect3 delta;
600 VECT3_SUM(delta, *last_sample, *sample);
601 VECT3_SMUL(delta, delta, 0.5f * dt);
602 if(beta != NULL) {
603 struct FloatVect3 coning_arm, coning;
604 VECT3_SUM_SCALED(coning_arm, *alpha, *last_delta, 1.f / 6.f);
605 VECT3_CROSS_PRODUCT(coning, coning_arm, delta);
606 VECT3_ADD_SCALED(*beta, coning, 0.5f);
607 *last_delta = delta;
608 }
609 VECT3_ADD(*alpha, delta);
610 *last_sample = *sample;
611}

921static void imu_gyro_raw_cb(uint8_t sender_id, uint32_t stamp, struct Int32Rates *data, uint8_t samples, float rate, float temp)
922{
923 // Find the correct gyro
924 struct imu_gyro_t *gyro = imu_get_gyro(sender_id, true);
925 if(gyro == NULL || samples < 1)
926 return;
927
928 // Reference rate for the filter and the measured integration rate:
929 // the calibrated sample frequency (e.g. measured from a log) instead of the nominal one of the driver
930 if(gyro->sample_freq > 0.f) {
931 rate = gyro->sample_freq;
932 }
933
934 // Initialize the filter on the first sample, it is disabled without a valid cutoff and sample frequency
935 if(gyro->calibrated.filter && gyro->last_stamp == 0) {
936 const float init[3] = {data[0].p, data[0].q, data[0].r};
937 gyro->calibrated.filter = imu_filter_init(gyro->filter, gyro->filter_freq, rate, init);
938 }
939
940 // Filter the gyro (data points to the filtered samples after the block, only allocate them if needed)
941 struct Int32Rates data_filtered[gyro->calibrated.filter ? samples : 1];
942 if(gyro->calibrated.filter) {
943 for(uint8_t i = 0; i < samples; i++) {
944 data_filtered[i].p = update_butterworth_2_low_pass(&gyro->filter[0], data[i].p);
945 data_filtered[i].q = update_butterworth_2_low_pass(&gyro->filter[1], data[i].q);
946 data_filtered[i].r = update_butterworth_2_low_pass(&gyro->filter[2], data[i].r);
947 }
948 data = data_filtered;
949 }
950
951 // Copy last sample as unscaled
952 RATES_COPY(gyro->unscaled, data[samples-1]);
953
954 // Scale the gyro
955 struct Int32Rates scaled, scaled_rot;
956 scaled.p = (gyro->unscaled.p - gyro->neutral.p) * gyro->scale_f.p;
957 scaled.q = (gyro->unscaled.q - gyro->neutral.q) * gyro->scale_f.q;
958 scaled.r = (gyro->unscaled.r - gyro->neutral.r) * gyro->scale_f.r;
959
960 // Rotate the sensor
961 int32_rmat_transp_ratemult(&scaled_rot, &gyro->body_to_sensor, &scaled);
962
963#if IMU_INTEGRATION
964 // Only integrate if we have gotten a previous measurement and didn't overflow the timer (rate 0)
965 // and the rate is valid (positive and finite, NaN if unknown)
966 // The measured sample rate is used if it is within IMU_RATE_EST_BOUND of the calibrated or driver rate
967 const float int_rate = imu_rate_est_update(&gyro->rate_est, "gyro", sender_id, rate, stamp, gyro->last_stamp, samples);
