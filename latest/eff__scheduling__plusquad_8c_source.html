113void init_RW_Model(void)
114{
115 // Inertia and mass
116 RW.I.xx = PLUSQUAD_EFF_SCHED_IXX; // [kgm²]
117 RW.I.yy = PLUSQUAD_EFF_SCHED_IYY; // [kgm²]
118 RW.I.zz = PLUSQUAD_EFF_SCHED_IZZ; // [kgm²]
119 RW.m = PLUSQUAD_EFF_SCHED_M; // [kg]
120
121 // Init the thrust curves
122 init_all_thrust_curve();
123 // Motor Front
124 RW.mF.dFdu = PLUSQUAD_EFF_SCHED_MF_dFdu / RW_G_SCALE;
125 RW.mF.dMdu = PLUSQUAD_EFF_SCHED_MF_dMdu / RW_G_SCALE;
126 RW.mF.dMdud = PLUSQUAD_EFF_SCHED_MF_dMdud / (RW_G_SCALE * RW_G_SCALE);
127 RW.mF.l = PLUSQUAD_EFF_SCHED_MF_l;
128
129 // Motor Right
130 RW.mR.dFdu = PLUSQUAD_EFF_SCHED_MR_dFdu / RW_G_SCALE;
131 RW.mR.dMdu = PLUSQUAD_EFF_SCHED_MR_dMdu / RW_G_SCALE;
132 RW.mR.dMdud = PLUSQUAD_EFF_SCHED_MR_dMdud / (RW_G_SCALE * RW_G_SCALE);
133 RW.mR.l = PLUSQUAD_EFF_SCHED_MR_l;
134
135 // Motor Back
136 RW.mB.dFdu = PLUSQUAD_EFF_SCHED_MB_dFdu / RW_G_SCALE;
137 RW.mB.dMdu = PLUSQUAD_EFF_SCHED_MB_dMdu / RW_G_SCALE;
138 RW.mB.dMdud = PLUSQUAD_EFF_SCHED_MB_dMdud / (RW_G_SCALE * RW_G_SCALE);
139 RW.mB.l = PLUSQUAD_EFF_SCHED_MB_l;
140
141 // Motor Left
142 RW.mL.dFdu = PLUSQUAD_EFF_SCHED_ML_dFdu / RW_G_SCALE;
143 RW.mL.dMdu = PLUSQUAD_EFF_SCHED_ML_dMdu / RW_G_SCALE;
144 RW.mL.dMdud = PLUSQUAD_EFF_SCHED_ML_dMdud / (RW_G_SCALE * RW_G_SCALE);
145 RW.mL.l = PLUSQUAD_EFF_SCHED_ML_l;
146
147 // Initialize attitude
148 RW.att.phi = 0.0;
149 RW.att.theta = 0.0;
150 RW.att.psi = 0.0;
151 RW.att.sphi = 0.0;
152 RW.att.cphi = 0.0;
153 RW.att.stheta = 0.0;
154 RW.att.ctheta = 0.0;
155 RW.att.spsi = 0.0;
156 RW.att.cpsi = 0.0;
157
158 cutoff_prev = oneloop_nB_filt_cutoff;
159}

161/*Update the attitude*/
162void update_attitude(void)
163{
164 if(cutoff_prev != oneloop_nB_filt_cutoff){
165 float tau_att = 1.0 / (2.0 * M_PI * oneloop_nB_filt_cutoff);
166 float sample_time = 1.0 / PERIODIC_FREQUENCY;
167 init_butterworth_2_low_pass(&phi_filt, tau_att, sample_time, RW.att.phi);
168 init_butterworth_2_low_pass(&theta_filt, tau_att, sample_time, RW.att.theta);
169 init_butterworth_2_low_pass(&psi_filt, tau_att, sample_time, RW.att.psi);
170 cutoff_prev = oneloop_nB_filt_cutoff;
171 }
172 float_eulers_of_quat_zxy(&eulers_zxy_RW_EFF, stateGetNedToBodyQuat_f());
173 update_butterworth_2_low_pass(&phi_filt, eulers_zxy_RW_EFF.phi);
174 update_butterworth_2_low_pass(&theta_filt, eulers_zxy_RW_EFF.theta);
175 update_butterworth_2_low_pass(&psi_filt, eulers_zxy_RW_EFF.psi);
176 RW.att.phi = phi_filt.o[0];//eulers_zxy_RW_EFF.phi;
177 RW.att.theta = theta_filt.o[0];//eulers_zxy_RW_EFF.theta;
178 RW.att.psi = psi_filt.o[0];//eulers_zxy_RW_EFF.psi;
179 //printf("Attitude filt: %f %f %f\n", RW.att.phi, RW.att.theta, RW.att.psi);
180 RW.att.sphi = sinf(RW.att.phi);
181 RW.att.cphi = cosf(RW.att.phi);
182 RW.att.stheta = sinf(RW.att.theta);
183 RW.att.ctheta = cosf(RW.att.theta);
184 RW.att.spsi = sinf(RW.att.psi);
185 RW.att.cpsi = cosf(RW.att.psi);
186}
187/* Function to precalculate once some constant effectiveness values to improve efficiency*/
188void calc_G1_G2_RW(void)
189{
190 // Inertia
191 int x = 0;
192 int y = 1;
193 int z = 2;
194 I_inv[x][x] = 1/RW.I.xx;
195 I_inv[x][y] = 0.0;
196 I_inv[x][z] = 0.0;
197 I_inv[y][x] = 0.0;
198 I_inv[y][y] = 1/RW.I.yy;
199 I_inv[y][z] = 0.0;
200 I_inv[z][x] = 0.0;
201 I_inv[z][y] = 0.0;
202 I_inv[z][z] = 1/RW.I.zz;
203 //printf("I_inv_x: %f %f %f\n", I_inv[x][x], I_inv[x][y], I_inv[x][z]);
204 //printf("I_inv_y: %f %f %f\n", I_inv[y][x], I_inv[y][y], I_inv[y][z]);
205 //printf("I_inv_z: %f %f %f\n", I_inv[z][x], I_inv[z][y], I_inv[z][z]);
206 //printf("Control - I: %f %f %f\n", 1.0/I_inv[x][x], 1.0/I_inv[y][y], 1.0/I_inv[z][z]);
207 // Calc motor and control effectiveness
208 calc_all_thrust_curve();
209
210 // Motor Front
211 G1_RW[RW_aZ][COMMAND_MOTOR_FRONT] = -RW.mF.dFdu / RW.m;
212 G1_RW[RW_ap][COMMAND_MOTOR_FRONT] = 0.0;
213 G1_RW[RW_aq][COMMAND_MOTOR_FRONT] = (RW.mF.dFdu * RW.mF.l) * I_inv[y][y];
214 G1_RW[RW_ar][COMMAND_MOTOR_FRONT] = -RW.mF.dMdu * I_inv[z][z];
215 G2_RW[COMMAND_MOTOR_FRONT] = -G2_on*RW.mF.dMdud * I_inv[z][z] * PERIODIC_FREQUENCY;
216 // Motor Right
217 G1_RW[RW_aZ][COMMAND_MOTOR_RIGHT] = -RW.mR.dFdu / RW.m;
218 G1_RW[RW_ap][COMMAND_MOTOR_RIGHT] = -RW.mR.dFdu * RW.mR.l * I_inv[x][x];
219 G1_RW[RW_aq][COMMAND_MOTOR_RIGHT] = 0.0;
220 G1_RW[RW_ar][COMMAND_MOTOR_RIGHT] = RW.mR.dMdu * I_inv[z][z];
221 G2_RW[COMMAND_MOTOR_RIGHT] = G2_on*RW.mR.dMdud * I_inv[z][z] * PERIODIC_FREQUENCY;
222 // Motor Back
223 G1_RW[RW_aZ][COMMAND_MOTOR_BACK] = -RW.mB.dFdu / RW.m;
224 G1_RW[RW_ap][COMMAND_MOTOR_BACK] = 0.0;
225 G1_RW[RW_aq][COMMAND_MOTOR_BACK] = -(RW.mB.dFdu * RW.mB.l) * I_inv[y][y];
226 G1_RW[RW_ar][COMMAND_MOTOR_BACK] = -RW.mB.dMdu * I_inv[z][z];
227 G2_RW[COMMAND_MOTOR_BACK] = -G2_on*RW.mB.dMdud * I_inv[z][z] * PERIODIC_FREQUENCY;
228 // Motor Left
229 G1_RW[RW_aZ][COMMAND_MOTOR_LEFT] = -RW.mL.dFdu / RW.m;
230 G1_RW[RW_ap][COMMAND_MOTOR_LEFT] = RW.mL.dFdu * RW.mL.l * I_inv[x][x];
231 G1_RW[RW_aq][COMMAND_MOTOR_LEFT] = 0.0;
232 G1_RW[RW_ar][COMMAND_MOTOR_LEFT] = RW.mL.dMdu * I_inv[z][z];
233 G2_RW[COMMAND_MOTOR_LEFT] = G2_on*RW.mL.dMdud * I_inv[z][z] * PERIODIC_FREQUENCY;
234}

250void sum_EFF_MAT_RW(void) {
251
252 // Thrust force estimation
253 float T = RW.T / RW.m; // Thrust specific force. Minus gravity is a guesstimate.
254 int i = 0;
255 int j = 0;
256
257 for (i = 0; i < EFF_MAT_COLS_NB; i++) {
258 switch (i) {
259 case (COMMAND_MOTOR_FRONT):
260 case (COMMAND_MOTOR_BACK):
261 case (COMMAND_MOTOR_RIGHT):
262 case (COMMAND_MOTOR_LEFT):
263 EFF_MAT_RW[RW_aN][i] = (RW.att.cpsi * RW.att.stheta + RW.att.ctheta * RW.att.sphi * RW.att.spsi) * G1_RW[RW_aZ][i];
264 EFF_MAT_RW[RW_aE][i] = (RW.att.spsi * RW.att.stheta - RW.att.cpsi * RW.att.ctheta * RW.att.sphi) * G1_RW[RW_aZ][i];
265 EFF_MAT_RW[RW_aD][i] = (RW.att.cphi * RW.att.ctheta ) * G1_RW[RW_aZ][i];
266 EFF_MAT_RW[RW_ap][i] = (G1_RW[RW_ap][i]) ;
267 EFF_MAT_RW[RW_aq][i] = (G1_RW[RW_aq][i]) ;
268 EFF_MAT_RW[RW_ar][i] = (G1_RW[RW_ar][i] + G2_RW[i]) ;
269 break;
270 case (COMMAND_ROLL):
271 EFF_MAT_RW[RW_aN][i] = (-RW.att.cphi * RW.att.ctheta * RW.att.spsi * T);
272 EFF_MAT_RW[RW_aE][i] = ( RW.att.cphi * RW.att.ctheta * RW.att.cpsi * T);
273 EFF_MAT_RW[RW_aD][i] = ( RW.att.sphi * RW.att.ctheta * T);
274 EFF_MAT_RW[RW_ap][i] = 0.0;
275 EFF_MAT_RW[RW_aq][i] = 0.0;
276 EFF_MAT_RW[RW_ar][i] = 0.0;
277 break;
278 case (COMMAND_PITCH):
279 EFF_MAT_RW[RW_aN][i] = (-(RW.att.ctheta * RW.att.cpsi - RW.att.sphi * RW.att.stheta * RW.att.spsi) * T);
280 EFF_MAT_RW[RW_aE][i] = (-(RW.att.ctheta * RW.att.spsi + RW.att.sphi * RW.att.stheta * RW.att.cpsi) * T);
281 EFF_MAT_RW[RW_aD][i] = ( RW.att.stheta * RW.att.cphi * T) ;
282 EFF_MAT_RW[RW_ap][i] = 0.0;
283 EFF_MAT_RW[RW_aq][i] = 0.0;
284 EFF_MAT_RW[RW_ar][i] = 0.0;
285 break;
286 default:
287 break;
288 }
289 }
290 for (i = 0; i < EFF_MAT_ROWS_NB; i++) {
291 for(j = 0; j < EFF_MAT_COLS_NB; j++) {
292 float abs = fabs(EFF_MAT_RW[i][j]);
293 switch (i) {
294 case (RW_aN):
295 case (RW_aE):
296 case (RW_aD):
297 if (abs < flt_cut_a) {
298 EFF_MAT_RW[i][j] = 0.0;
299 }
300 break;
301 case (RW_aq):
302 case (RW_ar):
303 if (abs < flt_cut) {
304 EFF_MAT_RW[i][j] = 0.0;
305 }
306 break;
307 case (RW_ap):
308 if (abs < flt_cut_ap) {
309 EFF_MAT_RW[i][j] = 0.0;
310 }
311 break;
312 }
313 }
314 }
315 if(manual_roll){
316 EFF_MAT_RW[RW_ap][1] = -0.0370*roll_mult;//-0.0317895*roll_mult;
317 EFF_MAT_RW[RW_ap][3] = 0.0370*roll_mult;// 0.0317895*roll_mult;
318
319 }
320 if(manual_pitch){
321 EFF_MAT_RW[RW_aq][0] = 0.0314*pitch_mult;// 0.0274249*pitch_mult;
322 EFF_MAT_RW[RW_aq][2] = -0.0314*pitch_mult;//-0.0274249*pitch_mult;
323 }
324 if(manual_yaw){
325 EFF_MAT_RW[RW_ar][0] = -0.0022*yaw_mult;//-0.0025*yaw_mult;
326 EFF_MAT_RW[RW_ar][1] = 0.0022*yaw_mult;// 0.0025*yaw_mult;
327 EFF_MAT_RW[RW_ar][2] = -0.0022*yaw_mult;//-0.0025*yaw_mult;
328 EFF_MAT_RW[RW_ar][3] = 0.0022*yaw_mult;// 0.0025*yaw_mult;
329 //G2_RW[0] = -2.1e-5;
330 //G2_RW[1] = 2.1e-5;
331 //G2_RW[2] = -2.1e-5;
332 //G2_RW[3] = 2.1e-5;
333 }
334}

336float calc_thrust_curve(float k1, float k2, float k3, float u){
337 return k1*u*u + k2*u + k3;
338}
339
340float calc_thrust_curve_d(float k1, float k2, float u){
341 float out= 2.0*k1*u + k2;
342 Bound(out, 1.0e-3, 1.0e-2);
343 return out;
344}

379void init_all_thrust_curve(void){
380 // dTdu = 2*k1*u + k2--------|k1 | k2 | u
381 RW.mF.dFdu = calc_thrust_curve_d(PLUSQUAD_EFF_SCHED_MF_k1, PLUSQUAD_EFF_SCHED_MF_k2, 4800.0);
382 RW.mR.dFdu = calc_thrust_curve_d(PLUSQUAD_EFF_SCHED_MR_k1, PLUSQUAD_EFF_SCHED_MR_k2, 4800.0);
383 RW.mB.dFdu = calc_thrust_curve_d(PLUSQUAD_EFF_SCHED_MB_k1, PLUSQUAD_EFF_SCHED_MB_k2, 4800.0);
