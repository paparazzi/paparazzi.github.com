67bool search_map_update(struct search_map_t *map, float data, struct NedCoor_f pos)
68{
69 int x = (int)((pos.x - map->center.x) / map->res + SEARCH_MAP_SIZE / 2.f + 0.5f);
70 int y = (int)((pos.y - map->center.y) / map->res + SEARCH_MAP_SIZE / 2.f + 0.5f);
71 DEBUG_PRINT("update %f | %f %f | %d %d\n", data, pos.x, pos.y, x, y);
72 if (x < 0 || x >= SEARCH_MAP_SIZE || y < 0 || y >= SEARCH_MAP_SIZE) {
73 return false; // out of map
74 }
75 map->grid[x][y] = data; // TODO filter with current data in cell ?
76 return true;
77}

79float search_map_get_barycenter(struct search_map_t *map, float threshold, struct NedCoor_f *pos)
80{
81 float bx = 0.f;
82 float by = 0.f;
83 float quality = 0.f;
84 int nb_qual = 0;
85 float sum = 0.f;
86 const int offset = (int)(SEARCH_MAP_SIZE / 2.f + 0.5f);
87 float max = 0.f;
88#if SEARCH_MAP_DEBUG
89 float mx = 0.f;
90 float my = 0.f;
91#endif
92 for (int i = 0; i < SEARCH_MAP_SIZE; i++) {
93 for (int j = 0; j < SEARCH_MAP_SIZE; j++) {
94 if (map->grid[i][j] > max) {
95 max = map->grid[i][j];
96#if SEARCH_MAP_DEBUG
97 mx = (float)(i - offset) * map->res;
98 my = (float)(j - offset) * map->res;
99#endif
100 }
101 }
102 }
103 if (max < 1e-5) {
104 *pos = map->center;
105 DEBUG_PRINT("nothing in map\n");
106 return 0.f; // nothing in the map
107 }
108 DEBUG_PRINT("map:\n");
109 Bound(threshold, 0.f, 1.f);
110 for (int i = 0; i < SEARCH_MAP_SIZE; i++) {
111 for (int j = 0; j < SEARCH_MAP_SIZE; j++) {
112 float val = map->grid[i][j] / max; // normalized value
113 DEBUG_PRINT("\t%.1f,", val);
114 if (val >= threshold) {
115 bx += val * (float)(i - offset);
116 by += val * (float)(j - offset);
117 sum += val;
118 quality += map->grid[i][j];
119 nb_qual++;
120 }
121 }
122 DEBUG_PRINT("\n");
123 }
124 pos->x = map->center.x + (bx * map->res / sum);
125 pos->y = map->center.y + (by * map->res / sum);
126 pos->z = map->center.z;
127 quality /= (float)nb_qual;
128
129 DEBUG_PRINT(" -> offset %d, sum %f, res %f, bx %f, by %f, cx %f, cy %f, mx %f, my %f\n", offset, sum, map->res, bx, by, map->center.x, map->center.y, mx, my);
130 DEBUG_PRINT(" -> pos %.4f %.4f, quality=%.2f\n", pos->x, pos->y, quality);
131
132 return quality;
133}
