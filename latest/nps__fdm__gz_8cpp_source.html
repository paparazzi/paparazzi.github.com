167class PaparazziSystem : public gz::sim::System,
168 public gz::sim::ISystemConfigure,
169 public gz::sim::ISystemPreUpdate,
170 public gz::sim::ISystemPostUpdate
171{
172public:
173 void Configure(const gz::sim::Entity &entity,
174 const std::shared_ptr<const sdf::Element> &,
175 gz::sim::EntityComponentManager &ecm,
176 gz::sim::EventManager &) override
177 {
178 world_ = entity;
179 FindModel(ecm);
180 }
181
182 void PreUpdate(const gz::sim::UpdateInfo &info, gz::sim::EntityComponentManager &ecm) override
183 {
184 if (model_ == gz::sim::kNullEntity) {
185 FindModel(ecm);
186 return;
187 }
188 if (info.paused) {
189 return;
190 }
191 for (int i = 0; i < NPS_COMMANDS_NB; i++) {
192 gz::sim::Link link(motor_links_[i]);
193 auto pose = link.WorldPose(ecm);
194 if (!pose) {
195 continue;
196 }
197 gz::math::Vector3d force = pose->Rot().RotateVector(gz::math::Vector3d(0, 0, gz_cmd.thrusts[i]));
198 gz::math::Vector3d torque = pose->Rot().RotateVector(gz::math::Vector3d(0, 0, gz_cmd.torques[i]));
199 link.AddWorldWrench(ecm, force, torque);
200 }
201 }
202
203 void PostUpdate(const gz::sim::UpdateInfo &info, const gz::sim::EntityComponentManager &ecm) override
204 {
205 if (model_ == gz::sim::kNullEntity) {
206 return;
207 }
208 gz::sim::Link body(body_link_);
209 auto pose = body.WorldPose(ecm);
210 auto vel = body.WorldLinearVelocity(ecm);
211 auto ang_vel = body.WorldAngularVelocity(ecm);
212 if (!pose || !vel || !ang_vel) {
213 return;
214 }
215 gz_state.pose = *pose;
216 gz_state.linear_vel = *vel;
217 gz_state.angular_vel = *ang_vel;
218
219 auto gravity = ecm.Component<gz::sim::components::Gravity>(world_);
220 gz_state.gravity = gravity ? gravity->Data() : gz::math::Vector3d(0, 0, -9.81);
221 auto mag = ecm.Component<gz::sim::components::MagneticField>(world_);
222 gz_state.magnetic_field = mag ? mag->Data() : gz::math::Vector3d::Zero;
223 auto sphere = ecm.Component<gz::sim::components::SphericalCoordinates>(world_);
224 gz_state.heading_offset = sphere ? sphere->Data().HeadingOffset().Radian() : M_PI;
225
226 gz_state.sim_time = std::chrono::duration<double>(info.simTime).count();
227 gz_state.valid = true;
228 }
229
230 bool Ready() const { return model_ != gz::sim::kNullEntity; }
231
232private:
233 void FindModel(gz::sim::EntityComponentManager &ecm)
234 {
235 gz::sim::Model model(gz::sim::World(world_).ModelByName(ecm, NPS_GAZEBO_AC_NAME));
236 if (!model.Valid(ecm)) {
237 return;
238 }
239 gz::sim::Entity motor_links[NPS_COMMANDS_NB];
240 for (int i = 0; i < NPS_COMMANDS_NB; i++) {
241 motor_links[i] = model.LinkByName(ecm, gazebo_actuators.names[i]);
242 if (motor_links[i] == gz::sim::kNullEntity) {
243 fprintf(stderr, "NPS gz: link '%s' not found in model '%s'\n",
244 gazebo_actuators.names[i].c_str(), NPS_GAZEBO_AC_NAME);
245 exit(EXIT_FAILURE);
246 }
247 }
248 body_link_ = model.CanonicalLink(ecm);
249 gz::sim::Link body(body_link_);
250 body.EnableVelocityChecks(ecm, true);
251 for (int i = 0; i < NPS_COMMANDS_NB; i++) {
252 motor_links_[i] = motor_links[i];
253 }
254 model_ = model.Entity();
255 }
256
257 gz::sim::Entity world_{gz::sim::kNullEntity};
258 gz::sim::Entity model_{gz::sim::kNullEntity};
259 gz::sim::Entity body_link_{gz::sim::kNullEntity};
260 gz::sim::Entity motor_links_[NPS_COMMANDS_NB];
261};

303static struct DoubleEulers eulers_ned_of_enu(const gz::math::Quaterniond &q_enu)
304{
305 struct DoubleEulers eulers;
306 eulers.phi = q_enu.Roll();
307 eulers.theta = -q_enu.Pitch();
308 eulers.psi = M_PI_2 - q_enu.Yaw();
309 eulers.psi = atan2(sin(eulers.psi), cos(eulers.psi));
310 return eulers;
311}

315void nps_fdm_init(double dt)
316{
317 fdm.init_dt = dt;
318 fdm.curr_dt = dt;
319 fdm.nan_count = 0;
320
321#ifdef NPS_ACTUATOR_TIME_CONSTANTS
322 // Low-pass filter to simulate the delayed actuator response
323 const float tau[NPS_COMMANDS_NB] = NPS_ACTUATOR_TIME_CONSTANTS;
324 for (int i = 0; i < NPS_COMMANDS_NB; i++) {
325 init_first_order_low_pass(&gazebo_actuators.lowpass[i], tau[i], dt, 0.f);
326 }
327#ifdef NPS_ACTUATOR_MAX_ANGULAR_MOMENTUM
328 // High-pass filter to simulate the spin-up torque
329 const float Iwmax[NPS_COMMANDS_NB] = NPS_ACTUATOR_MAX_ANGULAR_MOMENTUM;
330 for (int i = 0; i < NPS_COMMANDS_NB; i++) {
331 init_first_order_high_pass(&gazebo_actuators.highpass[i], tau[i], dt, 0.f);
332 gazebo_actuators.max_ang_momentum[i] = Iwmax[i];
333 }
334#endif
335#endif
336}

352 gazebo_read_video();
353#endif
354}

371static void init_gazebo(void)
372{
373 const char *home = getenv("PAPARAZZI_HOME");
374 if (home == NULL) {
375 fprintf(stderr, "NPS gz: PAPARAZZI_HOME is not set\n");
376 exit(EXIT_FAILURE);
377 }
378 std::string pprz_home(home);
379 const std::string dirs[] = { pprz_home + "/conf/simulator/gazebo/", pprz_home + "/sw/ext/tudelft_gazebo_models/" };
380
381 std::string resource_path;
382 for (const std::string &dir : dirs) {
383 sdf::addURIPath("model://", dir + "models/");
384 sdf::addURIPath("world://", dir + "world/");
385 resource_path += dir + "models/:";
386 }
387 // models that only gz-sim needs (e.g. model://ground_plane, from the Classic model database)
388 sdf::addURIPath("model://", pprz_home + "/conf/simulator/gazebo/models_gz/");
389 resource_path += pprz_home + "/conf/simulator/gazebo/models_gz/:";
390 // model:// URIs in the merged world are resolved by gz-sim through this path
391 const char *env_path = getenv("GZ_SIM_RESOURCE_PATH");
392 if (env_path != NULL) {
393 resource_path += env_path;
394 }
395 setenv("GZ_SIM_RESOURCE_PATH", resource_path.c_str(), 1);
396
397 std::string vehicle_uri = std::string("model://") + NPS_GAZEBO_AC_NAME + "/" + NPS_GAZEBO_AC_NAME + ".sdf";
398 std::string vehicle_file = sdf::findFile(vehicle_uri, false);
399 sdf::SDFPtr vehicle_sdf(new sdf::SDF());
400 sdf::init(vehicle_sdf);
401 if (vehicle_file.empty() || !sdf::readFile(vehicle_file, vehicle_sdf)) {
402 fprintf(stderr, "NPS gz: could not load vehicle '%s'\n", vehicle_uri.c_str());
403 exit(EXIT_FAILURE);
404 }
405
406 std::string world_uri = std::string("world://") + NPS_GAZEBO_WORLD;
407 std::string world_file = sdf::findFile(world_uri, false);
408 sdf::SDFPtr world_sdf(new sdf::SDF());
409 sdf::init(world_sdf);
410 if (world_file.empty() || !sdf::readFile(world_file, world_sdf)) {
411 fprintf(stderr, "NPS gz: could not load world '%s'\n", world_uri.c_str());
412 exit(EXIT_FAILURE);
413 }
414 printf("NPS gz: vehicle %s, world %s\n", vehicle_file.c_str(), world_file.c_str());
415
416 sdf::ElementPtr world = world_sdf->Root()->GetElement("world");
417#if NPS_SIMULATE_VIDEO
418 gazebo_setup_cameras(vehicle_sdf->Root()->GetElement("model"));
419 // Cameras need the Sensors system. gz-sim only loads its default systems for
420 // a world without plugins, so add them as well.
421 if (!world->HasElement("plugin")) {
422 const char *plugins =
423 "<sdf version='1.6'><world name='plugins'>"
424 "<plugin filename='gz-sim-physics-system' name='gz::sim::systems::Physics'/>"
425 "<plugin filename='gz-sim-user-commands-system' name='gz::sim::systems::UserCommands'/>"
426 "<plugin filename='gz-sim-scene-broadcaster-system' name='gz::sim::systems::SceneBroadcaster'/>"
427 "<plugin filename='gz-sim-sensors-system' name='gz::sim::systems::Sensors'>"
428 "<render_engine>ogre2</render_engine></plugin>"
429 "</world></sdf>";
430 sdf::SDFPtr plugins_sdf(new sdf::SDF());
431 sdf::init(plugins_sdf);
432 sdf::readString(plugins, plugins_sdf);
433 for (sdf::ElementPtr plugin = plugins_sdf->Root()->GetElement("world")->GetElement("plugin"); plugin;
434 plugin = plugin->GetNextElement("plugin")) {
435 world->InsertElement(plugin->Clone());
436 }
437 } else {
438 fprintf(stderr, "NPS gz: the world has its own plugins, make sure it loads the Sensors system for the cameras\n");
439 }
440#endif
441 world->InsertElement(vehicle_sdf->Root()->GetFirstElement());
442 // One gz iteration per NPS step: use the NPS time step for the physics
443 world->GetElement("physics")->GetElement("max_step_size")->Set(fdm.init_dt);
444 std::string merged_file = pprz_home + "/var/gazebo.world";
445 world_sdf->Write(merged_file);
446
447 gz::sim::ServerConfig config;
448 config.SetSdfFile(merged_file);
449 // Render the cameras without a window when there is no display
450 config.SetHeadlessRendering(getenv("DISPLAY") == NULL);
451 server = std::make_unique<gz::sim::Server>(config);
452 pprz_system = std::make_shared<PaparazziSystem>();
453 if (!server->AddSystem(pprz_system).value_or(false)) {
454 fprintf(stderr, "NPS gz: could not add the paparazzi system\n");
455 exit(EXIT_FAILURE);
456 }
457
458 // Run the first iteration paused: creates the entities and configures the system
459 server->RunOnce(true);
460 if (!pprz_system->Ready()) {
461 fprintf(stderr, "NPS gz: model '%s' not found in the world\n", NPS_GAZEBO_AC_NAME);
462 exit(EXIT_FAILURE);
463 }
464
465 // gz-sim catches SIGINT and SIGTERM to stop its own run loop, which NPS does not use:
466 // restore the default handlers so Ctrl-C and the Paparazzi Center can stop the simulator
467 signal(SIGINT, SIG_DFL);
468 signal(SIGTERM, SIG_DFL);
469
470 // Motor directions as defined by the motor mixing
471#ifdef MOTOR_MIXING_YAW_COEF
472 const double yaw_coef[] = MOTOR_MIXING_YAW_COEF;
473 for (int i = 0; i < NPS_COMMANDS_NB; i++) {
474 if (yaw_coef[i] != 0.) {
475 double sign = yaw_coef[i] > 0. ? -1. : 1.;
476 gazebo_actuators.torques[i] = sign * fabs(gazebo_actuators.torques[i]);
477 gazebo_actuators.max_ang_momentum[i] = sign * fabs(gazebo_actuators.max_ang_momentum[i]);
478 }
479 }
480#endif
481
482#if NPS_SIMULATE_VIDEO
483 gazebo_init_video();
484#endif
485#if NPS_SIMULATE_LASER_RANGE_ARRAY
486 fprintf(stderr, "NPS gz: laser range array simulation is not supported yet\n");
487#endif
488}

496static void gazebo_read(void)
497{
498 static gz::math::Vector3d vel_prev;
499 static double time_prev = -1.;
500
501 if (!gz_state.valid) {
502 return;
503 }
504 const gz::math::Pose3d &pose = gz_state.pose;
505 const gz::math::Vector3d &vel = gz_state.linear_vel;
506 gz::math::Quaterniond to_enu = local_to_enu();
507
508 fdm.time = gz_state.sim_time;
509
510 // World acceleration from the velocity (as the Gazebo IMU does)
511 double dt = fdm.time - time_prev;
512 gz::math::Vector3d accel = (time_prev >= 0. && dt > 0.) ? (vel - vel_prev) / dt : gz::math::Vector3d::Zero;
513 vel_prev = vel;
514 time_prev = fdm.time;
515
516 // ltp definition in double for accuracy
517 struct LtpDef_d ltpdef_d;
518 ltpdef_d.ecef.x = state.ned_origin_f.ecef.x;
519 ltpdef_d.ecef.y = state.ned_origin_f.ecef.y;
520 ltpdef_d.ecef.z = state.ned_origin_f.ecef.z;
521 ltpdef_d.lla.lat = state.ned_origin_f.lla.lat;
522 ltpdef_d.lla.lon = state.ned_origin_f.lla.lon;
523 ltpdef_d.lla.alt = state.ned_origin_f.lla.alt;
524 for (int i = 0; i < 3 * 3; i++) {
525 ltpdef_d.ltp_of_ecef.m[i] = state.ned_origin_f.ltp_of_ecef.m[i];
526 }
527 ltpdef_d.hmsl = state.ned_origin_f.hmsl;
528
529 /* position */
530 fdm.ltpprz_pos = ned_of_enu(to_enu.RotateVector(pose.Pos()));
531 fdm.hmsl = -fdm.ltpprz_pos.z;
532 ecef_of_ned_point_d(&fdm.ecef_pos, &ltpdef_d, &fdm.ltpprz_pos);
533 lla_of_ecef_d(&fdm.lla_pos, &fdm.ecef_pos);
534 fdm.lla_pos_pprz = fdm.lla_pos;
535 fdm.lla_pos_geod = fdm.lla_pos;
536 fdm.lla_pos_geoc = fdm.lla_pos;
537 fdm.lla_pos_geoc.lat = gc_of_gd_lat_d(fdm.lla_pos.lat, fdm.hmsl);
538 fdm.agl = pose.Pos().Z(); // TODO sonar
539
540 /* velocity */
541 fdm.ltp_ecef_vel = ned_of_enu(to_enu.RotateVector(vel));
542 fdm.ltpprz_ecef_vel = fdm.ltp_ecef_vel;
543 fdm.body_ecef_vel = frd_of_flu(pose.Rot().RotateVectorReverse(vel));
544 ecef_of_ned_vect_d(&fdm.ecef_ecef_vel, &ltpdef_d, &fdm.ltp_ecef_vel);
545
546 /* acceleration */
547 fdm.ltp_ecef_accel = ned_of_enu(to_enu.RotateVector(accel));
548 fdm.ltpprz_ecef_accel = fdm.ltp_ecef_accel;
549 fdm.body_ecef_accel = frd_of_flu(pose.Rot().RotateVectorReverse(accel));
550 fdm.body_inertial_accel = fdm.body_ecef_accel;
551 fdm.body_accel = frd_of_flu(pose.Rot().RotateVectorReverse(accel - gz_state.gravity));
552 ecef_of_ned_vect_d(&fdm.ecef_ecef_accel, &ltpdef_d, &fdm.ltp_ecef_accel);
553
554 /* attitude */
555 fdm.ltp_to_body_eulers = eulers_ned_of_enu(to_enu * pose.Rot());
556 double_quat_of_eulers(&fdm.ltp_to_body_quat, &fdm.ltp_to_body_eulers);
557 fdm.ltpprz_to_body_quat = fdm.ltp_to_body_quat;
558 fdm.ltpprz_to_body_eulers = fdm.ltp_to_body_eulers;
559
560 /* angular velocity */
561 fdm.body_ecef_rotvel = rates_frd_of_flu(pose.Rot().RotateVectorReverse(gz_state.angular_vel));
562 fdm.body_inertial_rotvel = fdm.body_ecef_rotvel;
563
564 /* gravity and magnetic field */
565 fdm.ltp_g = vect_ned_of_enu(to_enu.RotateVector(-gz_state.gravity));
566 fdm.ltp_h = vect_ned_of_enu(to_enu.RotateVector(gz_state.magnetic_field));
567
568 /* atmosphere: not simulated */
569 fdm.wind = (struct DoubleVect3) {0, 0, 0};
570 fdm.pressure_sl = PPRZ_ISA_SEA_LEVEL_PRESSURE;
571 fdm.airspeed = 0;
572 fdm.pressure = pprz_isa_pressure_of_height(fdm.hmsl, fdm.pressure_sl);
573 fdm.dynamic_pressure = fdm.pressure_sl;
574 fdm.temperature = 20.0;
575 fdm.aoa = 0;
576 fdm.sideslip = 0;
577
578 /* flight controls and engines: unused */
579 fdm.elevator = 0;
580 fdm.flap = 0;
581 fdm.left_aileron = 0;
582 fdm.right_aileron = 0;
583 fdm.rudder = 0;
584 fdm.num_engines = 0;
585}

591static void gazebo_write(double act_commands[], int commands_nb)
592{
593 for (int i = 0; i < commands_nb && i < NPS_COMMANDS_NB; ++i) {
594 double sp = autopilot.motors_on ? act_commands[i] : 0.0;
595
596#ifdef NPS_ACTUATOR_TIME_CONSTANTS
597 double u = update_first_order_low_pass(&gazebo_actuators.lowpass[i], sp);
598#else
599 double u = sp;
600#endif
601 double thrust = gazebo_actuators.thrusts[i] * u;
602 double torque = gazebo_actuators.torques[i] * u;
603
604#ifdef NPS_ACTUATOR_MAX_ANGULAR_MOMENTUM
605 double udot = update_first_order_high_pass(&gazebo_actuators.highpass[i], sp);
606 torque += gazebo_actuators.max_ang_momentum[i] / (2.0 * sqrt(u > 0.05 ? u : 0.05)) * udot;
607#endif
608
609 gz_cmd.thrusts[i] = thrust;
610 gz_cmd.torques[i] = torque;
611 }

741static void read_image(struct image_t *img, const gz::msgs::Image &msg, bool is_mt9f002)
742{
743 const int w = msg.width();
744 const int h = msg.height();
745 const int step = msg.step() ? msg.step() : 3 * w;
746 if (is_mt9f002) {
747 image_create(img, MT9F002_OUTPUT_WIDTH, MT9F002_OUTPUT_HEIGHT, IMAGE_YUV422);
748 } else {
749 image_create(img, w, h, IMAGE_YUV422);
750 }
751 const uint8_t *data_rgb = (const uint8_t *)msg.data().data();
752 uint8_t *data_yuv = (uint8_t *)(img->buf);
753 for (int y_yuv = 0; y_yuv < img->h; ++y_yuv) {
754 for (int x_yuv = 0; x_yuv < img->w; ++x_yuv) {
755 int x_rgb = x_yuv;
756 int y_rgb = y_yuv;
757 if (is_mt9f002) {
758 x_rgb = (mt9f002.offset_x + ((float)x_yuv / img->w) * mt9f002.sensor_width)
759 / CFG_MT9F002_PIXEL_ARRAY_WIDTH * w;
760 y_rgb = (mt9f002.offset_y + ((float)y_yuv / img->h) * mt9f002.sensor_height)
761 / CFG_MT9F002_PIXEL_ARRAY_HEIGHT * h;
762 }
763 x_rgb = Clip(x_rgb, 0, w - 1);
764 y_rgb = Clip(y_rgb, 0, h - 1);
765 const uint8_t *rgb = data_rgb + step * y_rgb + 3 * x_rgb;
766 int idx_yuv = 2 * (img->w * y_yuv + x_yuv);
767 if (x_yuv % 2 == 0) {
768 data_yuv[idx_yuv] = - 0.148 * rgb[0] - 0.291 * rgb[1] + 0.439 * rgb[2] + 128; // U
769 } else {
770 data_yuv[idx_yuv] = 0.439 * rgb[0] - 0.368 * rgb[1] - 0.071 * rgb[2] + 128; // V
771 }
772 data_yuv[idx_yuv + 1] = 0.257 * rgb[0] + 0.504 * rgb[1] + 0.098 * rgb[2] + 16; // Y
773 }
774 }
775 img->ts.tv_sec = msg.header().stamp().sec();
776 img->ts.tv_usec = msg.header().stamp().nsec() / 1000;
777 img->pprz_ts = msg.header().stamp().sec() * 1000000 + msg.header().stamp().nsec() / 1000;
778 img->buf_idx = 0; // unused
779}
