191static void air_data_filter_init(struct air_data_sensor_t *s, float value)
192{
193 // the bilinear transform needs a cut-off below the Nyquist frequency
194 float freq = Min(s->filter_freq, 0.4f / s->dt);
195 float init = (s->filter_dt > 0.f) ? get_butterworth_2_low_pass(&s->filter) : value;
196 init_butterworth_2_low_pass(&s->filter, 1.f / (2.f * M_PI * freq), s->dt, init);
197 s->filter_dt = s->dt;
198}
199
201static float air_data_sensor_filter(struct air_data_sensor_t *s, float raw)
202{
203 if (s->filter_freq <= 0.f || s->dt <= 0.f) {
204 s->filter_dt = 0.f;
205 return raw;
206 }
207 if (s->filter_dt <= 0.f || fabsf(s->dt - s->filter_dt) > 0.2f * s->filter_dt) {
208 air_data_filter_init(s, raw);
209 }
210 return update_butterworth_2_low_pass(&s->filter, raw);
211}

213struct air_data_sensor_t *air_data_get_sensor(struct air_data_sensor_t *sensors, uint8_t abi_id, bool create)
214{
215 if (abi_id == ABI_DISABLE || abi_id == ABI_BROADCAST) {
216 return NULL;
217 }
218 // First the sensor itself (for example from the airframe calibration), then a free slot
219 for (uint8_t i = 0; i < AIR_DATA_MAX_SENSORS; i++) {
220 if (sensors[i].abi_id == abi_id) {
221 return &sensors[i];
222 }
223 }
224 if (create) {
225 for (uint8_t i = 0; i < AIR_DATA_MAX_SENSORS; i++) {
226 if (sensors[i].abi_id == ABI_DISABLE) {
227 sensors[i].abi_id = abi_id;
228 air_data_sensor_defaults(&sensors[i]);
229 return &sensors[i];
230 }
231 }
232 }
233 return NULL;
234}

237static void air_data_sensor_update(struct air_data_sensor_t *s, uint32_t stamp, float raw, float now)
238{
239 if (s->time > 0.f) {
240 float dt = now - s->time;
241 s->dt = (s->dt > 0.f) ? 0.9f * s->dt + 0.1f * dt : dt;
242 }
243 s->raw = raw;
244 s->value = (air_data_sensor_filter(s, raw) - s->offset) * s->scale;
245 s->stamp = stamp;
246 s->time = now;
247 if (calib_running) {
248 s->calib_sum += raw;
249 s->calib_cnt++;
250 }
251}

254static bool air_data_sensor_healthy(struct air_data_sensor_t *s, float now)
255{
256 return s->abi_id != ABI_DISABLE && s->time > 0.f &&
257 (now - s->time) < Max(AIR_DATA_TIMEOUT, AIR_DATA_TIMEOUT_FACTOR * s->dt);
258}

263static struct air_data_sensor_t *air_data_select(struct air_data_sensor_t *sensors[], uint8_t nb, uint8_t id,
264 uint8_t (*rank)(struct air_data_sensor_t *s), float now)
265{
266 struct air_data_sensor_t *best = NULL;
267 uint8_t best_rank = UINT8_MAX;
268 for (uint8_t k = 0; k < nb; k++) {
269 for (uint8_t i = 0; i < AIR_DATA_MAX_SENSORS; i++) {
270 struct air_data_sensor_t *s = &sensors[k][i];
271 if (!air_data_sensor_healthy(s, now)) {
272 continue;
273 }
274 if (s->abi_id == id) {
275 return s;
276 }
277 uint8_t r = rank(s);
278 if (r < best_rank) {
279 best = s;
280 best_rank = r;
281 }
282 }
283 }
284 return best;
285}

333static void air_data_set_pressure(uint32_t stamp, float pressure)
334{
335 air_data.pressure = pressure;
336
337 // calculate QNH from pressure and absolute altitude if that is available
338 if (air_data.calc_qnh_once && stateIsGlobalCoordinateValid()) {
340 // in the meantime use geoid separation at local reference frame origin
341 float geoid_separation = 0;
342 if (state.ned_initialized_f) {
343 geoid_separation = stateGetLlaOrigin_f().alt - stateGetHmslOrigin_f();
344 }
345 float h = stateGetPositionLla_f()->alt - geoid_separation;
346 air_data.qnh = pprz_isa_ref_pressure_of_height_full(air_data.pressure, h) / 100.f;
347 air_data.calc_qnh_once = false;
348 }
349
350 if (air_data.qnh > 0) {
351 air_data.amsl_baro = pprz_isa_height_of_pressure_full(air_data.pressure,
352 air_data.qnh * 100.f);
353 air_data.amsl_baro_valid = true;
354 }
355
356 /* reset baro health counter */
357 baro_health_counter = 10;
358
359 AbiSendMsgBARO_ABS(AIR_DATA_SENDER_ID, stamp, pressure);
360}

378static void baro_raw_cb(uint8_t sender_id, uint32_t stamp, float pressure)
379{
380 // bad pressure (Pa), don't use
381 if (!isfinite(pressure) || pressure < 1000.f || pressure > 120000.f) {
382 return;
383 }
384 struct air_data_sensor_t *s = air_data_get_sensor(air_data.baros, sender_id, true);
385 if (s == NULL) {
386 return;
387 }
388 float now = get_sys_time_float();
389 air_data_sensor_update(s, stamp, pressure, now);
390#if AIR_DATA_SEND_ALL_SENSORS
391 AbiSendMsgBARO_ABS(sender_id, stamp, s->value);
392#endif
393
394 struct air_data_sensor_t *sel = air_data_select_baro(now);
395 air_data.baro_used = (sel != NULL) ? sel->abi_id : ABI_DISABLE;
396 if (sel == s) {
397 air_data_set_pressure(stamp, s->value);
398 } else if (!calib_running && !s->calibrated.offset && sel != NULL && sel->calibrated.offset) {
399 // a barometer that starts after the calibration: line it up with the one in use
400 s->calib_sum += pressure - air_data.pressure / s->scale;
401 s->calib_cnt++;
402 if (s->calib_cnt >= AIR_DATA_ALIGN_NB) {
403 s->offset = s->calib_sum / s->calib_cnt;
404 s->calibrated.offset = true;
405 }
406 }
407}

409static void diff_raw_cb(uint8_t sender_id, uint32_t stamp, float pressure)
410{
411 if (!isfinite(pressure) || fabsf(pressure) > AIR_DATA_DIFF_MAX) {
412 return;
413 }
414 struct air_data_sensor_t *s = air_data_get_sensor(air_data.diffs, sender_id, true);
415 if (s == NULL) {
416 return;
417 }
418 float now = get_sys_time_float();
419 air_data_sensor_update(s, stamp, pressure, now);
420#if AIR_DATA_SEND_ALL_SENSORS
421 AbiSendMsgBARO_DIFF(sender_id, s->value);
422 AbiSendMsgAIRSPEED(sender_id, eas_from_dynamic_pressure(s->value));
423#endif
424
425 struct air_data_sensor_t *sel = air_data_select_airspeed(now);
426 air_data.airspeed_used = (sel != NULL) ? sel->abi_id : ABI_DISABLE;
427 if (sel == s) {
428 air_data.airspeed_filter = s->filter_freq;
429 air_data.differential = s->value;
430 AbiSendMsgBARO_DIFF(AIR_DATA_SENDER_ID, s->value);
431 air_data_set_airspeed(eas_from_dynamic_pressure(s->value), true);
432 }
433}

435static void airspeed_raw_cb(uint8_t sender_id, uint32_t stamp, float eas)
436{
437 if (!isfinite(eas)) {
438 return;
439 }
440 struct air_data_sensor_t *s = air_data_get_sensor(air_data.airspeeds, sender_id, true);
441 if (s == NULL) {
442 return;
443 }
444 float now = get_sys_time_float();
445 air_data_sensor_update(s, stamp, eas, now);
446#if AIR_DATA_SEND_ALL_SENSORS
447 if (!AIR_DATA_IS_ESTIMATE(sender_id)) {
448 AbiSendMsgAIRSPEED(sender_id, s->value);
449 }
450#endif
451
452 struct air_data_sensor_t *sel = air_data_select_airspeed(now);
453 air_data.airspeed_used = (sel != NULL) ? sel->abi_id : ABI_DISABLE;
454 if (sel == s) {
455 air_data.airspeed_filter = s->filter_freq;
456 // an estimate isn't sent, the estimators (e.g. ins_ekf2) would fuse it
457 air_data_set_airspeed(s->value, !AIR_DATA_IS_ESTIMATE(sender_id));
458 }
459}

461static void temp_raw_cb(uint8_t sender_id, uint32_t stamp, float temp, uint8_t type)
462{
463 if (!isfinite(temp) || temp < -100.f || temp > 150.f) {
464 return;
465 }
466 // only air temperatures: not the IMU and other ones (their ids also overlap with the baro ids)
467 if (type == TEMPERATURE_TYPE_IMU || type == TEMPERATURE_TYPE_OTHER) {
468 return;
469 }
470 struct air_data_sensor_t *s = air_data_get_sensor(air_data.temps, sender_id, true);
471 if (s == NULL) {
472 return;
473 }
474 s->type = type;
475 float now = get_sys_time_float();
476 air_data_sensor_update(s, stamp, temp, now);
477
478 struct air_data_sensor_t *sel = air_data_select_temp(now);
479 air_data.temp_used = (sel != NULL) ? sel->abi_id : ABI_DISABLE;
480 if (sel == s) {
481 air_data.temperature = s->value;
482 AbiSendMsgTEMPERATURE(AIR_DATA_SENDER_ID, air_data.temperature);
483 }
484}

504static void air_data_start_calibration(bool boot, bool baro, bool diff)
505{
506 air_data.calibrate_baro = false;
507 air_data.zero_airspeed = false;
508 if (autopilot_get_motors_on() && !air_data.calib_armed) {
509 // refused while armed
510#if DOWNLINK
511 char msg[] = "Air data calibration refused: armed";
512 DOWNLINK_SEND_INFO_MSG(DefaultChannel, DefaultDevice, strlen(msg), msg);
513#endif
514 return;
515 }
516 struct air_data_sensor_t *sensors[] = {air_data.baros, air_data.diffs};
517 for (uint8_t k = 0; k < 2; k++) {
518 for (uint8_t i = 0; i < AIR_DATA_MAX_SENSORS; i++) {
519 sensors[k][i].calib_sum = 0.f;
520 sensors[k][i].calib_cnt = 0;
521 }
522 }
523 calib_boot = boot;
524 calib_baro = baro;
525 calib_diff = diff;
526 calib_start = get_sys_time_float();
527 calib_running = true;
528 air_data.calibrate_baro = baro;
529 air_data.zero_airspeed = diff;
530}

535static void air_data_finish_calibration(void)
536{
537 calib_running = false;
538 air_data.calibrate_baro = false;
539 air_data.zero_airspeed = false;
540
541 for (uint8_t i = 0; i < AIR_DATA_MAX_SENSORS; i++) {
542 struct air_data_sensor_t *s = &air_data.diffs[i];
543 if (s->calib_cnt == 0) {
544 continue;
545 }
546 float mean = s->calib_sum / s->calib_cnt;
547 if (calib_diff) {
548 s->offset = mean;
549 s->calibrated.offset = true;
550 } else if (s->calibrated.offset && fabsf((mean - s->offset) * s->scale) > AIR_DATA_DIFF_REST_MAX) {
551#if DOWNLINK
552 char msg[64];
553 int len = snprintf(msg, sizeof(msg), "Airspeed sensor %d: %d Pa at rest, check it or ZeroAirspeed",
554 s->abi_id, (int)((mean - s->offset) * s->scale));
555 if (len > 0) {
556 DOWNLINK_SEND_INFO_MSG(DefaultChannel, DefaultDevice, Min(len, (int)sizeof(msg) - 1), msg);
557 }
558#endif
559 }
560 }
561
562 if (!calib_baro) {
563 return;
564 }
565 struct air_data_sensor_t *ref = air_data_select_baro(get_sys_time_float());
566 if (ref != NULL && ref->calib_cnt > 0) {
567 float p_ref = (ref->calib_sum / ref->calib_cnt - ref->offset) * ref->scale;
568 air_data.qfe = p_ref;
569 ref->calibrated.offset = true;
570 for (uint8_t i = 0; i < AIR_DATA_MAX_SENSORS; i++) {
571 struct air_data_sensor_t *s = &air_data.baros[i];
572 if (s != ref && s->calib_cnt > 0 && !(calib_boot && s->calibrated.offset)) {
573 s->offset = s->calib_sum / s->calib_cnt - p_ref / s->scale;
574 s->calibrated.offset = true;
575 }
576 }
577 air_data.calc_qnh_once = true;
578 }
579}

595void air_data_SetAirspeedFilter(float freq)
596{
597 struct air_data_sensor_t *s = air_data_get_sensor(air_data.diffs, air_data.airspeed_used, false);

620static void air_data_preflight(struct preflight_result_t *result)
621{
622 uint16_t fail_cnt = result->fail_cnt;
623 uint16_t warning_cnt = result->warning_cnt;
624 if (calib_running) {
625 preflight_error(result, "Air data calibrating");
626 }
627 // Not zeroed (not in DIFF_CALIB): ZeroAirspeed before arming
628 // At rest before arming: warn on a large differential pressure (wrong offset, broken sensor or wind), don't block
629 float now = get_sys_time_float();
630 for (uint8_t i = 0; i < AIR_DATA_MAX_SENSORS; i++) {
631 struct air_data_sensor_t *s = &air_data.diffs[i];
632 if (s->abi_id != ABI_DISABLE && !s->calibrated.offset) {
633 preflight_error(result, "Airspeed sensor %d not zeroed, ZeroAirspeed", s->abi_id);
634 } else if (air_data_sensor_healthy(s, now) && fabsf(s->value) > AIR_DATA_DIFF_REST_MAX) {
635 preflight_warning(result, "Airspeed sensor %d: %d Pa at rest", s->abi_id, (int)s->value);
636 }
637 }
638 if (result->fail_cnt == fail_cnt && result->warning_cnt == warning_cnt) {
639 preflight_success(result, "Air data ok");
640 }
641}

667static void send_baro_raw(struct transport_tx *trans, struct link_device *dev)
668{
669 static uint8_t idx = 0;
670 if (air_data.baro_send_id == ABI_DISABLE) {
671 return;
672 }
673 struct air_data_sensor_t *s = (air_data.baro_send_id == ABI_BROADCAST) ? air_data_next(air_data.baros, &idx) :
674 air_data_get_sensor(air_data.baros, air_data.baro_send_id, false);
675 if (s == NULL) {
676 return;

882float eas_from_dynamic_pressure(float q)
883{
884 /* q (dynamic pressure) = total pressure - static pressure
885 * q = 1/2*rho*speed^2
886 * speed = sqrt(2*q/rho)
887 * With rho = air density at sea level.
888 * Lower bound of q at zero, no flying backwards guys...
889 */
890 const float two_div_rho_0 = 2.0 / PPRZ_ISA_AIR_DENSITY;
891 float sign = 1.0f;
892 if (q < 0) {
893 sign = -1.0f;
894 q = -q;
895 }
896 return sqrtf(q * two_div_rho_0) * sign;
897}

915float get_tas_factor(float p, float t)
916{
917 /* factor to convert EAS to TAS:
918 * sqrt(rho0 / rho) = sqrt((p0 * T) / (p * T0))
919 * convert input temp to Kelvin
920 */
921 return sqrtf((PPRZ_ISA_SEA_LEVEL_PRESSURE * KelvinOfCelsius(t)) /
922 (p * PPRZ_ISA_SEA_LEVEL_TEMP));
923}
924
928static void compute_tas_factor(void)
929{
930 if (air_data.pressure > 0.f && air_data.temperature > -900.f) {
931 // compute air density from pressure and temperature
932 air_data.tas_factor = get_tas_factor(air_data.pressure, air_data.temperature);
933 }
934 else {
935 // compute air density from altitude in ISA condition
936 const float z = air_data_get_amsl();
937 const float p = pprz_isa_pressure_of_altitude(z);
938 const float t = pprz_isa_temperature_of_altitude(z);
939 air_data.tas_factor = get_tas_factor(p, CelsiusOfKelvin(t));
940 }
941}
942
952float tas_from_eas(float eas)
953{
954 compute_tas_factor();
955 return air_data.tas_factor * eas;
956}
957
967float eas_from_tas(float tas)
968{
969 compute_tas_factor();
970 return tas / air_data.tas_factor;
971}

981float tas_from_dynamic_pressure(float q)
982{
983 return tas_from_eas(eas_from_dynamic_pressure(q));
