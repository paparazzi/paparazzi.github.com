118static void pressure_abs_cb(uint8_t __attribute__((unused)) sender_id, uint32_t __attribute__((unused)) stamp, float pressure)
119{
120 air_data.pressure = pressure;
121
122 // calculate QNH from pressure and absolute altitude if that is available
123 if (air_data.calc_qnh_once && stateIsGlobalCoordinateValid()) {
125 // in the meantime use geoid separation at local reference frame origin
126 float geoid_separation = 0;
127 if (state.ned_initialized_f) {
128 geoid_separation = stateGetLlaOrigin_f().alt - stateGetHmslOrigin_f();
129 }
130 float h = stateGetPositionLla_f()->alt - geoid_separation;
131 air_data.qnh = pprz_isa_ref_pressure_of_height_full(air_data.pressure, h) / 100.f;
132 air_data.calc_qnh_once = false;
133 }
134
135 if (air_data.calc_amsl_baro && air_data.qnh > 0) {
136 air_data.amsl_baro = pprz_isa_height_of_pressure_full(air_data.pressure,
137 air_data.qnh * 100.f);
138 air_data.amsl_baro_valid = true;
139 }
140
141 /* reset baro health counter */
142 baro_health_counter = 10;
143}

145static void pressure_diff_cb(uint8_t __attribute__((unused)) sender_id, float pressure)
146{
147 air_data.differential = pressure;
148 if (air_data.calc_airspeed) {
149 air_data.airspeed = eas_from_dynamic_pressure(air_data.differential);
150 air_data.tas = tas_from_eas(air_data.airspeed);
151#if USE_AIRSPEED_AIR_DATA
152 stateSetAirspeed_f(MODULE_AIR_DATA_ID, air_data.airspeed);
153#endif
154 /* for other modules (e.g. the airspeed fusion of ins_ekf2) when the sensor only gives
155 * the differential pressure (e.g. ms45xx, uavcan), not when it also sends its airspeed (e.g. sdp3x) */
156 if (get_sys_time_float() - sensor_airspeed_time > 1.f) {
157 AbiSendMsgAIRSPEED(AIRSPEED_AIR_DATA_ID, air_data.airspeed);
158 }
159 }
160}

172static void airspeed_cb(uint8_t sender_id, float eas)
173{
174 if (sender_id == AIRSPEED_AIR_DATA_ID) {
175 return; // computed here from the differential pressure
176 }
177 if (sender_id != AIRSPEED_WE_QUAD_ID) {
178 sensor_airspeed_time = get_sys_time_float(); // an estimate isn't a sensor
179 }
180 air_data.airspeed = eas;
181 if (air_data.calc_airspeed) {
182 air_data.tas = tas_from_eas(air_data.airspeed);
183#if USE_AIRSPEED_AIR_DATA
184 stateSetAirspeed_f(MODULE_AIR_DATA_ID, air_data.airspeed);
185#endif
186 }
187}

233void air_data_init(void)
234{
235 air_data.calc_airspeed = AIR_DATA_CALC_AIRSPEED;
236 air_data.calc_tas_factor = AIR_DATA_CALC_TAS_FACTOR;
237 air_data.calc_amsl_baro = AIR_DATA_CALC_AMSL_BARO;
238 air_data.tas_factor = AIR_DATA_TAS_FACTOR;
239 air_data.calc_qnh_once = true;
240 air_data.amsl_baro_valid = false;
241
242 /* initialize the output variables
243 * pressure, qnh, temperature and airspeed to invalid values,
244 * rest to zero
245 */
246 air_data.pressure = -1.0f;
247 air_data.qnh = -1.0f;
248 air_data.airspeed = -1.0f;
249 air_data.tas = -1.0f;
250 air_data.temperature = -1000.0f;
251 air_data.differential = 0.0f;
252 air_data.amsl_baro = 0.0f;
253 air_data.aoa = 0.0f;
254 air_data.sideslip = 0.0f;
255 air_data.wind_speed = 0.0f;
256 air_data.wind_dir = 0.0f;
257
258 /* internal variables */
259 baro_health_counter = 0;
260
261 AbiBindMsgBARO_ABS(AIR_DATA_BARO_ABS_ID, &pressure_abs_ev, pressure_abs_cb);
262 AbiBindMsgBARO_DIFF(AIR_DATA_BARO_DIFF_ID, &pressure_diff_ev, pressure_diff_cb);
263 AbiBindMsgTEMPERATURE(AIR_DATA_TEMPERATURE_ID, &temperature_ev, temperature_cb);
264 AbiBindMsgAIRSPEED(AIR_DATA_AIRSPEED_ID, &airspeed_ev, airspeed_cb);
265 AbiBindMsgINCIDENCE(AIR_DATA_INCIDENCE_ID, &incidence_ev, incidence_cb);
266
267#if PERIODIC_TELEMETRY
268 register_periodic_telemetry(DefaultPeriodic, PPRZ_MSG_ID_BARO_RAW, send_baro_raw);
269 register_periodic_telemetry(DefaultPeriodic, PPRZ_MSG_ID_AIR_DATA, send_air_data);
270 register_periodic_telemetry(DefaultPeriodic, PPRZ_MSG_ID_AMSL, send_amsl);
271#endif

294void air_data_parse_WIND_INFO(struct link_device *dev __attribute__((unused)), struct transport_tx *trans __attribute__((unused)), uint8_t *buf)
295{
296 if (DL_WIND_INFO_ac_id(buf) != AC_ID) { return; }
297 uint8_t flags = DL_WIND_INFO_flags(buf);
298 struct FloatVect2 wind = { 0.f, 0.f };
299 float upwind = 0.f;
300 if (bit_is_set(flags, 0)) {
301 wind.x = DL_WIND_INFO_north(buf);
302 wind.y = DL_WIND_INFO_east(buf);
303 stateSetHorizontalWindspeed_f(MODULE_AIR_DATA_ID, &wind);
304 air_data.wind_speed = float_vect2_norm(&wind);
305 air_data.wind_dir = atan2f(wind.y, wind.x);
306 }
307 if (bit_is_set(flags, 1)) {
308 upwind = DL_WIND_INFO_up(buf);
309 stateSetVerticalWindspeed_f(MODULE_AIR_DATA_ID, upwind);
310 }
311#if !USE_AIRSPEED
312 if (bit_is_set(flags, 2)) {
313 air_data.tas = DL_WIND_INFO_airspeed(buf);
314 air_data.airspeed = eas_from_tas(air_data.tas);
315 stateSetAirspeed_f(MODULE_AIR_DATA_ID, air_data.airspeed);
316 }
317#endif
318#ifdef WIND_INFO_RET
319 float airspeed = stateGetAirspeed_f();
320 pprz_msg_send_WIND_INFO_RET(trans, dev, AC_ID, &flags, &wind.y, &wind.x, &upwind, &airspeed);
321#endif
322}

337float eas_from_dynamic_pressure(float q)
338{
339 /* q (dynamic pressure) = total pressure - static pressure
340 * q = 1/2*rho*speed^2
341 * speed = sqrt(2*q/rho)
342 * With rho = air density at sea level.
343 * Lower bound of q at zero, no flying backwards guys...
344 */
345 const float two_div_rho_0 = 2.0 / PPRZ_ISA_AIR_DENSITY;
346 float sign = 1.0f;
347 if (q < 0) {
348 sign = -1.0f;
349 q = -q;
350 }
351 return sqrtf(q * two_div_rho_0) * sign;
352}

370float get_tas_factor(float p, float t)
371{
372 /* factor to convert EAS to TAS:
373 * sqrt(rho0 / rho) = sqrt((p0 * T) / (p * T0))
374 * convert input temp to Kelvin
375 */
376 return sqrtf((PPRZ_ISA_SEA_LEVEL_PRESSURE * KelvinOfCelsius(t)) /
377 (p * PPRZ_ISA_SEA_LEVEL_TEMP));
378}

383static void compute_tas_factor(void)
384{
385 // update tas factor if requested
386 if (air_data.calc_tas_factor) {
387 if (air_data.pressure > 0.f && air_data.temperature > -900.f) {
388 // compute air density from pressure and temperature
389 air_data.tas_factor = get_tas_factor(air_data.pressure, air_data.temperature);
390 }
391 else {
392 // compute air density from altitude in ISA condition
393 const float z = air_data_get_amsl();
394 const float p = pprz_isa_pressure_of_altitude(z);
395 const float t = pprz_isa_temperature_of_altitude(z);
396 air_data.tas_factor = get_tas_factor(p, CelsiusOfKelvin(t));
397 }
398 }
399}
400
410float tas_from_eas(float eas)
411{
412 compute_tas_factor();
413 return air_data.tas_factor * eas;
414}
415
425float eas_from_tas(float tas)
426{
427 compute_tas_factor();
428 return tas / air_data.tas_factor;
429}

439float tas_from_dynamic_pressure(float q)
440{
441 return tas_from_eas(eas_from_dynamic_pressure(q));
