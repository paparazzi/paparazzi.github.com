57void gps_feed_value(void)
58{
59 // FIXME, set proper time instead of hardcoded to May 2014
60 gps_nps.week = 1794;
61 gps_nps.tow = fdm.time * 1000;
62
63 gps_nps.ecef_pos.x = sensors.gps.ecef_pos.x * 100.;
64 gps_nps.ecef_pos.y = sensors.gps.ecef_pos.y * 100.;
65 gps_nps.ecef_pos.z = sensors.gps.ecef_pos.z * 100.;
66 SetBit(gps_nps.valid_fields, GPS_VALID_POS_ECEF_BIT);
67 gps_nps.ecef_vel.x = sensors.gps.ecef_vel.x * 100.;
68 gps_nps.ecef_vel.y = sensors.gps.ecef_vel.y * 100.;
69 gps_nps.ecef_vel.z = sensors.gps.ecef_vel.z * 100.;
70 SetBit(gps_nps.valid_fields, GPS_VALID_VEL_ECEF_BIT);
71
72 //ecef pos seems to be based on geocentric model, hence we get a very high alt when converted to lla
73 gps_nps.lla_pos.lat = DegOfRad(sensors.gps.lla_pos.lat) * 1e7;
74 gps_nps.lla_pos.lon = DegOfRad(sensors.gps.lla_pos.lon) * 1e7;
75 gps_nps.lla_pos.alt = sensors.gps.lla_pos.alt * 1000.;
76 SetBit(gps_nps.valid_fields, GPS_VALID_POS_LLA_BIT);
77
78 gps_nps.hmsl = sensors.gps.hmsl * 1000.;
79 SetBit(gps_nps.valid_fields, GPS_VALID_HMSL_BIT);
80
81 /* calc NED speed from ECEF */
82 struct LtpDef_d ref_ltp;
83 ltp_def_from_ecef_d(&ref_ltp, &sensors.gps.ecef_pos);
84 struct NedCoor_d ned_vel_d;
85 ned_of_ecef_vect_d(&ned_vel_d, &ref_ltp, &sensors.gps.ecef_vel);
86 gps_nps.ned_vel.x = ned_vel_d.x * 100;
87 gps_nps.ned_vel.y = ned_vel_d.y * 100;
88 gps_nps.ned_vel.z = ned_vel_d.z * 100;
89 SetBit(gps_nps.valid_fields, GPS_VALID_VEL_NED_BIT);
90
91 /* horizontal and 3d ground speed in cm/s */
92 gps_nps.gspeed = sqrt(ned_vel_d.x * ned_vel_d.x + ned_vel_d.y * ned_vel_d.y) * 100;
93 gps_nps.speed_3d = sqrt(ned_vel_d.x * ned_vel_d.x + ned_vel_d.y * ned_vel_d.y + ned_vel_d.z * ned_vel_d.z) * 100;
94
95#if GpsId(PRIMARY_GPS) == GPS_DATALINK_ID
96 /* vehicle heading in radians * 1e7 */
97 gps_nps.course = fdm.ltp_to_body_eulers.psi * 1e7;
98#else
99 /* ground course in radians * 1e7 */
100 gps_nps.course = atan2(ned_vel_d.y, ned_vel_d.x) * 1e7;
101#endif
102 SetBit(gps_nps.valid_fields, GPS_VALID_COURSE_BIT);
103
104 /* accuracies of a good GNSS receiver with a clear sky view */
105 gps_nps.pacc = GPS_NPS_PACC;
106 gps_nps.hacc = GPS_NPS_HACC;
107 gps_nps.vacc = GPS_NPS_VACC;
108 gps_nps.sacc = GPS_NPS_SACC;
109 gps_nps.pdop = GPS_NPS_PDOP;
110
111 if (gps_has_fix) {
112 gps_nps.num_sv = 11;
113 gps_nps.fix = GPS_FIX_3D;
114 } else {
115 gps_nps.num_sv = 1;
116 gps_nps.fix = GPS_FIX_NONE;
117 }
118
119 // publish gps data
120 uint32_t now_ts = get_sys_time_usec();
121 gps_nps.last_msg_ticks = sys_time.nb_sec_rem;
122 gps_nps.last_msg_time = sys_time.nb_sec;
123 if (gps_nps.fix == GPS_FIX_3D) {
124 gps_nps.last_3dfix_ticks = sys_time.nb_sec_rem;
125 gps_nps.last_3dfix_time = sys_time.nb_sec;
126 }
127 AbiSendMsgGPS(GPS_SIM_ID, now_ts, &gps_nps);
128}
