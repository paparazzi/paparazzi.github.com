137void nps_fdm_run_step(bool launch __attribute__((unused)), double *commands, int commands_nb __attribute__((unused)))
138{
139 fdm.time += fdm.curr_dt;
140
141 /****************************************************************************
142 PHYSICAL MODEL
143 -------------
144 The physical model of your rover goes here. This physics takes place in
145 the LTP plane (so we transfer every integration step to NED and ECEF at the end).
146 */
147
148 // From previous step...
149 ned_of_ecef_point_d(&rover_pos, &ltpdef, &fdm.ecef_pos);
150 ned_of_ecef_vect_d(&rover_vel, &ltpdef, &fdm.ecef_ecef_vel);
151 double phi = fdm.ltpprz_to_body_eulers.psi;
152
153 #if (defined COMMAND_STEERING) && (defined COMMAND_THROTTLE) // STEERING ROVER PHYSICS
154
155 // Steering rover cmds:
156 // COMMAND_STEERING -> delta parameter
157 // COMMAND_THROTTLE -> acceleration in heading direction
158
160 double delta = RadOfDeg(commands[COMMAND_STEERING] * MAX_DELTA);
161 double speed = FLOAT_VECT2_NORM(rover_vel);
162 double phi_d = tan(delta) / DRIVE_SHAFT_DISTANCE * speed;
163 double accel = NPS_ROVER_ACCELERATION * commands[COMMAND_THROTTLE];
164
165 // NED accel = R(psi) [accel; |V|*phi_d] - mu*V
166 rover_acc.x = accel * cos(phi) - speed * phi_d * sin(phi) - mu * rover_vel.x;
167 rover_acc.y = accel * sin(phi) + speed * phi_d * cos(phi) - mu * rover_vel.y;
168 // Velocities (EULER INTEGRATION)
169 rover_vel.x += rover_acc.x * fdm.curr_dt;
170 rover_vel.y += rover_acc.y * fdm.curr_dt;
171
172 #elif (defined COMMAND_TURN) && (defined COMMAND_SPEED) // 2 wheels rover dynamic
173
174 double phi_sp = NPS_ROVER_TURN_RATE * commands[COMMAND_TURN];
175 double phi_d = update_first_order_low_pass(&turn_filter, phi_sp);
176 double speed_sp = NPS_ROVER_MAX_SPEED * commands[COMMAND_SPEED];
177 double speed = update_first_order_low_pass(&speed_filter, speed_sp);
178 double vx = speed * cos(phi);
179 double vy = speed * sin(phi);
180 rover_acc.x = (vx - rover_vel.x) / fdm.curr_dt;
181 rover_acc.y = (vy - rover_vel.y) / fdm.curr_dt;
182 rover_vel.x = vx;
183 rover_vel.y = vy;
184
185 #else
186 #warning "The physics of this rover are not yet implemented in nps_fdm_rover!!"
187 #endif // STEERING ROVER PHYSICS
188
189 // Positions
190 rover_pos.x += rover_vel.x * fdm.curr_dt;
191 rover_pos.y += rover_vel.y * fdm.curr_dt;
192 // phi have to be contained in [-180º,180º). So:
193 phi += phi_d * fdm.curr_dt;
194 NormRadAngle(phi);
195 /****************************************************************************/
196 // Coordenates transformations |
197 // ----------------------------|
198
199 /* in ECEF */
200 ecef_of_ned_point_d(&fdm.ecef_pos, &ltpdef, &rover_pos);
201 ecef_of_ned_vect_d(&fdm.ecef_ecef_vel, &ltpdef, &rover_vel);
202 ecef_of_ned_vect_d(&fdm.ecef_ecef_accel, &ltpdef, &rover_acc);
203
204 /* in NED */
205 ned_of_ecef_point_d(&fdm.ltpprz_pos, &ltpdef, &fdm.ecef_pos);
206 ned_of_ecef_vect_d(&fdm.ltpprz_ecef_vel, &ltpdef, &fdm.ecef_ecef_vel);
207 ned_of_ecef_vect_d(&fdm.ltpprz_ecef_accel, &ltpdef, &fdm.ecef_ecef_accel);
208
209 /* Height above sea level */
210 fdm.hmsl = ltpdef.hmsl - rover_pos.z;
211
212 /* Eulers */
213 fdm.ltpprz_to_body_eulers.psi = phi;
214 fdm.ltp_to_body_eulers.psi = phi;
215
216 // Exporting Eulers to AHRS (in quaternions)
217 double_quat_of_eulers(&fdm.ltp_to_body_quat, &fdm.ltp_to_body_eulers);
218
219 // Angular vel & acc
220 fdm.body_ecef_rotaccel.r = (phi_d - fdm.body_ecef_rotvel.r) / fdm.curr_dt;
221 fdm.body_ecef_rotvel.r = phi_d;
222
223}

230static void init_ltp(void)
231{
232 struct LlaCoor_d llh_nav0;
233 llh_nav0.lat = RadOfDeg((double)NAV_LAT0 / 1e7);
234 llh_nav0.lon = RadOfDeg((double)NAV_LON0 / 1e7);
235 llh_nav0.alt = (double)(NAV_ALT0 + NAV_MSL0) / 1000.; /* Height above the ellipsoid */
236
237 struct EcefCoor_d ecef_nav0;
238 ecef_of_lla_d(&ecef_nav0, &llh_nav0);
239
240 ltp_def_from_ecef_d(&ltpdef, &ecef_nav0);
241 ltpdef.hmsl = (double)NAV_ALT0 / 1000.; /* Height above the geoid */
242
243 fdm.ecef_pos = ecef_nav0;
244 fdm.hmsl = ltpdef.hmsl;
245 fdm.ltp_g.x = 0.;
246 fdm.ltp_g.y = 0.;
247 fdm.ltp_g.z = 0.; // accel data are already with the correct format
248
249
250#ifdef AHRS_H_X
251 fdm.ltp_h.x = AHRS_H_X;
252 fdm.ltp_h.y = AHRS_H_Y;
253 fdm.ltp_h.z = AHRS_H_Z;
254 PRINT_CONFIG_MSG("Using magnetic field as defined in airframe file.")
255#else
256 fdm.ltp_h.x = 1.;
257 fdm.ltp_h.y = 0.;
258 fdm.ltp_h.z = 0.;
259#endif
260
261}
