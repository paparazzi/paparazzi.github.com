47static void airspeed_uavcan_cb(struct uavcan_iface_t *iface __attribute__((unused)), CanardRxTransfer *transfer) {
48
49 struct uavcan_equipment_air_data_RawAirData msg;
50
51 if(uavcan_equipment_air_data_RawAirData_decode(transfer, &msg)) {
52 return; // decode error
53 }
54
55 uint32_t now_ts = get_sys_time_usec();
56 float diff_p = msg.differential_pressure;
57 float static_air_temp = msg.static_air_temperature;
58
59
60 if(!isnan(diff_p)) {
61 // Apply a scaling factor, air_data filters and zeroes it and computes the airspeed
62 airspeed_uavcan.diff_p = diff_p * airspeed_uavcan.diff_p_scale;
63
64 AbiSendMsgBARO_DIFF_RAW(UAVCAN_SENDER_ID, now_ts, airspeed_uavcan.diff_p);
65 }
66
67 if(!isnan(static_air_temp)) {
68 // Kelvin to Celsius, if the node sends Kelvin
69 if (static_air_temp > AIRSPEED_UAVCAN_KELVIN_MIN) {
70 static_air_temp += PPRZ_ISA_ABS_NULL;
71 }
72 airspeed_uavcan.temperature = static_air_temp;
73 AbiSendMsgTEMPERATURE_RAW(UAVCAN_SENDER_ID, now_ts, airspeed_uavcan.temperature, TEMPERATURE_TYPE_AIR);
74 }
75}
