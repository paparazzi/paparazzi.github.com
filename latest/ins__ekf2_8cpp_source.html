367static uint64_t ekf2_time_us(uint32_t stamp)
368{
369 static uint32_t last_now = 0;
370 static uint64_t wraps = 0;
371 uint32_t now = get_sys_time_usec();
372 if (now < last_now && (last_now - now) > (1UL << 31)) {
373 wraps += (1ULL << 32);
374 }
375 last_now = now;
376
377 // A stamp more than half the range later than now was taken before the last wrap
378 uint64_t time_us = wraps + stamp;
379 if (stamp > now && (stamp - now) > (1UL << 31) && time_us >= (1ULL << 32)) {
380 time_us -= (1ULL << 32);
381 }
382 return time_us;
383}
384
386static uint64_t ekf2_delayed_time_us(uint32_t stamp, uint32_t delay_ms)
387{
388 uint64_t time_us = ekf2_time_us(stamp);
389 return (time_us > delay_ms * 1000ULL) ? time_us - delay_ms * 1000ULL : 0;
390}

436static void ekf2_log(uint8_t type, uint64_t time_us, uint8_t nb_ints, int32_t *ints, uint8_t nb_values, float *values)
437{
438 if (ekf2_log_state != EKF2_LOG_ON) {
439 return;
440 }
441 uint32_t t = (uint32_t)time_us;
442 int32_t no_int = 0;
443 float no_value = 0;
444 pprz_msg_send_EKF2_INPUT(EKF2_LOG_TP, EKF2_LOG_DEV, AC_ID, &t, &type,
445 nb_ints, nb_ints ? ints : &no_int, nb_values, nb_values ? values : &no_value);
446}

449static void ekf2_log_params(uint64_t time_us)
450{
451 for (int i = 0; i < ins_ekf2_params_nb; i += 32) {
452 float values[32];
453 uint8_t n = 0;
454 for (int j = i; j < ins_ekf2_params_nb && n < 32; j++) {
455 values[n++] = ins_ekf2_params[j].get(*ekf_params);
456 }
457 int32_t first = i;
458 ekf2_log(EKF2_IN_PARAMS, time_us, 1, &first, n, values);
459 }
460}

463static void ekf2_log_param_changed(uint64_t time_us, const char *name)
464{
465 for (int i = 0; i < ins_ekf2_params_nb; i++) {
466 if (!strcmp(ins_ekf2_params[i].name, name)) {
467 int32_t index = i;
468 float value = ins_ekf2_params[i].get(*ekf_params);
469 ekf2_log(EKF2_IN_PARAMS, time_us, 1, &index, 1, &value);
470 return;
471 }

489static void ekf2_log_check_start(void)
490{
491 if (ekf2_log_state != EKF2_LOG_WAITING) {
492 return;
493 }
494 if (ekf2_log_ready()) {
495 ekf2_log_state = EKF2_LOG_ON;
496 uint64_t now = ekf2_time_us(get_sys_time_usec());
497 ekf2_log_params(now);
498 float config[6] = {INS_EKF2_TAU_POS, INS_EKF2_TAU_VEL, INS_EKF2_GPS_YAW_RST,
499 INS_EKF2_RNG_MIN_DIST, INS_EKF2_RNG_MAX_DIST, INS_EKF2_OF_MAX_RATE};
500 ekf2_log(EKF2_IN_CONFIG, now, 0, NULL, 6, config);
501 if (ekf2_log_origin.set) {
502 ekf2_log_origin_set(now, ekf2_log_origin.lat, ekf2_log_origin.lon, ekf2_log_origin.alt);
503 }
504 } else if (get_sys_time_float() > INS_EKF2_LOG_WAIT) {
505 ekf2_log_state = EKF2_LOG_OFF;
506 }
507}

608 uint16_t soln_status = ekf.get_ekf_soln_status();
609
610 // Innovation test ratios, a bit is set in innov_test_status when the test fails (ratio > 1)
611 float mag = test_ratio(ekf.getHeadingInnovationTestRatio());
612 float vel = test_ratio(math::max(test_ratio(ekf.getHorizontalVelocityInnovationTestRatio()),
613 test_ratio(ekf.getVerticalVelocityInnovationTestRatio())));

615 float hgt = test_ratio(ekf.getVerticalPositionInnovationTestRatio());

619 float flow = test_ratio(math::max(test_ratio(ekf.aid_src_optical_flow().test_ratio[0]),
620 test_ratio(ekf.aid_src_optical_flow().test_ratio[1])));
621 const float ratios[] = {mag, vel, pos, hgt, tas, hagl, flow, beta};

625 innov_test_status |= (1 << i);
626 }
627 }
628

946template<typename T>
947static void ekf2_event_bits(const struct ekf2_status_text *texts, uint8_t nb, float *times, T &reported, T current, float now)
948{
949 for (uint8_t i = 0; i < nb; i++) {
950 T mask = ((T)1) << texts[i].bit;
951 if ((current & mask) == (reported & mask) || (times[i] > 0.f && now - times[i] < INS_EKF2_EVENT_HOLDOFF)) {
952 continue;
953 }
954 const char *text = (current & mask) ? texts[i].on : texts[i].off;
955 if (text != NULL) {
956 ekf2_event_add(text, texts[i].important);
957 times[i] = now;
958 }
959 reported = (reported & ~mask) | (current & mask);
960 }

1029 ekf.getVerticalPositionInnovationTestRatio()
1030 };

1032 for (uint8_t i = 0; i < EKF2_PREFLIGHT_INNOV_NB; i++) {

1034 ekf2_preflight.ratio[i] = NAN;

1036 ekf2_preflight.ratio[i] = ratio[i];

1043static void ins_ekf2_preflight(struct preflight_result_t *result)
1044{
1045 const auto &flags = ekf.control_status_flags();
1046 uint16_t fail_cnt = result->fail_cnt;
1047
1048 if (!flags.tilt_align) {
1049 preflight_error(result, "EKF2 attitude not aligned");
1050 }
1051 if (!flags.yaw_align) {
1052 preflight_error(result, "EKF2 heading not aligned");
1053 }
1054 if (ekf.fault_status().value != 0) {
1055 preflight_error(result, "EKF2 numerical fault 0x%x", (unsigned int)ekf.fault_status().value);
1056 }
1057 if (flags.mag_fault || flags.gnss_fault || flags.baro_fault || flags.gnss_hgt_fault || flags.gnss_yaw_fault) {
1058 preflight_error(result, "EKF2 sensor fault (mag %d, GNSS %d, baro %d, GNSS hgt %d, GNSS yaw %d)",
1059 flags.mag_fault, flags.gnss_fault, flags.baro_fault, flags.gnss_hgt_fault, flags.gnss_yaw_fault);
1060 }
1061 if (flags.inertial_dead_reckoning) {
1062 preflight_error(result, "EKF2 dead reckoning");
1063 }
1064
1065#if INS_EKF2_PREFLIGHT_POS
1066 const int32_t gnss_pos = static_cast<int32_t>(GnssCtrl::HPOS) | static_cast<int32_t>(GnssCtrl::VEL);
1067 if ((ekf_params->ekf2_gps_ctrl & gnss_pos) && !(flags.gnss_pos || flags.gnss_vel)) {
1068 static const char *const names[] = {"nsats", "pdop", "hacc", "vacc", "sacc", "hdrift", "vdrift", "hspeed",
1069 "vspeed", "spoofed", "fix", "jammed"
1070 };
1071 char failed[80] = "";
1072 uint16_t flags_fail = gnss_checks.getStrictFailFlags();
1073 for (uint8_t i = 0; i < sizeof(names) / sizeof(names[0]); i++) {
1074 if (flags_fail & (1 << i)) {
1075 size_t l = strlen(failed);
1076 snprintf(failed + l, sizeof(failed) - l, "%s%s", l ? "," : "", names[i]);
1077 }
1078 }
1079 preflight_error(result, "EKF2 not using GNSS (failed: %s)", failed[0] ? failed : "no data");
1080 }
1081 const int32_t ev_pos = static_cast<int32_t>(EvCtrl::HPOS) | static_cast<int32_t>(EvCtrl::VEL);
1082 if ((ekf_params->ekf2_ev_ctrl & ev_pos) && !(flags.ev_pos || flags.ev_vel)) {
1083 preflight_error(result, "EKF2 not using external vision");
1084 }
1085#endif
1086
1087 // Filtered innovation test ratios above 1 mean the measurements are rejected
1088 for (uint8_t i = 0; i < EKF2_PREFLIGHT_INNOV_NB; i++) {
1089 if (PX4_ISFINITE(ekf2_preflight.ratio[i]) && ekf2_preflight.ratio[i] > 1.f) {
1090 preflight_error(result, "EKF2 %s innovation too high (%.1f)", ekf2_preflight_innov_names[i], ekf2_preflight.ratio[i]);
1091 }
1092 }
1093
1094 if (flags.mag_field_disturbed) {
1095 preflight_warning(result, "EKF2 mag field disturbed");
1096 }
1097 if (result->fail_cnt == fail_cnt) {
1098 preflight_success(result, "EKF2 ok");
1099 }
1100}

1102#define ins_ekf2_preflight_filter() {}
1103#endif /* PREFLIGHT_CHECKS */

1289{
1290 bool in_air = autopilot_in_flight();

1371 return;
1372 }

1375

1377 if (ekf2.got_imu_data) {

1391 if (ekf.control_status_flags().tilt_align) {

1400 stateSetPositionNed_f(MODULE_INS_EKF2_ID, &pos);

1405 speed.x = vel_f(0);
1406 speed.y = vel_f(1);
1407 speed.z = vel_f(2);
1408

1410 stateSetSpeedNed_f(MODULE_INS_EKF2_ID, &speed);
1411

1413 ekf2.accel_ned.y = vel_deriv_f(1);

1418 accel.x = ekf2.accel_ned.x;
1419 accel.y = ekf2.accel_ned.y;

1428 uint64_t origin_time;
1429
1430 // Only update the origin when the state estimator has updated the origin
1431 if (ekf.global_origin_valid()) {
1432 ekf.getEkfGlobalOrigin(origin_time, ekf_origin_lat, ekf_origin_lon, ref_alt);
1433 } else {
1434 origin_time = 0;
1435 }
1436 if (origin_time > ekf2.ltp_stamp) {
1437#if INS_EKF2_FORCE_NAV_REF && USE_INS_NAV_INIT
1438 ekf2_log_origin_set(ekf2_time_us(get_sys_time_usec()), NAV_LAT0 * 1e-7, NAV_LON0 * 1e-7, (NAV_ALT0) * 1e-3);
1439 if (ekf.setEkfGlobalOrigin(NAV_LAT0 * 1e-7, NAV_LON0 * 1e-7, (NAV_ALT0) * 1e-3)) {
1440 struct LlaCoor_i llh_nav0; /* Height above the ellipsoid */
1441 llh_nav0.lat = NAV_LAT0;
1442 llh_nav0.lon = NAV_LON0;
1443 llh_nav0.alt = NAV_ALT0 + NAV_MSL0; // in millimeters above WGS84 reference ellipsoid
1444
1445 ltp_def_from_lla_i(&ekf2.ltp_def, &llh_nav0);
1446 ekf2.ltp_def.hmsl = NAV_ALT0;
1447 stateSetLocalOrigin_i(MODULE_INS_EKF2_ID, &ekf2.ltp_def);
1448 }
1449#else
1450 lla_ref.lat = ekf_origin_lat * 1e7; // WGS-84 lat
1451 lla_ref.lon = ekf_origin_lon * 1e7; // WGS-84 lon
1452 lla_ref.alt = ref_alt * 1e3 + wgs84_ellipsoid_to_geoid_i(lla_ref.lat, lla_ref.lon); // in millimeters above WGS84 reference ellipsoid (ref_alt is in HMSL)
1453 ltp_def_from_lla_i(&ekf2.ltp_def, &lla_ref);
1454 ekf2.ltp_def.hmsl = ref_alt * 1e3;
1455 stateSetLocalOrigin_i(MODULE_INS_EKF2_ID, &ekf2.ltp_def);
1456#endif
1457 /* update local ENU coordinates of global waypoints */
1458 waypoints_localize_all();
1459
1460 ekf2.ltp_stamp = origin_time;
1461 }
1462 }
1463 }

1466 if (nps_bypass_ins) {
1467 sim_overwrite_ins();
1468 }
1469#endif
1470
1471 ekf2.got_imu_data = false;
