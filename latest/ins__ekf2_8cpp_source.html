369static uint64_t ekf2_time_us(uint32_t stamp)
370{
371 static uint32_t last_now = 0;
372 static uint64_t wraps = 0;
373 uint32_t now = get_sys_time_usec();
374 if (now < last_now && (last_now - now) > (1UL << 31)) {
375 wraps += (1ULL << 32);
376 }
377 last_now = now;
378
379 // A stamp more than half the range later than now was taken before the last wrap
380 uint64_t time_us = wraps + stamp;
381 if (stamp > now && (stamp - now) > (1UL << 31) && time_us >= (1ULL << 32)) {
382 time_us -= (1ULL << 32);
383 }
384 return time_us;
385}
386
388static uint64_t ekf2_delayed_time_us(uint32_t stamp, uint32_t delay_ms)
389{
390 uint64_t time_us = ekf2_time_us(stamp);
391 return (time_us > delay_ms * 1000ULL) ? time_us - delay_ms * 1000ULL : 0;
392}

451static void ekf2_log_params(uint64_t time_us)
452{
453 for (int i = 0; i < ins_ekf2_params_nb; i += 32) {
454 float values[32];
455 uint8_t n = 0;
456 for (int j = i; j < ins_ekf2_params_nb && n < 32; j++) {

459 int32_t first = i;
460 ekf2_log(EKF2_IN_PARAMS, time_us, 1, &first, n, values);

491static void ekf2_log_check_start(void)
492{
493 if (ekf2_log_state != EKF2_LOG_WAITING) {
494 return;
495 }
496 if (ekf2_log_ready()) {
497 ekf2_log_state = EKF2_LOG_ON;
498 uint64_t now = ekf2_time_us(get_sys_time_usec());
499 ekf2_log_params(now);
500 float config[6] = {INS_EKF2_TAU_POS, INS_EKF2_TAU_VEL, INS_EKF2_GPS_YAW_RST,
501 INS_EKF2_RNG_MIN_DIST, INS_EKF2_RNG_MAX_DIST, INS_EKF2_OF_MAX_RATE};
502 ekf2_log(EKF2_IN_CONFIG, now, 0, NULL, 6, config);
503 if (ekf2_log_origin.set) {
504 ekf2_log_origin_set(now, ekf2_log_origin.lat, ekf2_log_origin.lon, ekf2_log_origin.alt);
505 }
506 } else if (get_sys_time_float() > INS_EKF2_LOG_WAIT) {
507 ekf2_log_state = EKF2_LOG_OFF;
508 }
509}

615 test_ratio(ekf.getVerticalVelocityInnovationTestRatio())));

949static void ekf2_event_bits(const struct ekf2_status_text *texts, uint8_t nb, float *times, T &reported, T current, float now)
950{
951 for (uint8_t i = 0; i < nb; i++) {
952 T mask = ((T)1) << texts[i].bit;
953 if ((current & mask) == (reported & mask) || (times[i] > 0.f && now - times[i] < INS_EKF2_EVENT_HOLDOFF)) {
954 continue;
955 }
956 const char *text = (current & mask) ? texts[i].on : texts[i].off;
957 if (text != NULL) {
958 ekf2_event_add(text, texts[i].important);
959 times[i] = now;
960 }
961 reported = (reported & ~mask) | (current & mask);
962 }

1038 ekf2_preflight.ratio[i] = ratio[i];

1046{
1047 const auto &flags = ekf.control_status_flags();
1048 uint16_t fail_cnt = result->fail_cnt;
1049
1050 if (!flags.tilt_align) {
1051 preflight_error(result, "EKF2 attitude not aligned");
1052 }
1053 if (!flags.yaw_align) {
1054 preflight_error(result, "EKF2 heading not aligned");
1055 }
1056 if (ekf.fault_status().value != 0) {
1057 preflight_error(result, "EKF2 numerical fault 0x%x", (unsigned int)ekf.fault_status().value);
1058 }
1059 if (flags.mag_fault || flags.gnss_fault || flags.baro_fault || flags.gnss_hgt_fault || flags.gnss_yaw_fault) {
1060 preflight_error(result, "EKF2 sensor fault (mag %d, GNSS %d, baro %d, GNSS hgt %d, GNSS yaw %d)",
1061 flags.mag_fault, flags.gnss_fault, flags.baro_fault, flags.gnss_hgt_fault, flags.gnss_yaw_fault);
1062 }
1063 if (flags.inertial_dead_reckoning) {
1064 preflight_error(result, "EKF2 dead reckoning");
1065 }
1066
1067#if INS_EKF2_PREFLIGHT_POS
1068 const int32_t gnss_pos = static_cast<int32_t>(GnssCtrl::HPOS) | static_cast<int32_t>(GnssCtrl::VEL);
1069 if ((ekf_params->ekf2_gps_ctrl & gnss_pos) && !(flags.gnss_pos || flags.gnss_vel)) {
1070 static const char *const names[] = {"nsats", "pdop", "hacc", "vacc", "sacc", "hdrift", "vdrift", "hspeed",
1071 "vspeed", "spoofed", "fix", "jammed"
1072 };
1073 char failed[80] = "";
1074 uint16_t flags_fail = gnss_checks.getStrictFailFlags();
1075 for (uint8_t i = 0; i < sizeof(names) / sizeof(names[0]); i++) {
1076 if (flags_fail & (1 << i)) {
1077 size_t l = strlen(failed);
1078 snprintf(failed + l, sizeof(failed) - l, "%s%s", l ? "," : "", names[i]);
1079 }
1080 }
1081 preflight_error(result, "EKF2 not using GNSS (failed: %s)", failed[0] ? failed : "no data");
1082 }
1083 const int32_t ev_pos = static_cast<int32_t>(EvCtrl::HPOS) | static_cast<int32_t>(EvCtrl::VEL);
1084 if ((ekf_params->ekf2_ev_ctrl & ev_pos) && !(flags.ev_pos || flags.ev_vel)) {
1085 preflight_error(result, "EKF2 not using external vision");
1086 }
1087#endif
1088
1089 // Filtered innovation test ratios above 1 mean the measurements are rejected
1090 for (uint8_t i = 0; i < EKF2_PREFLIGHT_INNOV_NB; i++) {
1091 if (PX4_ISFINITE(ekf2_preflight.ratio[i]) && ekf2_preflight.ratio[i] > 1.f) {
1092 preflight_error(result, "EKF2 %s innovation too high (%.1f)", ekf2_preflight_innov_names[i], ekf2_preflight.ratio[i]);
1093 }
1094 }
1095
1096 if (flags.mag_field_disturbed) {
1097 preflight_warning(result, "EKF2 mag field disturbed");
1098 }
1099 if (result->fail_cnt == fail_cnt) {
1100 preflight_success(result, "EKF2 ok");
1101 }
1102}

1291{

1294 // airspeed and sideslip fusion, no ground effect, airspeed for the yaw estimator: always for the fixedwing

1300 return;

1305 flags.armed = armed;

1310 | (flags.gnd_effect << 4) | (flags.constant_pos << 5) | (flags.in_transition << 6);

1463 }
1464 }
1465 }
