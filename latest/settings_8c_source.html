81int32_t settings_store(void)
82{
83#if USE_PERSISTENT_SETTINGS
84 if (settings_store_flag) {
85 /* from generated/settings.h */
86 persistent_settings_store();
87 if (!persistent_write((void *)&pers_settings, sizeof(struct PersistentSettings))) {
88 /* persistent write was successful */

116void settings_parse_msg_SETTING(struct link_device *dev, struct transport_tx *trans, uint8_t *buf)
117{
118#ifndef INTERMCU_FBW
119 if (DL_SETTING_ac_id(buf) != AC_ID) { return; }
120 uint8_t sender_id = pprzlink_get_msg_sender_id(buf);
121 uint8_t i = DL_SETTING_index(buf);
122 float var = DL_SETTING_value(buf);
123 DlSetting(i, var);
124 settings_send_dl_value(dev, trans, sender_id, i, var);
125#else
126 (void)dev;
127 (void)trans;
128 (void)buf;
129#endif
130}
