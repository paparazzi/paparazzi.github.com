109void light_scheduler_set_color(uint8_t color)
110{
111 light_scheduler_color = color;
112 if (color >= LIGHT_SCHEDULER_COLOR_FLAG_FIRST && color < LIGHT_SCHEDULER_COLOR_NB) {
113 light_scheduler_mode = LIGHT_SCHEDULER_MODE_COLOR;
114 } else if (color != LIGHT_SCHEDULER_COLOR_CUSTOM && color < LIGHT_SCHEDULER_COLOR_FLAG_FIRST) {
115 light_scheduler_red = light_scheduler_presets[color][0];
116 light_scheduler_green = light_scheduler_presets[color][1];
117 light_scheduler_blue = light_scheduler_presets[color][2];
118 light_scheduler_mode = LIGHT_SCHEDULER_MODE_COLOR;
119 }
120}

132void light_scheduler_periodic(void)
133{
134 uint32_t n, s0;
135#ifdef LIGHT_SCHEDULER_ALERT
136 if (LIGHT_SCHEDULER_ALERT) {
137 const uint8_t *c = light_scheduler_presets[LIGHT_SCHEDULER_ALERT_COLOR];
138 for (n = 0; n < WS2812_NB_LEDS; n++) {
139 light_scheduler_set(n, c[0], c[1], c[2]);
140 }
141 return;
142 }
143#endif
144
145 for (n = 0; n < WS2812_NB_LEDS; n++) {
146 if (light_scheduler_mode == LIGHT_SCHEDULER_MODE_COLOR &&
147 light_scheduler_color >= LIGHT_SCHEDULER_COLOR_FLAG_FIRST && light_scheduler_color < LIGHT_SCHEDULER_COLOR_NB) {
148 // flag: three equal bands, led n is in the band containing its center (5 leds: 2 | 1 | 2)
149 uint32_t band = ((2 * n + 1) * 3) / (2 * WS2812_NB_LEDS);
150 const uint8_t *c = light_scheduler_presets[light_scheduler_flags[light_scheduler_color - LIGHT_SCHEDULER_COLOR_FLAG_FIRST][band]];
151 light_scheduler_set(n, c[0], c[1], c[2]);
152 } else if (light_scheduler_mode == LIGHT_SCHEDULER_MODE_COLOR) {
153 light_scheduler_set(n, light_scheduler_red, light_scheduler_green, light_scheduler_blue);
154 } else if (light_scheduler_mode == LIGHT_SCHEDULER_MODE_OFF) {
155 light_ws2812_arch_set(n, 0, 0, 0);
156 } else if (WS2812_NB_LEDS == 32) {
157 if (n<=7) light_scheduler_set(n, 0,255,0); // LEFT = GREEN
158 else if (n<=15)
159 light_scheduler_set(n, 0,0,255); // BACK = BLUE
160 else if (n<=23)
161 light_scheduler_set(n, 255,0,0); // RIGHT = RED
162 else
163 light_scheduler_set(n, 255,255,255); // FRONT = WHITE
164 } else {
165 s0 = s + 10 * n;
166 light_scheduler_set(n, s0 % 255, (s0 + 85) % 255, (s0 + 170) % 255);
167 }
168 }
169 s += 10;
170}
