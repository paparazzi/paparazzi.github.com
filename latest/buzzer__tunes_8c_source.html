288static uint16_t rtttl_number(const char **p)
289{
290 uint16_t n = 0;
291 while (**p >= '0' && **p <= '9') {
292 n = n * 10 + (*(*p)++ - '0');
293 }
294 return n;
295}
296
298static bool rtttl_start(const char *str)
299{
300 const char *p = str;
301 while (*p && *p != ':') { p++; } // skip name
302 if (*p != ':') { return false; }
303 p++;
304 rtttl_d = 4; rtttl_o = 6; rtttl_bpm = 63;
305 while (*p && *p != ':') {
306 char key = *p;
307 if (key == 'd' || key == 'o' || key == 'b') {
308 p += 2; // skip "x="
309 uint16_t v = rtttl_number(&p);
310 if (key == 'd' && v > 0) { rtttl_d = v; }
311 if (key == 'o' && v > 0) { rtttl_o = v; }
312 if (key == 'b' && v > 0) { rtttl_bpm = v; }
313 } else {
314 p++;
315 }
316 }
317 if (*p != ':') { return false; }
318 rtttl = p + 1;
319 return true;
320}

323static bool rtttl_next(struct buzzer_note *n)
324{
325 // semitone frequencies of octave 4 [Hz], C to B
326 static const uint16_t octave4[12] = {262, 277, 294, 311, 330, 349, 370, 392, 415, 440, 466, 494};
327 static const int8_t semitone['h' - 'a' + 1] = {9, 11, 0, 2, 4, 5, 7, 11}; // a b c d e f g h(=b)
328 const char *p = rtttl;
329 while (*p == ' ' || *p == ',') { p++; }
330 if (*p == '\0') { return false; }
331 uint16_t d = rtttl_number(&p);
332 if (d == 0) { d = rtttl_d; }
333 char c = *p | 0x20; // lower case
334 if (c != 'p' && (c < 'a' || c > 'h')) { return false; }
335 p++;
336 int8_t semi = (c == 'p') ? -1 : semitone[c - 'a'];
337 if (*p == '#') { semi++; p++; }
338 bool dotted = false;
339 if (*p == '.') { dotted = true; p++; }
340 uint8_t o = rtttl_o;
341 if (*p >= '0' && *p <= '9') { o = *p++ - '0'; }
342 if (*p == '.') { dotted = true; p++; }
343 while (*p && *p != ',') { p++; } // ignore anything else
344 rtttl = p;
345
346 uint32_t ms = (240000UL / rtttl_bpm) / d; // whole note = 4 beats
347 if (dotted) { ms = ms * 3 / 2; }
348 n->ms = Min(ms, 65535);
349 if (semi < 0) {
350 n->freq = REST;
351 } else {
352 uint8_t oct = o + semi / 12; // B# goes to the next octave
353 uint32_t f = octave4[semi % 12];
354 n->freq = (oct >= 4) ? (f << (oct - 4)) : (f >> (4 - oct));
355 }
356 return true;
357}

360static bool next_note(struct buzzer_note *n)
361{
362 if (tune_notes != NULL) {
363 if (tune_notes->ms == 0) { return false; }
364 *n = *tune_notes++;
365 return true;
366 }
367 if (rtttl != NULL) {
368 return rtttl_next(n);
369 }
370 return false;
371}

373static void start_next_note(void)
374{
375 if (!next_note(&note)) {
376 buzzer_tunes_play(BUZZER_TUNES_NONE); // end of tune
377 return;
378 }
379 note_start_ms = get_sys_time_msec();
380 note_gap = false;
381 buzzer_arch_tone(note.freq);
382}

384void buzzer_tunes_play(uint8_t tune)
385{
386 playing = false;
387 tune_notes = NULL;
388 rtttl = NULL;
389 buzzer_arch_tone(0);
390 buzzer_tunes_tune = BUZZER_TUNES_NONE;
391 if (tune == BUZZER_TUNES_NONE || tune >= BUZZER_TUNES_NB) {
392 return;
393 }
394 if (tune >= BUZZER_TUNES_CUSTOM_1) {
395 if (!rtttl_start(buzzer_tunes_custom[tune - BUZZER_TUNES_CUSTOM_1])) {
396 return; // empty or invalid custom tune
397 }
398 } else {
399 tune_notes = buzzer_tunes_list[tune];
400 }
401 buzzer_tunes_tune = tune;
402 playing = true;
403 start_next_note();
404}

412void buzzer_tunes_periodic(void)
413{
414 if (!playing) {
415 return;
416 }
417 uint32_t elapsed = get_sys_time_msec() - note_start_ms;
418 if (!note_gap && elapsed + BUZZER_TUNES_GAP_MS >= note.ms) {
419 buzzer_arch_tone(0);
420 note_gap = true;
421 }
422 if (elapsed >= note.ms) {
423 start_next_note();
424 }
