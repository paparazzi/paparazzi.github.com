105 void sys_time_arch_init( void ) {
106  sys_time.cpu_ticks_per_sec = PCLK / T0_PCLK_DIV;
107  /* cpu ticks per desired sys_time timer step */
108  sys_time.resolution_cpu_ticks = (uint32_t)(sys_time.resolution * sys_time.cpu_ticks_per_sec + 0.5);
109 
110  /* setup Timer 0 to count forever */
111  /* reset & disable timer 0 */
112  T0TCR = TCR_RESET;
113  /* set the prescale divider */
114  T0PR = T0_PCLK_DIV - 1;
115  /* enable interrupt on match0 for sys_ticks */
116  T0MCR = TMCR_MR0_I;
117  /* disable capture registers */
118  T0CCR = 0;
119  /* disable external match register */
120  T0EMR = 0;
121 
122  /* set first sys tick interrupt */
123  /* We need to wait long enough to be sure
124  * that all the init part is finished before
125  * the first interrupt. Since the global
126  * interrupts are enable at the end of the init
127  * phase, if we miss the first one, the
128  * sys_tick_handler is not called afterward
129  */
130  T0MR0 = 4*sys_time.resolution_cpu_ticks;
131 
132  /* enable timer 0 */
133  T0TCR = TCR_ENABLE;
134 
135  /* select TIMER0 as IRQ */
136  VICIntSelect &= ~VIC_BIT(VIC_TIMER0);
137  /* enable TIMER0 interrupt */
138  VICIntEnable = VIC_BIT(VIC_TIMER0);
139  /* on slot vic slot 1 */
140  _VIC_CNTL(TIMER0_VIC_SLOT) = VIC_ENABLE | VIC_TIMER0;
141  /* address of the ISR */
142  _VIC_ADDR(TIMER0_VIC_SLOT) = (uint32_t)TIMER0_ISR;
143 }

146 // FIXME : nb_tick rollover ???
147 //
148 // 97 days at 512hz
149 // 12 hours at 100khz
150 //
151 static inline void sys_tick_irq_handler(void) {
152 
153  /* set match register for next interrupt */
154  T0MR0 += sys_time.resolution_cpu_ticks - 1;
155 
156  sys_time.nb_tick++;
157  sys_time.nb_sec_rem += sys_time.resolution_cpu_ticks;
158  if (sys_time.nb_sec_rem >= sys_time.cpu_ticks_per_sec) {
159  sys_time.nb_sec_rem -= sys_time.cpu_ticks_per_sec;
160  sys_time.nb_sec++;
161 #ifdef SYS_TIME_LED
162  LED_TOGGLE(SYS_TIME_LED);
163 #endif
164  }
165  for (unsigned int i=0; i<SYS_TIME_NB_TIMER; i++) {
166  if (sys_time.timer[i].in_use &&
167  sys_time.nb_tick >= sys_time.timer[i].end_time) {
168  sys_time.timer[i].end_time += sys_time.timer[i].duration;
169  sys_time.timer[i].elapsed = TRUE;
170  if (sys_time.timer[i].cb) {
171  sys_time.timer[i].cb(i);
172  }
173  }
174  }
175 }

177 void TIMER0_ISR ( void ) {
178  ISR_ENTRY();
179 
180  while (T0IR & TIMER0_IT_MASK) {
181 
182  if (T0IR&SYS_TICK_IT) {
183  sys_tick_irq_handler();
184  T0IR = SYS_TICK_IT;
185  }
186 
187 #if defined ACTUATORS && ( defined SERVOS_4017 || defined SERVOS_4015_MAT || defined SERVOS_PPM_MAT)
188  if (T0IR&ACTUATORS_IT) {
189 #ifdef SERVOS_4017
190  SERVOS_4017_ISR();
191 #endif
192 #ifdef SERVOS_4015_MAT
193  Servos4015Mat_ISR();
194 #endif
195 #ifdef SERVOS_PPM_MAT
196  ServosPPMMat_ISR();
197 #endif
198  T0IR = ACTUATORS_IT;
199  }
200 #endif /* ACTUATORS && (SERVOS_4017 || SERVOS_4015_MAT || SERVOS_PPM_MAT) */
201 
202 #if defined RADIO_CONTROL && defined RADIO_CONTROL_TYPE_PPM
203  if (T0IR&PPM_IT) {
204  PPM_ISR();
205  T0IR = PPM_IT;

214 #endif
215 #ifdef MB_SCALE

219  }
220 #endif
221 #ifdef MB_TACHO
