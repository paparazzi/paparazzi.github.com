68void pump_control_init(void)
69{
70 pump_control.time = 0.f;
71 pump_control.time_pumping = 0.f;
72 adc_buf_channel(PUMP_CONTROL_ADC, &pump_control.adc_b, DEFAULT_AV_NB_SAMPLE);
73 gpio_setup_output(PUMP_CONTROL_GPIO_ACTIVATE);
74 gpio_setup_output(PUMP_CONTROL_GPIO_PURGE);
75 gpio_clear(PUMP_CONTROL_GPIO_ACTIVATE);
76 gpio_clear(PUMP_CONTROL_GPIO_PURGE);
77}
78
79void pump_activate(void)
80{
81 pump_control.time = 0.f;
82 pump_control.time_pumping = 0.f;
83 gpio_set(PUMP_CONTROL_GPIO_ACTIVATE);
84 gpio_clear(PUMP_CONTROL_GPIO_PURGE);
85 pump_ctrl_state = PUMP_CTRL_PUMP;
86}
87
88void pump_purge(void)
89{
90 pump_control.time = 0.f;
91 pump_control.time_pumping = 0.f;
92 gpio_set(PUMP_CONTROL_GPIO_PURGE);
93 gpio_clear(PUMP_CONTROL_GPIO_ACTIVATE);
94 pump_ctrl_state = PUMP_CTRL_VALVE;
95}
96
97void pump_stop(void)
98{
99 gpio_clear(PUMP_CONTROL_GPIO_ACTIVATE);
100 gpio_clear(PUMP_CONTROL_GPIO_PURGE);
101 pump_ctrl_state = PUMP_CTRL_OFF;
102}
103
104bool pump_pumping(void) {
105 return adc_buf_get(&pump_control.adc_b) > PUMP_CONTROL_ADC_EMPTY;
106}
107
108bool pump_done(float duration, float timeout)
109{
110 pump_control.time += nav_dt;
111 if (pump_pumping()) {
112 pump_control.time_pumping += nav_dt;
113 }
114 if ((timeout > 0.f && pump_control.time > timeout)
115 || pump_control.time_pumping > duration) {
116 return true;
117 }
118 return false;
119}
120
121void pump_control_handler(float value) {
122 int val = value;
123 switch (val)
124 {
125 case PUMP_CTRL_OFF:
126 pump_stop();
127 break;
128 case PUMP_CTRL_PUMP:
129 pump_activate();
130 break;
131 case PUMP_CTRL_VALVE:
132 pump_purge();
133 break;
134
135 default:
136 break;
137 }
138}

140void pump_control_report(void)
141{
142 uint16_t val1 = adc_buf_get(&pump_control.adc_b);
143 uint16_t val2 = adc_buf_get(&pump_control.adc_b) > PUMP_CONTROL_ADC_EMPTY;
144
145 DOWNLINK_SEND_ADC_GENERIC(DefaultChannel, DefaultDevice,
146 &val1,
147 &val2);
148}
