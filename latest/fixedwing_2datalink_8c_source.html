84 {

87 #if 0 // not ready yet

90  /* parse telemetry messages coming from other AC */
91  if (sender_id != 0) {
92  switch (msg_id) {
93 #ifdef TCAS
94  case DL_TCAS_RA: {
95  if (DL_TCAS_RESOLVE_ac_id(dl_buffer) == AC_ID && SenderIdOfMsg(dl_buffer) != AC_ID) {
96  uint8_t ac_id_conflict = SenderIdOfMsg(dl_buffer);
97  tcas_acs_status[the_acs_id[ac_id_conflict]].resolve = DL_TCAS_RA_resolve(dl_buffer);
98  }
99  }
100 #endif
101  }
102  return;
103  }
104 #endif
105 
106  switch (msg_id) {
107 
108  case DL_PING: {
109  DOWNLINK_SEND_PONG(DefaultChannel, DefaultDevice);
110  }
111  break;
112 
113 #ifdef TRAFFIC_INFO
114  case DL_ACINFO: {
115  if (DL_ACINFO_ac_id(dl_buffer) == AC_ID) { break; }
116  uint8_t id = DL_ACINFO_ac_id(dl_buffer);
117  float ux = MOfCm(DL_ACINFO_utm_east(dl_buffer));
118  float uy = MOfCm(DL_ACINFO_utm_north(dl_buffer));
119  float a = MOfCm(DL_ACINFO_alt(dl_buffer));
120  float c = RadOfDeg(((float)DL_ACINFO_course(dl_buffer)) / 10.);
121  float s = MOfCm(DL_ACINFO_speed(dl_buffer));
122  float cl = MOfCm(DL_ACINFO_climb(dl_buffer));
123  uint32_t t = DL_ACINFO_itow(dl_buffer);
124  SetAcInfo(id, ux, uy, c, a, s, cl, t);
125  }
126  break;
127 #endif
128 
129 #ifdef NAV
130  case DL_MOVE_WP: {
131  if (DL_MOVE_WP_ac_id(dl_buffer) != AC_ID) { break; }
132  uint8_t wp_id = DL_MOVE_WP_wp_id(dl_buffer);
133  float a = MOfMM(DL_MOVE_WP_alt(dl_buffer));
134 
135  /* Computes from (lat, long) in the referenced UTM zone */
136  struct LlaCoor_f lla;
137  lla.lat = RadOfDeg((float)(DL_MOVE_WP_lat(dl_buffer) / 1e7));
138  lla.lon = RadOfDeg((float)(DL_MOVE_WP_lon(dl_buffer) / 1e7));
139  struct UtmCoor_f utm;
140  utm.zone = nav_utm_zone0;
141  utm_of_lla_f(&utm, &lla);
142  nav_move_waypoint(wp_id, utm.east, utm.north, a);
143 
144  /* Waypoint range is limited. Computes the UTM pos back from the relative
145  coordinates */
146  utm.east = waypoints[wp_id].x + nav_utm_east0;
147  utm.north = waypoints[wp_id].y + nav_utm_north0;
148  DOWNLINK_SEND_WP_MOVED(DefaultChannel, DefaultDevice, &wp_id, &utm.east, &utm.north, &a, &nav_utm_zone0);
149  }
150  break;
151 
152  case DL_BLOCK: {
153  if (DL_BLOCK_ac_id(dl_buffer) != AC_ID) { break; }
154  nav_goto_block(DL_BLOCK_block_id(dl_buffer));
155  SEND_NAVIGATION(&(DefaultChannel).trans_tx, &(DefaultDevice).device);
156  }
157  break;
158 #endif
161 #ifdef WIND_INFO
