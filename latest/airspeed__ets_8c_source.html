129void airspeed_ets_read_periodic(void)
130{
131#ifndef SITL
132 if (!airspeed_ets_delay_done) {
133 if (SysTimeTimer(airspeed_ets_delay_time) < USEC_OF_SEC(AIRSPEED_ETS_START_DELAY)) { return; }
134 else { airspeed_ets_delay_done = true; }
135 }
136 if (airspeed_ets_i2c_trans.status == I2CTransDone) {
137 i2c_receive(&AIRSPEED_ETS_I2C_DEV, &airspeed_ets_i2c_trans, AIRSPEED_ETS_ADDR, 2);
138 }
139#elif !defined USE_NPS
140 extern float sim_air_speed;
141 AbiSendMsgAIRSPEED_RAW(AIRSPEED_ETS_ID, get_sys_time_usec(), sim_air_speed);
142#endif //SITL
143}

145void airspeed_ets_read_event(void)
146{
147 float airspeed_tmp = 0.0;
148
149 // Get raw airspeed from buffer
150 airspeed_ets_raw = ((uint16_t)(airspeed_ets_i2c_trans.buf[1]) << 8) | (uint16_t)(airspeed_ets_i2c_trans.buf[0]);
151 // Check if this is valid airspeed
152 if (airspeed_ets_raw == 0) {
153 airspeed_ets_valid = false;
154 } else {
155 airspeed_ets_valid = true;
156 }
157
158 // Continue only if a new airspeed value was received
159 if (airspeed_ets_valid) {
160#if !AIRSPEED_ETS_3RD_PARTY_MODE
161
162 // Calculate offset average if not done already
163 if (!airspeed_ets_offset_init) {
164 --airspeed_ets_cnt;
165 // Check if averaging completed
166 if (airspeed_ets_cnt == 0) {
167 // Calculate average
168 airspeed_ets_offset = (uint16_t)(airspeed_ets_offset_tmp / AIRSPEED_ETS_OFFSET_NBSAMPLES_AVRG);
169 // Limit offset
170 if (airspeed_ets_offset < AIRSPEED_ETS_OFFSET_MIN) {
171 airspeed_ets_offset = AIRSPEED_ETS_OFFSET_MIN;
172 }
173 if (airspeed_ets_offset > AIRSPEED_ETS_OFFSET_MAX) {
174 airspeed_ets_offset = AIRSPEED_ETS_OFFSET_MAX;
175 }
176 airspeed_ets_offset_init = true;
177 }
178 // Check if averaging needs to continue
179 else if (airspeed_ets_cnt <= AIRSPEED_ETS_OFFSET_NBSAMPLES_AVRG) {
180 airspeed_ets_offset_tmp += airspeed_ets_raw;
181 }
182 }
183
184 // Convert raw to m/s
185#ifdef AIRSPEED_ETS_REVERSE
186 if (airspeed_ets_offset_init && airspeed_ets_raw < airspeed_ets_offset) {
187 airspeed_tmp = AIRSPEED_ETS_SCALE * sqrtf((float)(airspeed_ets_offset - airspeed_ets_raw)) - (float)AIRSPEED_ETS_OFFSET;
188 }
189#else
190 if (airspeed_ets_offset_init && airspeed_ets_raw > airspeed_ets_offset) {
191 airspeed_tmp = AIRSPEED_ETS_SCALE * sqrtf((float)(airspeed_ets_raw - airspeed_ets_offset)) - (float)AIRSPEED_ETS_OFFSET;
192 }
193#endif
194 else {
195 airspeed_tmp = 0.0;
196 }
197//use raw value for sensor set to third-party mode
198#else
199 airspeed_tmp = airspeed_ets_raw;
200#endif //AIRSPEED_ETS_3RD_PARTY_MODE
201
202 // Airspeed should always be positive
203 if (airspeed_tmp < 0.0) {
204 airspeed_tmp = 0.0;
205 }
206 // air_data filters it
207 airspeed_ets = airspeed_tmp;
208
209 // Publish airspeed sensor, air_data filters and calibrates it and sets the airspeed
210 AbiSendMsgAIRSPEED_RAW(AIRSPEED_ETS_ID, get_sys_time_usec(), airspeed_ets);
211 } else {
212 airspeed_ets = 0.0;
213 }
214
215
216#if AIRSPEED_ETS_SDLOG
217#ifndef SITL
218 if (pprzLogFile != -1) {
219 if (!log_airspeed_ets_started) {
220 sdLogWriteLog(pprzLogFile, "AIRSPEED_ETS: raw offset airspeed(m/s) GPS_fix TOW(ms) Week Lat(1e7deg) Lon(1e7deg) HMSL(mm) gpseed(cm/s) course(1e7rad) climb(cm/s)\n");
221 log_airspeed_ets_started = true;
222 }
223 sdLogWriteLog(pprzLogFile, "airspeed_ets: %d %d %8.4f %d %d %d %d %d %d %d %d %d\n",
224 airspeed_ets_raw, airspeed_ets_offset, airspeed_ets,
225 gps.fix, gps.tow, gps.week,
226 gps.lla_pos.lat, gps.lla_pos.lon, gps.hmsl,
227 gps.gspeed, gps.course, -gps.ned_vel.z);
228 }
229#endif
230#endif
231

233
234 // Transaction has been read
235 airspeed_ets_i2c_trans.status = I2CTransDone;
