236void nps_fdm_init(double dt)
237{
238 fdm.init_dt = dt; // JSBsim specific
239 fdm.curr_dt = dt; // JSBsim specific
240 fdm.nan_count = 0; // JSBsim specific
241
242#ifdef NPS_ACTUATOR_TIME_CONSTANTS
243 // Set up low-pass filter to simulate delayed actuator response
244 const float tau[NPS_COMMANDS_NB] = NPS_ACTUATOR_TIME_CONSTANTS;
245 for (int i = 0; i < NPS_COMMANDS_NB; i++) {
246 init_first_order_low_pass(&gazebo_actuators.lowpass[i], tau[i], dt, 0.f);
247 }
248#ifdef NPS_ACTUATOR_MAX_ANGULAR_MOMENTUM
249 // Set up high-pass filter to simulate spinup torque
250 const float Iwmax[NPS_COMMANDS_NB] = NPS_ACTUATOR_MAX_ANGULAR_MOMENTUM;
251 for (int i = 0; i < NPS_COMMANDS_NB; i++) {
252 init_first_order_high_pass(&gazebo_actuators.highpass[i], tau[i], dt, 0.f);
253 gazebo_actuators.max_ang_momentum[i] = Iwmax[i];
254 }
255#endif
256#endif
257}

265void nps_fdm_run_step(
266 bool launch __attribute__((unused)),
267 double *act_commands,
268 int commands_nb)
269{
270 // Initialize Gazebo if req'd.
271 // Initialization is peformed here instead of in nps_fdm_init because:
272 // - Video devices need to added at this point. Video devices have not been
273 // added yet when nps_fdm_init is called.
274 // - nps_fdm_init runs on a different thread then nps_fdm_run_step, which
275 // causes problems with Gazebo.
276 if (!gazebo_initialized) {
277 init_gazebo();
278 gazebo_read();
279#if NPS_SIMULATE_VIDEO
280 init_gazebo_video();
281#endif
282#if NPS_SIMULATE_LASER_RANGE_ARRAY
283 gazebo_init_range_sensors();
284#endif
285 gazebo_initialized = true;
286 }
287
288 // Update the simulation for a single timestep.
289 gazebo::runWorld(model->GetWorld(), 1);
290 gazebo::sensors::run_once();
291 gazebo_write(act_commands, commands_nb);
292 gazebo_read();
293#if NPS_SIMULATE_VIDEO
294 gazebo_read_video();
295#endif
296#if NPS_SIMULATE_LASER_RANGE_ARRAY
297 gazebo_read_range_sensors();
298#endif
299
300}

342static void init_gazebo(void)
343{
344 string gazebo_home = "/conf/simulator/gazebo/";
345 string pprz_home(getenv("PAPARAZZI_HOME"));
346 string gazebodir = pprz_home + gazebo_home;
347 cout << "Gazebo directory: " << gazebodir << endl;
348
349 if (getenv("ROS_MASTER_URI")) {
350 // Launch with ROS support
351 cout << "Add ROS plugins... ";
352 gazebo::addPlugin("libgazebo_ros_paths_plugin.so");
353 gazebo::addPlugin("libgazebo_ros_api_plugin.so");
354 cout << "ok" << endl;
355 }
356
357 if (!gazebo::setupServer(0, NULL)) {
358 cout << "Failed to start Gazebo, exiting." << endl;
359 std::exit(-1);
360 }
361
362 cout << "Add Paparazzi paths: " << gazebodir << endl;
363 gazebo::common::SystemPaths::Instance()->AddModelPaths(gazebodir + "models/");
364 sdf::addURIPath("model://", gazebodir + "models/");
365 sdf::addURIPath("world://", gazebodir + "world/");
366
367 cout << "Add TU Delft paths: " << pprz_home + "/sw/ext/tudelft_gazebo_models/" << endl;
368 gazebo::common::SystemPaths::Instance()->AddModelPaths(pprz_home + "/sw/ext/tudelft_gazebo_models/models/");
369 sdf::addURIPath("model://", pprz_home + "/sw/ext/tudelft_gazebo_models/models/");
370 sdf::addURIPath("world://", pprz_home + "/sw/ext/tudelft_gazebo_models/world/");
371
372 // get vehicles
373 string vehicle_uri = "model://" + string(NPS_GAZEBO_AC_NAME) + "/" + string(NPS_GAZEBO_AC_NAME) + ".sdf";
374 string vehicle_filename = sdf::findFile(vehicle_uri, false);
375 if (vehicle_filename.empty()) {
376 cout << "ERROR, could not find vehicle " + vehicle_uri << endl;
377 std::exit(-1);
378 }
379 cout << "Load vehicle: " << vehicle_filename << endl;
380 sdf::SDFPtr vehicle_sdf(new sdf::SDF());
381 sdf::init(vehicle_sdf);
382 if (!sdf::readFile(vehicle_filename, vehicle_sdf)) {
383 cout << "ERROR, could not read vehicle " + vehicle_filename << endl;
384 std::exit(-1);
385 }
386
387 // add or set up sensors before the vehicle gets loaded
388#if NPS_SIMULATE_VIDEO
389 // Cameras
390 sdf::ElementPtr link = vehicle_sdf->Root()->GetFirstElement()->GetElement("link");
391 while (link) {
392 if (link->Get<string>("name") == "front_camera" && link->GetElement("sensor")->Get<string>("name") == "mt9f002") {
393 if (NPS_MT9F002_SENSOR_RES_DIVIDER != 1) {
394 int w = link->GetElement("sensor")->GetElement("camera")->GetElement("image")->GetElement("width")->Get<int>();
395 int h = link->GetElement("sensor")->GetElement("camera")->GetElement("image")->GetElement("height")->Get<int>();
396 int env = link->GetElement("sensor")->GetElement("camera")->GetElement("lens")->GetElement("env_texture_size")->Get<int>();
397 link->GetElement("sensor")->GetElement("camera")->GetElement("image")->GetElement("width")->Set(w / NPS_MT9F002_SENSOR_RES_DIVIDER);
398 link->GetElement("sensor")->GetElement("camera")->GetElement("image")->GetElement("height")->Set(h / NPS_MT9F002_SENSOR_RES_DIVIDER);
399 link->GetElement("sensor")->GetElement("camera")->GetElement("lens")->GetElement("env_texture_size")->Set(env / NPS_MT9F002_SENSOR_RES_DIVIDER);
400 }
401 if (MT9F002_TARGET_FPS){
402 int fps = Min(MT9F002_TARGET_FPS, link->GetElement("sensor")->GetElement("update_rate")->Get<int>());
403 link->GetElement("sensor")->GetElement("update_rate")->Set(fps);
404 }
405 } else if (link->Get<string>("name") == "bottom_camera" && link->GetElement("sensor")->Get<string>("name") == "mt9v117") {
406 if (MT9V117_TARGET_FPS){
407 int fps = Min(MT9V117_TARGET_FPS, link->GetElement("sensor")->GetElement("update_rate")->Get<int>());
408 link->GetElement("sensor")->GetElement("update_rate")->Set(fps);
409 }
410 }
411 link = link->GetNextElement("link");
412 }
413#endif
414
415 // laser range array
416#if NPS_SIMULATE_LASER_RANGE_ARRAY
417 vehicle_sdf->Root()->GetFirstElement()->AddElement("include")->GetElement("uri")->Set("model://range_sensors");
418 sdf::ElementPtr range_joint = vehicle_sdf->Root()->GetFirstElement()->AddElement("joint");
419 range_joint->GetAttribute("name")->Set("range_sensor_joint");
420 range_joint->GetAttribute("type")->Set("fixed");
421 range_joint->GetElement("parent")->Set("chassis");
422 range_joint->GetElement("child")->Set("range_sensors::base");
423#endif
424
425 // get world
426 string world_uri = "world://" + string(NPS_GAZEBO_WORLD);
427 string world_filename = sdf::findFile(world_uri, false);
428 if (world_filename.empty()) {
429 cout << "ERROR, could not find world " + world_uri << endl;
430 std::exit(-1);
431 }
432 cout << "Load world: " << world_filename << endl;
433 sdf::SDFPtr world_sdf(new sdf::SDF());
434 sdf::init(world_sdf);
435 if (!sdf::readFile(world_filename, world_sdf)) {
436 cout << "ERROR, could not read world " + world_filename << endl;
437 std::exit(-1);
438 }
439
440 // add vehicles
441 world_sdf->Root()->GetFirstElement()->InsertElement(vehicle_sdf->Root()->GetFirstElement());
442
443 world_sdf->Write(pprz_home + "/var/gazebo.world");
444
445 gazebo::physics::WorldPtr world = gazebo::loadWorld(pprz_home + "/var/gazebo.world");
446 if (!world) {
447 cout << "Failed to open world, exiting." << endl;
448 std::exit(-1);
449 }
450
451 cout << "Get pointer to aircraft: " << NPS_GAZEBO_AC_NAME << endl;
452 model = world->ModelByName(NPS_GAZEBO_AC_NAME);
453 if (!model) {
454 cout << "Failed to find '" << NPS_GAZEBO_AC_NAME << "', exiting."
455 << endl;
456 std::exit(-1);
457 }
458
459 // Initialize sensors
460 gazebo::sensors::run_once(true);
461 gazebo::sensors::run_threads();
462 gazebo::runWorld(world, 1);
463 cout << "Sensors initialized..." << endl;
464
465 // Find sensors
466 // Contact sensor
467 gazebo::sensors::SensorManager *mgr = gazebo::sensors::SensorManager::Instance();
468 ct = static_pointer_cast < gazebo::sensors::ContactSensor > (mgr->GetSensor("contactsensor"));
469 ct->SetActive(true);
470 // Sonar
471 sonar = static_pointer_cast<gazebo::sensors::SonarSensor>(mgr->GetSensor("sonarsensor"));
472 if(sonar) {
473 cout << "Found sonar" << endl;
474 }
475
476 gazebo::physics::LinkPtr sonar_link = model->GetLink("sonar");
477 if (sonar_link) {
478 // Get a pointer to the sensor using its full name
479 if (sonar_link->GetSensorCount() != 1) {
480 cout << "ERROR: Link '" << sonar_link->GetName()
481 << "' should only contain 1 sensor!" << endl;
482 } else {
483 string name = sonar_link->GetSensorName(0);
484 sonar = static_pointer_cast< gazebo::sensors::SonarSensor > (mgr->GetSensor(name));
485 if (!sonar) {
486 cout << "ERROR: Could not get pointer to '" << name << "'!" << endl;
487 } else {
488 // Activate sensor
489 sonar->SetActive(true);
490 }
491 }
492 }
493
494 // Overwrite motor directions as defined by motor_mixing
495#ifdef MOTOR_MIXING_YAW_COEF
496 const double yaw_coef[] = MOTOR_MIXING_YAW_COEF;
497
498 for (uint8_t i = 0; i < NPS_COMMANDS_NB; i++) {
499 gazebo_actuators.torques[i] = -fabs(gazebo_actuators.torques[i]) * yaw_coef[i] / fabs(yaw_coef[i]);
500 gazebo_actuators.max_ang_momentum[i] = -fabs(gazebo_actuators.max_ang_momentum[i]) * yaw_coef[i] / fabs(yaw_coef[i]);
501 }
502#endif
503 cout << "Gazebo initialized successfully!" << endl;
504}

514static void gazebo_read(void)
515{
516 static ignition::math::Vector3d vel_prev;
517 static double time_prev;
518
519 gazebo::physics::WorldPtr world = model->GetWorld();
520 ignition::math::Pose3d pose = model->WorldPose(); // In LOCAL xyz frame
521 ignition::math::Vector3d vel = model->WorldLinearVel();
522 ignition::math::Vector3d ang_vel = model->WorldAngularVel();
523 gazebo::common::SphericalCoordinatesPtr sphere = world->SphericalCoords();
524 ignition::math::Quaterniond local_to_global_quat(0, 0, -sphere->HeadingOffset().Radian());
525
526 /* Fill FDM struct */
527 fdm.time = world->SimTime().Double();
528
529 // Find world acceleration by differentiating velocity
530 // model->GetWorldLinearAccel() does not seem to take the velocity_decay into account!
531 // Derivation of the velocity also follows the IMU implementation of Gazebo itself:
532 // https://bitbucket.org/osrf/gazebo/src/e26144434b932b4b6a760ddaa19cfcf9f1734748/gazebo/sensors/ImuSensor.cc?at=default&fileviewer=file-view-default#ImuSensor.cc-370
533 double dt = fdm.time - time_prev;
534 ignition::math::Vector3d accel = (vel - vel_prev) / dt;
535 vel_prev = vel;
536 time_prev = fdm.time;
537
538 // init_dt: unused
539 // curr_dt: unused
540 // on_ground: unused
541 // nan_count: unused
542
543 // Transform ltp definition to double for accuracy
544 struct LtpDef_d ltpdef_d;
545 ltpdef_d.ecef.x = state.ned_origin_f.ecef.x;
546 ltpdef_d.ecef.y = state.ned_origin_f.ecef.y;
547 ltpdef_d.ecef.z = state.ned_origin_f.ecef.z;
548 ltpdef_d.lla.lat = state.ned_origin_f.lla.lat;
549 ltpdef_d.lla.lon = state.ned_origin_f.lla.lon;
550 ltpdef_d.lla.alt = state.ned_origin_f.lla.alt;
551 for (int i = 0; i < 3 * 3; i++) {
552 ltpdef_d.ltp_of_ecef.m[i] = state.ned_origin_f.ltp_of_ecef.m[i];
553 }
554 ltpdef_d.hmsl = state.ned_origin_f.hmsl;
555
556 /* position */
557 fdm.ltpprz_pos = to_pprz_ned(sphere->PositionTransform(pose.Pos(), gazebo::common::SphericalCoordinates::LOCAL,
558 gazebo::common::SphericalCoordinates::GLOBAL)); // Allows Gazebo worlds rotated wrt north.
559 fdm.hmsl = -fdm.ltpprz_pos.z;
560 ecef_of_ned_point_d(&fdm.ecef_pos, &ltpdef_d, &fdm.ltpprz_pos);
561 lla_of_ecef_d(&fdm.lla_pos, &fdm.ecef_pos);
562
563 /* debug positions */
564 fdm.lla_pos_pprz = fdm.lla_pos; // Don't really care...
565 fdm.lla_pos_geod = fdm.lla_pos;
566 fdm.lla_pos_geoc = fdm.lla_pos;
567 fdm.lla_pos_geoc.lat = gc_of_gd_lat_d(fdm.lla_pos.lat, fdm.hmsl);
568
569 if(sonar) {
570 double agl = sonar->Range();
571 if (agl > sonar->RangeMax()) agl = -1.0;
572 fdm.agl = agl;
573 } else {
574 fdm.agl = pose.Pos().Z(); // TODO Measure with sensor
575 }
576
577 /* velocity */
578 fdm.ltp_ecef_vel = to_pprz_ned(sphere->VelocityTransform(vel, gazebo::common::SphericalCoordinates::LOCAL,
579 gazebo::common::SphericalCoordinates::GLOBAL));
580 fdm.ltpprz_ecef_vel = fdm.ltp_ecef_vel; // ???
581 fdm.body_ecef_vel = to_pprz_body(pose.Rot().RotateVectorReverse(vel)); // Note: unused
582 ecef_of_ned_vect_d(&fdm.ecef_ecef_vel, &ltpdef_d, &fdm.ltp_ecef_vel);
583
584 /* acceleration */
585 fdm.ltp_ecef_accel = to_pprz_ned(sphere->VelocityTransform(accel, gazebo::common::SphericalCoordinates::LOCAL,
586 gazebo::common::SphericalCoordinates::GLOBAL)); // Note: unused
587 fdm.ltpprz_ecef_accel = fdm.ltp_ecef_accel; // ???
588 fdm.body_ecef_accel = to_pprz_body(pose.Rot().RotateVectorReverse(accel));
589 fdm.body_inertial_accel = fdm.body_ecef_accel; // Approximate, unused.
590 fdm.body_accel = to_pprz_body(pose.Rot().RotateVectorReverse(accel - world->Gravity()));
591 ecef_of_ned_vect_d(&fdm.ecef_ecef_accel, &ltpdef_d, &fdm.ltp_ecef_accel);
592
593 /* attitude */
594 // ecef_to_body_quat: unused
595 fdm.ltp_to_body_eulers = to_global_pprz_eulers(local_to_global_quat * pose.Rot());

656static void gazebo_write(double act_commands[], int commands_nb)
657{
658 for (int i = 0; i < commands_nb; ++i) {
659 // Thrust setpoint
660 double sp = autopilot.motors_on ? act_commands[i] : 0.0; // Normalized thrust setpoint
661
662 // Actuator dynamics, forces and torques
663#ifdef NPS_ACTUATOR_TIME_CONSTANTS
664 // Delayed response from actuator
665 double u = update_first_order_low_pass(&gazebo_actuators.lowpass[i], sp);// Normalized actual thrust
666#else
667 double u = sp;
668#endif
669 double thrust = gazebo_actuators.thrusts[i] * u;
670 double torque = gazebo_actuators.torques[i] * u;
671
672#ifdef NPS_ACTUATOR_MAX_ANGULAR_MOMENTUM
673 // Spinup torque
674 double udot = update_first_order_high_pass(&gazebo_actuators.highpass[i], sp);
675 double spinup_torque = gazebo_actuators.max_ang_momentum[i] / (2.0 * sqrt(u > 0.05 ? u : 0.05)) * udot;
676 torque += spinup_torque;
677#endif
678
679 // Apply force and torque to gazebo model
680 gazebo::physics::LinkPtr link = model->GetLink(gazebo_actuators.names[i]);
681 link->AddRelativeForce(ignition::math::Vector3d(0, 0, thrust));
682 link->AddRelativeTorque(ignition::math::Vector3d(0, 0, torque));
683 }
684}

830static void read_image(struct image_t *img, gazebo::sensors::CameraSensorPtr cam)
831{
832 bool is_mt9f002 = false;
833 if (cam->Name() == "mt9f002") {
834 image_create(img, MT9F002_OUTPUT_WIDTH, MT9F002_OUTPUT_HEIGHT, IMAGE_YUV422);
835 is_mt9f002 = true;
836 } else {
837 image_create(img, cam->ImageWidth(), cam->ImageHeight(), IMAGE_YUV422);
838 }
839
840 // Convert Gazebo's *RGB888* image to Paparazzi's YUV422
841 const uint8_t *data_rgb = cam->ImageData();
842 uint8_t *data_yuv = (uint8_t *)(img->buf);
843 for (int x_yuv = 0; x_yuv < img->w; ++x_yuv) {
844 for (int y_yuv = 0; y_yuv < img->h; ++y_yuv) {
845 int x_rgb = x_yuv;
846 int y_rgb = y_yuv;
847 if (is_mt9f002) {
848 // Change sampling points for zoomed and/or cropped image.
849 // Use nearest-neighbour sampling for now.
850 x_rgb = (mt9f002.offset_x + ((float)x_yuv / img->w) * mt9f002.sensor_width)
851 / CFG_MT9F002_PIXEL_ARRAY_WIDTH * cam->ImageWidth();
852 y_rgb = (mt9f002.offset_y + ((float)y_yuv / img->h) * mt9f002.sensor_height)
853 / CFG_MT9F002_PIXEL_ARRAY_HEIGHT * cam->ImageHeight();
854 }
855 int idx_rgb = 3 * (cam->ImageWidth() * y_rgb + x_rgb);
856 int idx_yuv = 2 * (img->w * y_yuv + x_yuv);
857 int idx_px = img->w * y_yuv + x_yuv;
858 if (idx_px % 2 == 0) { // Pick U or V
859 data_yuv[idx_yuv] = - 0.148 * data_rgb[idx_rgb]
860 - 0.291 * data_rgb[idx_rgb + 1]
861 + 0.439 * data_rgb[idx_rgb + 2] + 128; // U
862 } else {
863 data_yuv[idx_yuv] = 0.439 * data_rgb[idx_rgb]
864 - 0.368 * data_rgb[idx_rgb + 1]
865 - 0.071 * data_rgb[idx_rgb + 2] + 128; // V
866 }
867 data_yuv[idx_yuv + 1] = 0.257 * data_rgb[idx_rgb]
868 + 0.504 * data_rgb[idx_rgb + 1]
869 + 0.098 * data_rgb[idx_rgb + 2] + 16; // Y
870 }
871 }
872 // Fill miscellaneous fields
873 gazebo::common::Time ts = cam->LastMeasurementTime();
874 img->ts.tv_sec = ts.sec;
875 img->ts.tv_usec = ts.nsec / 1000.0;
876 img->pprz_ts = ts.Double() * 1e6;
877 img->buf_idx = 0; // unused
878}

882/*
883 * Simulate range sensors:
884 *
885 * In the airframe file, set NPS_SIMULATE_LASER_RANGE_ARRAY if you want to make use of the integrated Ray sensors.
886 * These are defined in their own model which is added to the aircraft sdf (called range_sensors). Here you can add
887 * single ray point sensors with a specified position and orientation. It is also possible add noise.
888 *
889 * Within the airframe file, the amount of sensors
890 * (LASER_RANGE_ARRAY_NUM_SENSORS) and their orientations
891 * (LASER_RANGE_ARRAY_ORIENTATIONS={phi_1,theta_1,psi_1...phi_n,theta_n,psi_n} n = number of sensors) need to be
892 * specified as well. This is to keep it generic since this need to be done on the real platform with an external
893 * ray sensor. The function will compare the orientations from the ray sensors of gazebo, with the ones specified
894 * in the airframe, and will fill up an array to send and abi message to be used by other modules.
895 *
896 * NPS_GAZEBO_RANGE_SEND_AGL defines if the sensor that is defined as down should be used to send an AGL message.
897 * to send an OBSTACLE_DETECTION message.
898 *
899 * Functions:
900 * gazebo_init_range_sensors() -> Finds and initializes all ray sensors in gazebo
901 * gazebo_read_range_sensors() -> Reads and evaluates the ray sensors values, and sending it to other pprz modules
902 */
903
904struct gazebo_ray_t {
905 gazebo::sensors::RaySensorPtr sensor;
906 gazebo::common::Time last_measurement_time;
907};
908
909static struct gazebo_ray_t rays[LASER_RANGE_ARRAY_NUM_SENSORS] = {{NULL, 0}};
910static float range_orientation[] = LASER_RANGE_ARRAY_ORIENTATIONS;
911static uint8_t ray_sensor_agl_index = 255;
912
913#define VL53L0_MAX_VAL 8.191f

915static void gazebo_init_range_sensors(void)
916{
917 gazebo::sensors::SensorManager *mgr = gazebo::sensors::SensorManager::Instance();
918 gazebo::sensors::Sensor_V sensors = mgr->GetSensors(); // list of all sensors
919
920 uint16_t sensor_count = model->GetSensorCount();
921
922 gazebo::sensors::RaySensorPtr ray_sensor;
923 uint8_t ray_sensor_count_selected = 0;
924
925 // Loop though all sensors and only select ray sensors, which are saved within a struct
926 for (uint16_t i = 0; i < sensor_count; i++) {
927 if (ray_sensor_count_selected > LASER_RANGE_ARRAY_NUM_SENSORS) {
928 break;
929 }
930
931 if (sensors.at(i)->Type() == "ray") {
932 ray_sensor = static_pointer_cast<gazebo::sensors::RaySensor>(sensors.at(i));
933
934 if (!ray_sensor) {
935 cout << "ERROR: Could not get pointer to raysensor " << i << "!" << endl;
936 continue;
937 }
938
939 // Read out the pose from per ray sensors in gazebo relative to body
940 struct DoubleEulers pose_sensor = to_pprz_eulers(ray_sensor->Pose().Rot());
941
942 struct DoubleQuat q_ray, q_def;
943 double_quat_of_eulers(&q_ray, &pose_sensor);
944
945 /* Check the orientations of the ray sensors found in gazebo, if they are similar (within 5 deg) to the orientations
946 * given in the airframe file in LASER_RANGE_ARRAY_RANGE_ORIENTATION
947 */
948 for (int j = 0; j < LASER_RANGE_ARRAY_NUM_SENSORS; j++) {
949 struct DoubleEulers def = {0, range_orientation[j * 2], range_orientation[j * 2 + 1]};
950 double_quat_of_eulers(&q_def, &def);
951 // get angle between required angle and ray angle
952 double angle = acos(QUAT_DOT_PRODUCT(q_ray, q_def));
953
954 if (fabs(angle) < RadOfDeg(5)) {
955 ray_sensor_count_selected++;
956 rays[j].sensor = ray_sensor;
957 rays[j].sensor->SetActive(true);
958
959#if LASER_RANGE_ARRAY_SEND_AGL
960 // find the sensor pointing down
961 if (fabs(range_orientation[j * 2] + M_PI_2) < RadOfDeg(5)) {
962 ray_sensor_agl_index = j;
963 }
964#endif
965 break;
966 }
967 }
968 }
969 }
970
971 if (ray_sensor_count_selected != LASER_RANGE_ARRAY_NUM_SENSORS) {
972 cout << "ERROR: you have defined " << LASER_RANGE_ARRAY_NUM_SENSORS << " sensors in your airframe file, but only "
973 << ray_sensor_count_selected << " sensors have been found in the gazebo simulator, "
974 "with the same orientation as in the airframe file " << endl;
975 exit(0);
976 }
977 cout << "Initialized laser range array" << endl;
978}
