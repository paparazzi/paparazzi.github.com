67static bool sdp3x_crc(const uint8_t data[], unsigned size, uint8_t checksum)
68{
69 uint8_t crc_value = 0xff;
70
71 // calculate 8-bit checksum with polynomial 0x31 (x^8 + x^5 + x^4 + 1)
72 for (unsigned i = 0; i < size; i++) {
73 crc_value ^= (data[i]);
74
75 for (int bit = 8; bit > 0; --bit) {
76 if (crc_value & 0x80) {
77 crc_value = (crc_value << 1) ^ 0x31;
78
79 } else {
80 crc_value = (crc_value << 1);
81 }
82 }
83 }
84
85 // verify checksum
86 return (crc_value == checksum);
87}

99void sdp3x_periodic(void)
100{
101 if (sdp3x_trans.status != I2CTransDone) {
102 return; // not ready
103 }
104
105 if (sdp3x.initialized) {
106 // Initiate next read
107 i2c_receive(&SDP3X_I2C_DEV, &sdp3x_trans, SDP3X_I2C_ADDR, 6);
108 }
109 else {
110 // Init sensor in continuous mode
111 sdp3x_trans.buf[0] = SDP3X_MODE >> 8;
112 sdp3x_trans.buf[1] = SDP3X_MODE & 0xFF;
113 i2c_transmit(&SDP3X_I2C_DEV, &sdp3x_trans, SDP3X_I2C_ADDR, 2);
114 sdp3x.initialized = true;
115 }
116}

118void sdp3x_event(void)
119{
120 /* Check if transaction is succesfull */
121 if (sdp3x_trans.status == I2CTransSuccess) {
122
123 if (sdp3x.initialized) {
124 uint8_t buf[6];
125 for (uint8_t i = 0; i < 6; i++) {
126 buf[i] = sdp3x_trans.buf[i];
127 }
128
129 // Check the CRC
130 if (!sdp3x_crc(&buf[0], 2, buf[2]) || !sdp3x_crc(&buf[3], 2, buf[5])) {
131 // error
132 sdp3x_trans.status = I2CTransDone;
133 return;
134 }
135
136 int16_t p_raw = ((int16_t)(buf[0]) << 8) | (int16_t)(buf[1]);
137
138 float p_out = (float)p_raw / sdp3x.pressure_scale;
139
140 sdp3x.pressure = p_out;
141
142 int16_t t_raw = ((int16_t)(buf[3]) << 8) | (int16_t)(buf[4]);
143 sdp3x.temperature = (float)t_raw / SDP3X_SCALE_TEMPERATURE;
144
145 // Send (differential) pressure via ABI, air_data filters and zeroes it and computes the airspeed
146 uint32_t now_ts = get_sys_time_usec();
147 AbiSendMsgBARO_DIFF_RAW(SDP3X_SENDER_ID, now_ts, sdp3x.pressure);
148 // Send temperature as float in deg Celcius via ABI
149 AbiSendMsgTEMPERATURE_RAW(SDP3X_SENDER_ID, now_ts, sdp3x.temperature, TEMPERATURE_TYPE_DIFF);
150 }
151
152 // Set to done
153 sdp3x_trans.status = I2CTransDone;
154 } else if (sdp3x_trans.status == I2CTransFailed) {
155 // Just retry if failed
156 sdp3x_trans.status = I2CTransDone;
157 }
158}
