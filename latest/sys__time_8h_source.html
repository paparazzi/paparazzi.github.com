226static inline bool sys_time_check_and_ack_timer(tid_t id)
227{
228 if ((id < SYS_TIME_NB_TIMER) && (id >= 0)) {
229 if (sys_time.timer[id].elapsed) {
230 sys_time.timer[id].elapsed = false;
231 return true;
232 }
233 }
234 return false;
235}
