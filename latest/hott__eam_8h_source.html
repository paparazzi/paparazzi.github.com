131 static void hott_init_eam_msg(struct HOTT_EAM_MSG* hott_eam_msg) {
132  memset(hott_eam_msg, 0, sizeof(struct HOTT_EAM_MSG));
133  hott_eam_msg->start_byte = 0x7C;
134  hott_eam_msg->eam_sensor_id = HOTT_TELEMETRY_EAM_SENSOR_ID;
135  hott_eam_msg->sensor_id = 0xE0;
136  hott_eam_msg->stop_byte = 0x7D;
137 }

139 static void hott_update_eam_msg(struct HOTT_EAM_MSG* hott_eam_msg) {
140 
141  hott_eam_msg->batt1_voltage = electrical.vsupply;
142  hott_eam_msg->batt2_voltage = electrical.vsupply;
143  //hott_eam_msg->temp1 = 20 + imu.temperature / 10;
144  //hott_eam_msg->temp2 = 20 + imu.temperature / 10;
145  hott_eam_msg->altitude = (uint16_t)(500 + (stateGetPositionEnu_i()->z) / (1 << INT32_POS_FRAC));
146  hott_eam_msg->current = electrical.current / 100;
147  hott_eam_msg->main_voltage = electrical.vsupply;
148  hott_eam_msg->batt_cap = 0;
149  uint16_t speed_buf = (uint16_t)(*stateGetHorizontalSpeedNorm_i() * 36 / 10 / (1 << INT32_SPEED_FRAC));
150  hott_eam_msg->speed_L = speed_buf && 0xFF;
151  hott_eam_msg->speed_H = (speed_buf >> 8) && 0xFF;
152 
153  hott_eam_msg->climbrate = (uint16_t)(30000 + (stateGetSpeedEnu_i()->z) * 100 / (1 << INT32_SPEED_FRAC));
154  hott_eam_msg->climbrate3s = (uint8_t)(120 + (stateGetSpeedEnu_i()->z) * 3 / (1 << INT32_SPEED_FRAC));
155 
156  //display ON when motors are armed
157  if (autopilot_motors_on) {
158  hott_eam_msg->alarm_invers2 |= 0x80;
159  } else {
160  hott_eam_msg->alarm_invers2 &= 0x7f;
161  }
162 }
