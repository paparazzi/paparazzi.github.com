12/*
13 * One producer (write) and one consumer (read) can run in different threads: each only writes its own
14 * offset, with release semantics after the data, and reads the other one with acquire semantics.
15 */
16#define RB_LOAD(_x) __atomic_load_n(&(_x), __ATOMIC_ACQUIRE)
17#define RB_STORE(_x, _v) __atomic_store_n(&(_x), (_v), __ATOMIC_RELEASE)

27size_t ring_buffer_available(ring_buffer_t *rb) {
28 size_t read_offset = RB_LOAD(rb->read_offset);
29 size_t write_offset = RB_LOAD(rb->write_offset);
30 if (write_offset >= read_offset) {
31 return write_offset - read_offset;
32 }
33 else {
34 return rb->size - (read_offset - write_offset);
35 }
36}

38size_t ring_buffer_free_space(ring_buffer_t *ring_buffer) {
39 // One byte stays free: a full buffer would have the same offsets as an empty one
40 return ring_buffer->size - ring_buffer_available(ring_buffer) - 1;
41}

43size_t ring_buffer_write(ring_buffer_t *rb, const uint8_t *data, size_t len) {
44 size_t free_space = ring_buffer_free_space(rb);
45 if (len > free_space) {
46 len = free_space;
47 }
48
49 size_t first_chunk = rb->size - rb->write_offset;
50 if (first_chunk > len) {
51 first_chunk = len;
52 }
53
54 // Copy first chunk
55 memcpy(&rb->buf[rb->write_offset], data, first_chunk);
56
57 // Copy second chunk if wrap needed
58 size_t remaining = len - first_chunk;
59 if (remaining > 0) {
60 memcpy(&rb->buf[0], data + first_chunk, remaining);
61 }
62
63 RB_STORE(rb->write_offset, (rb->write_offset + len) % rb->size);
64
65 return len;
66}
