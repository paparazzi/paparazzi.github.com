101static void send_esc_telemetry(struct transport_tx *trans, struct link_device *dev)
102{
103 uint8_t idx[ACTUATORS_TELEMETRY_ESC_MAX], age[ACTUATORS_TELEMETRY_ESC_MAX];
104 int16_t rpm[ACTUATORS_TELEMETRY_ESC_MAX], current[ACTUATORS_TELEMETRY_ESC_MAX];
105 uint16_t voltage[ACTUATORS_TELEMETRY_ESC_MAX], errors[ACTUATORS_TELEMETRY_ESC_MAX];
106 int8_t temp[ACTUATORS_TELEMETRY_ESC_MAX], motor_temp[ACTUATORS_TELEMETRY_ESC_MAX];
107 uint8_t output[ACTUATORS_TELEMETRY_ESC_MAX];
108 struct act_feedback_set_t any = {0}; // values reported by at least one ESC
109 uint8_t nb = 0;
110 float now = get_sys_time_float();
111
112#ifdef ACTUATORS_TELEMETRY_ESC_IDX
113 static const uint8_t esc_idx[] = ACTUATORS_TELEMETRY_ESC_IDX;
114 const uint8_t nb_idx = sizeof(esc_idx) / sizeof(esc_idx[0]);
115#else
116 const uint8_t nb_idx = ACTUATORS_NB;
117#endif
118
119 for (uint8_t i = 0; i < nb_idx && nb < ACTUATORS_TELEMETRY_ESC_MAX; i++) {
120#ifdef ACTUATORS_TELEMETRY_ESC_IDX
121 const uint8_t a = esc_idx[i];
122 if (a >= ACTUATORS_NB) {
123 continue;
124 }
125#else
126 const uint8_t a = i;
127 if (!act_feedback_is_esc(&actuators_feedback[a])) {
128 continue;
129 }
130#endif
131 const struct act_feedback_t *t = &actuators_feedback[a];
132 ACT_FEEDBACK_ANY(any, t->set);
133 idx[nb] = a;
134 age[nb] = (actuators_feedback_time[a] > 0.f) ? (uint8_t)Min((now - actuators_feedback_time[a]) * 100.f, 255.f) : 255;
135 rpm[nb] = ACT_TELEM_SCALE(t, rpm, (float)t->rpm, 0.1f, INT16_MIN, INT16_MAX);
136 current[nb] = ACT_TELEM_SCALE(t, current, t->current, 100.f, INT16_MIN, INT16_MAX);
137 voltage[nb] = ACT_TELEM_SCALE(t, voltage, t->voltage, 100.f, 0, UINT16_MAX);
138 temp[nb] = ACT_TELEM_SCALE(t, temperature, t->temperature, 1.f, INT8_MIN, INT8_MAX);
139 motor_temp[nb] = ACT_TELEM_SCALE(t, motor_temperature, t->motor_temperature, 1.f, INT8_MIN, INT8_MAX);
140 output[nb] = t->set.output ? t->output_pct : 0;
141 errors[nb] = t->set.errors ? (uint16_t)Min(t->errors, UINT16_MAX) : 0;
142 nb++;
143 }
144
145 if (nb == 0) {
146 return;
147 }
148 pprz_msg_send_ESC_TELEMETRY(trans, dev, AC_ID, nb, idx, nb, age,
149 ACT_TELEM_NB(any, rpm, nb), rpm,
150 ACT_TELEM_NB(any, current, nb), current,
151 ACT_TELEM_NB(any, voltage, nb), voltage,
152 ACT_TELEM_NB(any, temperature, nb), temp,
153 ACT_TELEM_NB(any, motor_temperature, nb), motor_temp,
154 ACT_TELEM_NB(any, output, nb), output,
155 ACT_TELEM_NB(any, errors, nb), errors);
156}

224#define ACT_FEEDBACK_COPY(_to, _from, _field, _set) { if ((_from)->set._set) { (_to)->_field = (_from)->_field; (_to)->set._set = true; } }
225
227static void act_feedback_cb(uint8_t sender_id UNUSED, struct act_feedback_t *feedback, uint8_t num_act)
228{
229 float now = get_sys_time_float();
230 for (uint8_t i = 0; i < num_act; i++) {
231 const struct act_feedback_t *f = &feedback[i];
232 if (f->idx >= ACTUATORS_NB) {
233 continue;
234 }
235 struct act_feedback_t *t = &actuators_feedback[f->idx];
236 t->idx = f->idx;
237 ACT_FEEDBACK_COPY(t, f, rpm, rpm);
238 ACT_FEEDBACK_COPY(t, f, position, position);
239 ACT_FEEDBACK_COPY(t, f, force, force);
240 ACT_FEEDBACK_COPY(t, f, speed, speed);
241 ACT_FEEDBACK_COPY(t, f, power_pct, power);
242 ACT_FEEDBACK_COPY(t, f, current, current);
243 ACT_FEEDBACK_COPY(t, f, voltage, voltage);
244 ACT_FEEDBACK_COPY(t, f, temperature, temperature);
245 ACT_FEEDBACK_COPY(t, f, motor_temperature, motor_temperature);
246 ACT_FEEDBACK_COPY(t, f, output_pct, output);
247 ACT_FEEDBACK_COPY(t, f, errors, errors);
248 actuators_feedback_time[f->idx] = now;
249 }
250}

272static void cmd_actuator(shell_stream_t *sh, int argc, const char *const argv[])
273{
274 (void) argv;
275 if (argc == 0) {
276 chprintf(sh, "Usage: actuator [command] [options]\r\n");
277 chprintf(sh, " show [pprz_idx] Print list of actuators with their config or a single one\r\n");
278 chprintf(sh, " get_pprz pprz_idx Get current actuator value in pprz unit\r\n");
279 chprintf(sh, " get_driver driver_no servo_idx Get current actuator value in driver unit\r\n");
280 chprintf(sh, " set_pprz pprz_idx value Set actuator value in pprz unit (values are bounded to [-9600; 9600])\r\n");
281 chprintf(sh, " set_driver driver_no servo_idx value Set actuator value in driver unit (warning: values are not bounded)\r\n");
282 chprintf(sh, " set_conf pprz_idx min max neutral Set actuator min, max and neutral values in driver unit (warning: values are not bounded)\r\n");
283 return;
284 }
285
286 if (argc >= 1) {
287 if (strcmp(argv[0], "show") == 0) {
288 if (argc == 1) {
289 for (int i = 0; i < ACTUATORS_NB; i++) {
290 show_actuator(sh, i);
291 }
292 } else {
293 uint8_t idx = atoi(argv[1]);
294 show_actuator(sh, idx);
295 }
296 } else if (strcmp(argv[0], "get_pprz") == 0) {
297 if (argc >= 2) {
298 uint8_t idx = atoi(argv[1]);
299 chprintf(sh, "%d\r\n", actuators[idx].pprz_val);
300 }
301 } else if (strcmp(argv[0], "get_driver") == 0) {
302 if (argc >= 3) {
303 uint8_t d_idx = atoi(argv[1]);
304 uint8_t s_idx = atoi(argv[2]);
305 chprintf(sh, "%d\r\n", actuators[actuator_get_idx(d_idx, s_idx)].driver_val);
306 }
307 } else if (strcmp(argv[0], "set_pprz") == 0) {
308 if (argc >= 3) {
309 uint8_t idx = atoi(argv[1]);
310 int16_t val = atoi(argv[2]);
311 actuator_set(idx, val);
312 }
313 } else if (strcmp(argv[0], "set_driver") == 0) {
314 if (argc >= 4) {
315 uint8_t d_idx = atoi(argv[1]);
316 uint8_t s_idx = atoi(argv[2]);
317 uint8_t p_idx = actuator_get_idx(d_idx, s_idx);
318 int16_t val = atoi(argv[3]);
319 actuators[p_idx].driver_val = val;
320 if (actuators[p_idx].set != NULL) {
321 actuators[p_idx].set(actuators[p_idx].config.servo_idx, actuators[p_idx].driver_val);
322 }
323 }
324 } else if (strcmp(argv[0], "set_conf") == 0) {
325 if (argc >= 5) {
326 uint8_t idx = atoi(argv[1]);
327 int32_t min = atoi(argv[2]);
328 int32_t max = atoi(argv[3]);
329 int32_t neutral = atoi(argv[4]);
330 uint8_t d_idx = actuators[idx].config.driver_no;
331 uint8_t s_idx = actuators[idx].config.servo_idx;
332 actuator_set_min(d_idx, s_idx, min);
333 actuator_set_max(d_idx, s_idx, max);
334 actuator_set_neutral(d_idx, s_idx, neutral);
335 }
336 } else {
337 chprintf(sh, "unknown actuator command\r\n");
338 }
339 }
340}

418void actuator_set(uint8_t pprz_idx, pprz_t pprz_value)
419{
420 int32_t servo_value;
421 int32_t command_value;
422
423 actuators[pprz_idx].pprz_val = ClipAbs(pprz_value, MAX_PPRZ);
424 command_value = actuators[pprz_idx].pprz_val * (pprz_value>0 ? actuators[pprz_idx].config.travel_up : actuators[pprz_idx].config.travel_down);
425 servo_value = actuators[pprz_idx].config.neutral + command_value;
426 actuators[pprz_idx].driver_val = Clip(servo_value, actuators[pprz_idx].config.min, actuators[pprz_idx].config.max);
427 if (actuators[pprz_idx].set != NULL) {
428 actuators[pprz_idx].set(actuators[pprz_idx].config.servo_idx, actuators[pprz_idx].driver_val);
429 }
430}

479void actuator_set_min(uint8_t driver_no, uint8_t servo_idx, int32_t min)
480{
481 uint8_t idx = actuator_get_idx(driver_no, servo_idx);
482 if (idx == ACTUATORS_NB) return; // wrong index
483 actuators[idx].config.min = min;
484 int32_t delta = actuators[idx].config.neutral - min;
485 actuators[idx].config.travel_down = (float) delta / MAX_PPRZ;
486}
487
488void actuator_set_max(uint8_t driver_no, uint8_t servo_idx, int32_t max)
489{
490 uint8_t idx = actuator_get_idx(driver_no, servo_idx);
491 if (idx == ACTUATORS_NB) return; // wrong index
492 actuators[idx].config.max = max;
493 int32_t delta = max - actuators[idx].config.neutral;
494 actuators[idx].config.travel_up = (float) delta / MAX_PPRZ;
495}
496
497void actuator_set_neutral(uint8_t driver_no, uint8_t servo_idx, int32_t neutral)
498{
499 uint8_t idx = actuator_get_idx(driver_no, servo_idx);
500 if (idx == ACTUATORS_NB) return; // wrong index
501 actuators[idx].config.neutral = neutral;
502 int32_t delta_max = actuators[idx].config.max - neutral;
503 actuators[idx].config.travel_up = (float) delta_max / MAX_PPRZ;
504 int32_t delta_min = neutral - actuators[idx].config.min;
505 actuators[idx].config.travel_down = (float) delta_min / MAX_PPRZ;
506}
