68int pprz_bsem_wait_timeout(pprz_bsem_t* bsem, float timeout) {
69 struct timespec ts;
70 clock_gettime(CLOCK_REALTIME, &ts);
71 time_t timeout_secs = (time_t)timeout;
72 long timeout_nsecs = (timeout - timeout_secs) * 1e9;
73 ts.tv_sec += timeout_secs;
74 ts.tv_nsec += timeout_nsecs;
75 // tv_nsec must stay below 1 s, else sem_timedwait fails at once with EINVAL
76 if (ts.tv_nsec >= 1000000000L) {
77 ts.tv_sec += 1;
78 ts.tv_nsec -= 1000000000L;
79 }
80 return sem_timedwait(&bsem->sem, &ts);
81}

83void pprz_bsem_signal(pprz_bsem_t* bsem) {
84 int val;
85 pthread_mutex_lock(&bsem->mtx);
86 if(!sem_getvalue(&bsem->sem, &val)) {
87 if(val < 1) {
88 sem_post(&bsem->sem);
89 }
90 }
91 pthread_mutex_unlock(&bsem->mtx);
92}

103// stub function, ran in a new thread
104static void* stub(void* arg) {
105 struct stub_arg* sa = (struct stub_arg*) arg;
106 sa->func(sa->arg);
107 free(sa);

109}

111int pprz_thread_create(pprz_thread_t* thread, size_t size, const char *name, uint8_t prio, void (*func)(void*), void* arg) {
112 (void)size;
113 (void)prio;
114 struct stub_arg *sa = malloc(sizeof(struct stub_arg)); // allocate a stub arg. Must be allocated on the heap
115 sa->func = func;
116 sa->arg = arg;
117 int ret = pthread_create(thread, NULL, stub, (void*)sa); // launch the stub in a new thread
118 if(ret) {return ret;}
119 return pthread_setname_np(*thread, name);
