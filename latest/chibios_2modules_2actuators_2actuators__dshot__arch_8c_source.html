327 // update public structure with latest telemetry data
328 // and send actuator feedback message
329 struct act_feedback_t feedback[ACTUATORS_DSHOT_NB] = { 0 };
330 for (uint8_t i = 0; i < ACTUATORS_DSHOT_NB; i++) {
331 feedback[i].idx = get_servo_idx_DSHOT(i);
332 if (actuators_dshot_values[i].activated) {
333 DshotTelemetry dtelem = dshotGetTelemetry(actuators_dshot_private[i].driver, actuators_dshot_private[i].channel);
334 // RPM in telemetry is electrical RMP with a scale of 100,
335 // so physical value is 2 * 100 / motor_poles
336 const int32_t rpm = dtelem.frame.rpm * 200 / DSHOT_MOTOR_POLES;
337 feedback[i].rpm = rpm;
338 feedback[i].set.rpm = true;
339 actuators_dshot_values[i].rpm = (float)rpm;
340 actuators_dshot_values[i].current = (float)dtelem.frame.current * 0.01f;
341 actuators_dshot_values[i].voltage = (float)dtelem.frame.voltage * 0.01f;
342 actuators_dshot_values[i].energy = (float)dtelem.frame.consumption;
343 actuators_dshot_values[i].temp = (float)dtelem.frame.temp;
344 feedback[i].current = actuators_dshot_values[i].current;
345 feedback[i].set.current = true;
346 feedback[i].voltage = actuators_dshot_values[i].voltage;
347 feedback[i].set.voltage = true;
348 feedback[i].temperature = actuators_dshot_values[i].temp;
349 feedback[i].set.temperature = true;
350#if DSHOT_BIDIR
351 // overwrite RPM if DSHOT_BIDIR is used as it provides faster update
352 const uint32_t erpm = dshotGetRpm(actuators_dshot_private[i].driver, actuators_dshot_private[i].channel);
353 if ((erpm != DSHOT_BIDIR_ERR_CRC) && (erpm != DSHOT_BIDIR_TLM_EDT)) {
354 feedback[i].rpm = erpm * 2 / DSHOT_MOTOR_POLES;
355 feedback[i].set.rpm = true;
356 actuators_dshot_values[i].rpm = (float)feedback[i].rpm;
357 } else {
358 feedback[i].set.rpm = false;
359 }
360#endif
361 }
362 }
363 AbiSendMsgACT_FEEDBACK(ACT_FEEDBACK_DSHOT_ID, feedback, ACTUATORS_DSHOT_NB);
