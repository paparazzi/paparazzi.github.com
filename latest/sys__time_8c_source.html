64static uint64_t uptime_msec(void)
65{
66 uint32_t ms = get_sys_time_msec();
67 int64_t diff = (int64_t)sys_time.nb_sec * 1000 - ms;
68 uint64_t wraps = (uint64_t)((diff + (1LL << 31)) >> 32);
69 return (wraps << 32) + ms;
70}

77uint64_t get_sys_time_unix_msec(void)
78{
79 // 64-bit is not read/written atomically on 32-bit MCUs,

94static uint64_t msec_of_stamp(uint32_t stamp, uint64_t *now)
95{
96 *now = uptime_msec();
97 uint32_t age_ms = (get_sys_time_usec() - stamp) / 1000;
98 return *now - age_ms;
99}

103 sys_time.tow_offset = (tow % SYS_TIME_MSEC_PER_WEEK + SYS_TIME_MSEC_PER_WEEK - msec % SYS_TIME_MSEC_PER_WEEK)

107static void set_unix(uint64_t unix_ms, uint64_t msec, uint64_t now)
108{
109 sys_time.unix_offset = unix_ms - msec;
110 sys_time.unix_synced = true;
111
112 // Keep the RTC in sync
113 if (sys_time.rtc_sync == 0 || (uint32_t)now - sys_time.rtc_sync >= SYS_TIME_RTC_SYNC_PERIOD) {
114 sys_time.rtc_sync = (uint32_t)now;
115 uint64_t unix_now = sys_time.unix_offset + now;
116 sys_time_rtc_set((uint32_t)(unix_now / 1000), (uint16_t)(unix_now % 1000));
117 }
118}

120void sys_time_sync_gps(uint16_t week, uint32_t tow, uint32_t stamp)
121{
122 uint64_t now;
123 uint64_t msec = msec_of_stamp(stamp, &now);
124
125 set_tow(tow, msec);
126 sys_time.gps_tow_sync = (uint32_t)now;
127 sys_time.gps_tow_synced = true;
128
129 if (week != 0) {
130 uint64_t gps_ms = (uint64_t)week * SYS_TIME_MSEC_PER_WEEK + tow;
131 set_unix(gps_ms + SYS_TIME_UNIX_GPS_EPOCH_MS - SYS_TIME_GPS_LEAP_MS, msec, now);
132 sys_time.gps_unix_sync = (uint32_t)now;
133 sys_time.gps_unix_synced = true;
134 }
135}

137void sys_time_sync_unix(uint32_t unix_time, uint16_t unix_ms, uint32_t stamp)
138{
139 uint64_t unix_msec = (uint64_t)unix_time * 1000 + unix_ms;
140 if (unix_msec < SYS_TIME_UNIX_GPS_EPOCH_MS) {
141 return; // invalid time
142 }
143
144 uint64_t now;
145 uint64_t msec = msec_of_stamp(stamp, &now);
146
147 // GPS has priority
148 bool use_unix = !sys_time.gps_unix_synced || (uint32_t)now - sys_time.gps_unix_sync > SYS_TIME_GPS_SYNC_TIMEOUT;
149 bool use_tow = !sys_time.gps_tow_synced || (uint32_t)now - sys_time.gps_tow_sync > SYS_TIME_GPS_SYNC_TIMEOUT;
150 if (!use_unix && !use_tow) {
151 return;
152 }
153
154 // Only step when the difference with the last applied ground time is significant
155 uint64_t offset = unix_msec - msec;
156 int64_t diff = (int64_t)(offset - sys_time.ground_offset);
157 if (sys_time.ground_synced && diff < SYS_TIME_GROUND_SYNC_THRESHOLD && diff > -SYS_TIME_GROUND_SYNC_THRESHOLD) {
158 return;
159 }
160 sys_time.ground_offset = offset;
161 sys_time.ground_synced = true;
162
163 if (use_unix) {
164 set_unix(unix_msec, msec, now);
165 }
166 if (use_tow) {
167 uint64_t gps_ms = unix_msec - SYS_TIME_UNIX_GPS_EPOCH_MS + SYS_TIME_GPS_LEAP_MS;
168 set_tow((uint32_t)(gps_ms % SYS_TIME_MSEC_PER_WEEK), msec);
169 }
170}

172tid_t sys_time_register_timer(float duration, sys_time_cb cb)
173{
174 uint32_t start_time = sys_time.nb_tick;
175 for (tid_t i = 0; i < SYS_TIME_NB_TIMER; i++) {
176 if (!sys_time.timer[i].in_use) {
177 sys_time.timer[i].cb = cb;
178 sys_time.timer[i].elapsed = false;
179 sys_time.timer[i].end_time = start_time + sys_time_ticks_of_sec(duration);
180 sys_time.timer[i].duration = sys_time_ticks_of_sec(duration);
181 sys_time.timer[i].in_use = true;
182 return i;
183 }
184 }
185 return -1;
186}

188tid_t sys_time_register_timer_offset(tid_t timer, float offset, sys_time_cb cb)
189{
190 if (timer >=0 && timer < SYS_TIME_NB_TIMER) {
191 for (tid_t i = 0; i < SYS_TIME_NB_TIMER; i++) {
192 if (!sys_time.timer[i].in_use && (timer < i)) { // timer should be already registered
193 sys_time.timer[i].cb = cb;
194 sys_time.timer[i].elapsed = false;
195 sys_time.timer[i].end_time = sys_time.timer[timer].end_time + sys_time_ticks_of_sec(offset); // add offset to end time
196 sys_time.timer[i].duration = sys_time.timer[timer].duration; // copy duration
197 sys_time.timer[i].in_use = true;
198 return i;
199 }
200 }
201 }
202 return -1;
203}

232static void cmd_sys_time(shell_stream_t *sh, int argc, const char * const argv[])
233{
234 (void) argv;

236 chprintf(sh, "Usage: sys_time\r\n");

239
240 chprintf(sh, "Current sys_time\r\n");
241 chprintf(sh, " nb_sec: %lu\r\n", sys_time.nb_sec);
242 chprintf(sh, " nb_sec_rem: %lu\r\n", sys_time.nb_sec_rem);
243 chprintf(sh, " nb_tick: %lu\r\n", sys_time.nb_tick);
244 chprintf(sh, " tow: %lu\r\n", get_sys_time_tow());
245 chprintf(sh, " unix: %lu\r\n", get_sys_time_unix());
246
247 uint32_t rtc_ms = sys_time_rtc_millisecond();
248 uint32_t rtc_h = rtc_ms / 3600000;
249 uint32_t rtc_m = (rtc_ms % 3600000) / 60000;
250 float rtc_s = (rtc_ms % 60000) / 1000.0;
251 chprintf(sh, " rtc_time: %02u:%02u:%02.03f\r\n", rtc_h, rtc_m, rtc_s);
252
253 chprintf(sh, "Registered timers\r\n");
254 for (tid_t i = 0; i < SYS_TIME_NB_TIMER; i++) {
255 if (sys_time.timer[i].in_use) {
256 chprintf(sh, "id: %d, duration: %lu, end_time: %lu, cb: 0x%lx\r\n", i,
257 sys_time.timer[i].duration, sys_time.timer[i].end_time, sys_time.timer[i].cb);
258 }
259 }
260}

264{

266 sys_time.nb_sec_rem = 0;
267 sys_time.nb_tick = 0;
268 sys_time.tow_offset = 0;
269 sys_time.unix_offset = 0;

272 sys_time.gps_unix_sync = 0;

279 sys_time.ticks_per_sec = SYS_TIME_FREQUENCY;
280 sys_time.resolution = 1.0 / sys_time.ticks_per_sec;
281

285 sys_time.timer[i].elapsed = false;
286 sys_time.timer[i].end_time = 0;
287 sys_time.timer[i].duration = 0;
