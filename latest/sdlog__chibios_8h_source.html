63static inline const char *sdlog_error_string(SdioError error)
64{
65 switch (error) {
66 case SDLOG_OK: return "no error";
67 case SDLOG_NOCARD: return "no SD card";
68 case SDLOG_FATFS_ERROR: return "file system error";
69 case SDLOG_FATFS_NOENT: return "file or folder not found";
70 case SDLOG_FSFULL: return "SD card full";
71 case SDLOG_FDFULL: return "too many open files";
72 case SDLOG_QUEUEFULL: return "write queue full";
73 case SDLOG_MEMFULL: return "out of memory";
74 case SDLOG_NOTHREAD: return "logger thread not started";
75 case SDLOG_INTERNAL_ERROR: return "internal error";
76 case SDLOG_CANNOT_EXPAND: return "cannot reserve space on the SD card";
77 case SDLOG_LOGNUM_ERROR: return "too many log files";
78 case SDLOG_WAS_LAUNCHED: return "already started";
79 default: return "unknown error";
80 }
81}
