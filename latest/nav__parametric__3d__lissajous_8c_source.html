123bool nav_gvf_parametric_3D_lissajous_XYZ(float xo, float yo, float zo, float cx, float cy, float cz, float wx, float wy,
124 float wz, float dx, float dy, float dz, float alpha)
125{
126 // Safety first! If the asked altitude is low
127 if ((zo - cz) < 1) {
128 zo = 10;
129 cz = 0;
130 }
131
132 gvf_parametric_trajectory.type = LISSAJOUS_3D;
133 gvf_parametric_trajectory.p_parametric[0] = xo;
134 gvf_parametric_trajectory.p_parametric[1] = yo;
135 gvf_parametric_trajectory.p_parametric[2] = zo;
136 gvf_parametric_trajectory.p_parametric[3] = cx;
137 gvf_parametric_trajectory.p_parametric[4] = cy;
138 gvf_parametric_trajectory.p_parametric[5] = cz;
139 gvf_parametric_trajectory.p_parametric[6] = wx;
140 gvf_parametric_trajectory.p_parametric[7] = wy;
141 gvf_parametric_trajectory.p_parametric[8] = wz;
142 gvf_parametric_trajectory.p_parametric[9] = dx;
143 gvf_parametric_trajectory.p_parametric[10] = dy;
144 gvf_parametric_trajectory.p_parametric[11] = dz;
145 gvf_parametric_trajectory.p_parametric[12] = alpha;
146 gvf_parametric_trajectory.p_len = 13 + gvf_parametric_p_len_wps;
147 gvf_parametric_p_len_wps = 0;
148
149 float f1, f2, f3, f1d, f2d, f3d, f1dd, f2dd, f3dd;
150 float wb = gvf_parametric_control.w * gvf_parametric_control.beta * gvf_parametric_control.s;
151
152 gvf_parametric_3d_lissajous_info(&f1, &f2, &f3, &f1d, &f2d, &f3d, &f1dd, &f2dd, &f3dd, wb);
153 gvf_parametric_control_3D(gvf_parametric_3d_lissajous_par.kx, gvf_parametric_3d_lissajous_par.ky,
154 gvf_parametric_3d_lissajous_par.kz, f1, f2, f3, f1d, f2d, f3d, f1dd, f2dd, f3dd);
155
156 return true;
157}
