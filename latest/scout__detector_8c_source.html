41static void scout_detector_uavcan_cb(struct uavcan_iface_t *iface __attribute__((unused)), CanardRxTransfer *transfer) {
42 struct uavcan_protocol_debug_KeyValue msg;
43
44 if (uavcan_protocol_debug_KeyValue_decode(transfer, &msg)) {
45 return;
46 }
47
48 if (msg.key.len != 3) {
49 return;
50 }
51
52 if (strncmp("snr", (const char*)msg.key.data, 3) == 0) {
53 // first receive signal to noise value
54 scout_data.snr = msg.value;
55 } else if (strncmp("lit", (const char*)msg.key.data, 3) == 0) {
56 // lit (value in [0,1]) always received after snr
57 scout_data.lit = msg.value;
58 // FIXME better sync of data or single message !!!
59 float data = scout_data.snr * (1.f + scout_data.lit);
60 // update map
61 search_map_update(&scout_map, data, *stateGetPositionNed_f());
62 }
63}

81void scout_detector_sim(void) {
82#if (defined SITL) && (defined WP_SCOUT)
83 struct EnuCoor_f scout = *waypoint_get_enu_f(WP_SCOUT);
84 struct EnuCoor_f pos = *stateGetPositionEnu_f();
85 struct FloatVect3 dp;
86 VECT3_DIFF(dp, pos, scout);
87 float dist = float_vect3_norm(&dp);
88 // scout at 3 m ~ 95 dB
89 // sound level: L(d) = L(d0) - 20 * log10(d/d0)
90 // snr = 20 db at 3 m with propellers -> L(d0) = 20
91 float snr = 20. - 20.f*log10f(dist/3.f);
92 if (snr > 5.f) { // avoid to detect at too long range
93 scout_data.snr = snr;
94 } else {
95 scout_data.snr = 0.f;
96 }
97 search_map_update(&scout_map, scout_data.snr, *stateGetPositionNed_f());
98#endif
99}
