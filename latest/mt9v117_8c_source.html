45static bool i2c_checked(struct i2c_periph *p, struct i2c_transaction *t, uint8_t slave, uint16_t reg,
46 uint8_t len_w, uint16_t len_r, float timeout, const char *tag)
47{
48 for (int tries = 1; tries <= MT9V117_I2C_TRIES; tries++) {
49 for (int w = 0; w < 500 && (t->status == I2CTransPending || t->status == I2CTransRunning); w++) {
50 usleep(1000);
51 }
52 enum I2CTransactionStatus s;
53 if (len_r > 0) {
54 t->buf[0] = reg >> 8;
55 t->buf[1] = reg & 0xFF;
56 s = i2c_blocking_transceive(p, t, slave, len_w, len_r, timeout);
57 } else {
58 s = i2c_blocking_transmit(p, t, slave, len_w, timeout);
59 }
60 if (s == I2CTransSuccess) {
61 return true;
62 }
63 fprintf(stderr, "[%s] I2C %s of 0x%04x failed (status %d, try %d of %d)\n", tag,
64 len_r > 0 ? "read" : "write", reg, s, tries, MT9V117_I2C_TRIES);
65 usleep(10000);
66 }
67 return false;
68}

261static void write_reg(struct mt9v117_t *mt, uint16_t addr, uint32_t val, uint16_t len)
262{
263 mt->i2c_trans.buf[0] = addr >> 8;
264 mt->i2c_trans.buf[1] = addr & 0xFF;
265
266 // Fix sigdness based on length
267 if (len == 1) {
268 mt->i2c_trans.buf[2] = val & 0xFF;
269 } else if (len == 2) {
270 mt->i2c_trans.buf[2] = (val >> 8) & 0xFF;
271 mt->i2c_trans.buf[3] = val & 0xFF;
272 } else if (len == 4) {
273 mt->i2c_trans.buf[2] = (val >> 24) & 0xFF;
274 mt->i2c_trans.buf[3] = (val >> 16) & 0xFF;
275 mt->i2c_trans.buf[4] = (val >> 8) & 0xFF;
276 mt->i2c_trans.buf[5] = val & 0xFF;
277 } else {
278 printf("[MT9V117] write_reg with incorrect length %d\r\n", len);
279 }
280
281 // Transmit the buffer
282 i2c_checked(mt->i2c_periph, &mt->i2c_trans, MT9V117_ADDRESS, addr, len + 2, 0, 0.1, "MT9V117");
283}

288static uint32_t read_reg(struct mt9v117_t *mt, uint16_t addr, uint16_t len)
289{
290 uint32_t ret = 0;
291 mt->i2c_trans.buf[0] = addr >> 8;
292 mt->i2c_trans.buf[1] = addr & 0xFF;
293
294 // Transmit the buffer and receive back
295 i2c_checked(mt->i2c_periph, &mt->i2c_trans, MT9V117_ADDRESS, addr, 2, len, 0.1, "MT9V117");
296
297 /* Fix sigdness */
298 for (uint8_t i = 0; i < len; i++) {
299 ret |= mt->i2c_trans.buf[len - i - 1] << (8 * i);
300 }
301 return ret;
302}

304/* Write a byte to a var */
305static void write_var(struct mt9v117_t *mt, uint16_t var, uint16_t offset, uint32_t val, uint16_t len)
306{
307 uint16_t addr = 0x8000 | (var << 10) | offset;
308 write_reg(mt, addr, val, len);
309}

311/* Read a byte from a var */
312static uint32_t read_var(struct mt9v117_t *mt, uint16_t var, uint16_t offset, uint16_t len)
313{
314 uint16_t addr = 0x8000 | (var << 10) | offset;
315 return read_reg(mt, addr, len);
316}

318static inline void mt9v117_write_patch(struct mt9v117_t *mt)
319{
320 /* Errata item 2 */
321 write_reg(mt, 0x301a, 0x10d0, 2);
322 write_reg(mt, 0x31c0, 0x1404, 2);
323 write_reg(mt, 0x3ed8, 0x879c, 2);
324 write_reg(mt, 0x3042, 0x20e1, 2);
325 write_reg(mt, 0x30d4, 0x8020, 2);
326 write_reg(mt, 0x30c0, 0x0026, 2);
327 write_reg(mt, 0x301a, 0x10d4, 2);
328
329 /* Errata item 6 */
330 write_var(mt, MT9V117_AE_TRACK_VAR, 0x0002, 0x00d3, 2);
331 write_var(mt, MT9V117_CAM_CTRL_VAR, 0x0078, 0x00a0, 2);
332 write_var(mt, MT9V117_CAM_CTRL_VAR, 0x0076, 0x0140, 2);
333
334 /* Errata item 8 */
335 write_var(mt, MT9V117_LOW_LIGHT_VAR, 0x0004, 0x00fc, 2);
336 write_var(mt, MT9V117_LOW_LIGHT_VAR, 0x0038, 0x007f, 2);
337 write_var(mt, MT9V117_LOW_LIGHT_VAR, 0x003a, 0x007f, 2);
338 write_var(mt, MT9V117_LOW_LIGHT_VAR, 0x003c, 0x007f, 2);
339 write_var(mt, MT9V117_LOW_LIGHT_VAR, 0x0004, 0x00f4, 2);
340
341 /* Patch 0403; Critical; Sensor optimization */
342 write_reg(mt, MT9V117_ACCESS_CTL_STAT, 0x0001, 2);
343 write_reg(mt, MT9V117_PHYSICAL_ADDRESS_ACCESS, 0x7000, 2);
344
345 /* Write patch */
346 for (uint8_t i = 0; i < MT9V117_PATCH_LINE_NUM; ++i) {
347 // Copy buffer
348 for (uint8_t j = 0; j < mt9v117_patch_lines[i].len; ++j) {
349 mt->i2c_trans.buf[j] = mt9v117_patch_lines[i].data[j];
350 }
351
352 // Transmit the buffer
353 i2c_checked(mt->i2c_periph, &mt->i2c_trans, mt->i2c_trans.slave_addr, i, mt9v117_patch_lines[i].len, 0, 0.1, "MT9V117 patch");
354 }
355
356 write_reg(mt, MT9V117_LOGICAL_ADDRESS_ACCESS, 0x0000, 2);
357 write_var(mt, MT9V117_PATCHLDR_VAR, MT9V117_PATCHLDR_LOADER_ADDRESS_OFFSET, 0x05d8, 2);
358 write_var(mt, MT9V117_PATCHLDR_VAR, MT9V117_PATCHLDR_PATCH_ID_OFFSET, 0x0403, 2);
359 write_var(mt, MT9V117_PATCHLDR_VAR, MT9V117_PATCHLDR_FIRMWARE_ID_OFFSET, 0x00430104, 4);
360 write_reg(mt, MT9V117_COMMAND, MT9V117_COMMAND_OK | MT9V117_COMMAND_APPLY_PATCH, 2);
361
362 /* Wait for command OK */
363 for (uint8_t retries = 100; retries > 0; retries--) {
364 /* Wait 10ms */
365 usleep(10000);
366
367 /* Check the command */
368 uint16_t cmd = read_reg(mt, MT9V117_COMMAND, 2);
369 if ((cmd & MT9V117_COMMAND_APPLY_PATCH) == 0) {
370 if ((cmd & MT9V117_COMMAND_OK) == 0) {
371 printf("[MT9V117] Applying patch failed (No OK)\r\n");
372 }
373 return;
374 }
375 }
376
377 printf("[MT9V117] Applying patch failed after 10 retries\r\n");
378}

428void mt9v117_init(struct mt9v117_t *mt)
429{
430 /* bytes written to gpios/pwm */
431 int wc = 0;
432 /* Reset the device */
433 int gpio129 = open("/sys/class/gpio/gpio129/value", O_WRONLY | O_CREAT | O_TRUNC, 0666);
434 wc += write(gpio129, "0", 1);
435 wc += write(gpio129, "1", 1);
436 close(gpio129);
437
438 if (wc != 2) {
439 printf("[MT9V117] Couldn't write to GPIO 129\n");
440 }
441
442 /* Start PWM 9 (Which probably is the clock of the MT9V117) */
443 //#define BEBOP_CAMV_PWM_FREQ 43333333
444 int pwm9 = open("/sys/class/pwm/pwm_9/run", O_WRONLY | O_CREAT | O_TRUNC, 0666);
445 wc = 0;
446 wc += write(pwm9, "0", 1);
447 wc += write(pwm9, "1", 1);
448 close(pwm9);
449
450 if (wc != 2) {
451 printf("[MT9V117] Couldn't write to PWM\n");
452 }
453
454 //TODO: Make PWM and GPIO generic
455
456 /* Wait 50ms */
457 usleep(50000);
458
459 /* Setup i2c transaction */
460 mt->i2c_trans.status = I2CTransDone;
461
462 /* See if the device is there and correct */
463 uint16_t chip_id = read_reg(mt, MT9V117_CHIP_ID, 2);
464 if (chip_id != MT9V117_CHIP_ID_RESP) {
465 printf("[MT9V117] Didn't get correct response from CHIP_ID (expected: 0x%04X, got: 0x%04X)\r\n", MT9V117_CHIP_ID_RESP,
466 chip_id);
467 return;
468 }
469
470 /* Reset the device with software */
471 write_reg(mt, MT9V117_RESET_MISC_CTRL, MT9V117_RESET_SOC_I2C, 2);
472 write_reg(mt, MT9V117_RESET_MISC_CTRL, 0, 2);
473
474 /* Wait 50ms */
475 usleep(50000);
476
477 /* Apply MT9V117 software patch */
478 mt9v117_write_patch(mt);
479
480 /* Set basic settings */
481 write_var(mt, MT9V117_AWB_VAR, MT9V117_AWB_PIXEL_THRESHOLD_COUNT_OFFSET, 50000, 4);
482 write_var(mt, MT9V117_AE_RULE_VAR, MT9V117_AE_RULE_ALGO_OFFSET, MT9V117_AE_RULE_ALGO_AVERAGE, 2);
483
484 /* Set pixclk pad slew to 6 and data out pad slew to 1 */
485 write_reg(mt, MT9V117_PAD_SLEW, read_reg(mt, MT9V117_PAD_SLEW, 2) | 0x0600 | 0x0001, 2);
486
487 /* Configure the MT9V117 sensor */
488 mt9v117_config(mt);
489
490 /* Enable ITU656 */
491 write_var(mt, MT9V117_CAM_CTRL_VAR, MT9V117_CAM_OUTPUT_FORMAT_OFFSET,
492 read_var(mt, MT9V117_CAM_CTRL_VAR, MT9V117_CAM_OUTPUT_FORMAT_OFFSET, 2) |
493 MT9V117_CAM_OUTPUT_FORMAT_BT656_ENABLE, 2);
494
495 /* Set autoexposure luma */
496 write_var(mt, MT9V117_CAM_CTRL_VAR, MT9V117_AE_LUMA, MT9V117_TARGET_LUMA, 2);
497
498 /* Apply the configuration */
499 write_var(mt, MT9V117_SYSMGR_VAR, MT9V117_SYSMGR_NEXT_STATE_OFFSET, MT9V117_SYS_STATE_ENTER_CONFIG_CHANGE, 1);
500 write_reg(mt, MT9V117_COMMAND, MT9V117_COMMAND_OK | MT9V117_COMMAND_SET_STATE, 2);
501
502 /* Wait for command OK */
503 for (uint8_t retries = 100; retries > 0; retries--) {
504 /* Wait 10ms */
505 usleep(10000);
506
507 /* Check the command */
508 uint16_t cmd = read_reg(mt, MT9V117_COMMAND, 2);
509 if ((cmd & MT9V117_COMMAND_SET_STATE) == 0) {
510 if ((cmd & MT9V117_COMMAND_OK) == 0) {
511 printf("[MT9V117] Switching config failed (No OK)\r\n");
512 }
513
514 // Successfully configured!
515 //printf("[MT9V117] Switching config OK\r\n");
516 return;
517 }
518 }
519
520 printf("[MT9V117] Could not switch to new config\r\n");
521}
