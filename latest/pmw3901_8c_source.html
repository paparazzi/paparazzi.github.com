54// Blocking read/write functions
55static uint8_t readRegister_blocking(struct pmw3901_t *pmw, uint8_t addr) {
56 spi_slave_select(OPTICFLOW_PMW3901_SPI_SLAVE_IDX);
57
58 sys_time_usleep(10);
59
60 pmw->trans.output_buf[0] = addr & 0x7F; // MSB 0 => read
61 pmw->trans.output_length = 1;
62 pmw->trans.input_length = 0;
63 pmw->trans.select = SPINoSelect;
64 spi_blocking_transceive(pmw->periph, &pmw->trans, 0.5);
65
66 pmw->trans.output_buf[0] = 0;
67 pmw->trans.output_length = 1;
68 pmw->trans.input_length = 1;
69 pmw->trans.select = SPINoSelect;
70 spi_blocking_transceive(pmw->periph, &pmw->trans, 0.5);
71
72 spi_slave_unselect(OPTICFLOW_PMW3901_SPI_SLAVE_IDX);
73 return pmw->trans.input_buf[0];
74
75}

77static void writeRegister_blocking(struct pmw3901_t *pmw, uint8_t addr, uint8_t data) {
78 spi_slave_select(OPTICFLOW_PMW3901_SPI_SLAVE_IDX);
79
80 sys_time_usleep(35);
81
82 pmw->trans.output_buf[0] = addr | 0x80; // MSB 1 => write
83 pmw->trans.output_length = 1;
84 pmw->trans.input_length = 0;
85 pmw->trans.select = SPINoSelect;
86 spi_blocking_transceive(pmw->periph, &pmw->trans, 0.5);
87
88 pmw->trans.output_buf[0] = data;
89 pmw->trans.output_length = 1;
90 pmw->trans.input_length = 0;
91 pmw->trans.select = SPINoSelect;
92 spi_blocking_transceive(pmw->periph, &pmw->trans, 0.5);
93 sys_time_usleep(35);
94 spi_slave_unselect(OPTICFLOW_PMW3901_SPI_SLAVE_IDX);
95 sys_time_usleep(200);
96}

211static void pmw3901_thd(void* arg) {
212 struct pmw3901_t* pmw = (struct pmw3901_t*) arg;
213
214 while(true) {
215
216 pprz_bsem_wait(&pmw->bsem); // wait to be woken up by the AP thread
217
218 uint8_t tp = readRegister_blocking(pmw, PMW3901_REG_MOTION);
219 uint8_t xh = readRegister_blocking(pmw, PMW3901_REG_DELTA_X_H);
220 uint8_t xl = readRegister_blocking(pmw, PMW3901_REG_DELTA_X_L);
221 uint8_t yh = readRegister_blocking(pmw, PMW3901_REG_DELTA_Y_H);
222 uint8_t yl = readRegister_blocking(pmw, PMW3901_REG_DELTA_Y_L);
223
224 pmw->delta_x = (xh << 8) | xl;
225 pmw->delta_y = (yh << 8) | yl;
226 pmw->data_available = true;
227 }
228
229}

232void pmw3901_init(struct pmw3901_t *pmw, struct spi_periph *periph, uint8_t slave_idx) {
233 // Set up SPI peripheral and transaction
234 pmw->periph = periph;
235 pmw->trans.input_buf = pmw->spi_input_buf;
236 pmw->trans.output_buf = pmw->spi_output_buf;
237 pmw->trans.slave_idx = slave_idx;
238 pmw->trans.select = SPINoSelect;
239 pmw->trans.cpol = SPICpolIdleHigh;
240 pmw->trans.cpha = SPICphaEdge2;
241 pmw->trans.dss = SPIDss8bit;
242 pmw->trans.bitorder = SPIMSBFirst;
243 pmw->trans.cdiv = PMW3901_SPI_CDIV;
244 pmw->trans.before_cb = NULL;
245 pmw->trans.after_cb = NULL;
246 pmw->trans.status = SPITransDone;
247 // Initialize sensor registers
248 initializeSensor(pmw);
249 // Set up remaining fields
250 pmw->state = PMW3901_IDLE;
251 pmw->delta_x = 0;
252 pmw->delta_y = 0;
253 pmw->data_available = false;
254 pmw->rad_per_px = PMW3901_RAD_PER_PX;
255
256 pprz_bsem_init(&pmw->bsem, true);
257 pprz_thread_create(&pmw->thd_handle, 1024, "pmw3901", PPRZ_NORMAL_PRIO+1, pmw3901_thd, pmw);
258}
