81void baro_bmp280_event(void)
82{
83 bmp280_event(&baro_bmp280);
84
85 if (baro_bmp280.data_available) {
86 uint32_t now_ts = get_sys_time_usec();
87 // send ABI message
88 AbiSendMsgBARO_ABS_RAW(BARO_BMP_SENDER_ID, now_ts, baro_bmp280.pressure);
89 AbiSendMsgTEMPERATURE_RAW(BARO_BMP_SENDER_ID, now_ts, baro_bmp280.temperature, TEMPERATURE_TYPE_BARO);
90 baro_bmp280.data_available = false;
91 baro_alt = pprz_isa_altitude_of_pressure(baro_bmp280.pressure);
92 baro_alt_valid = true;
93 baro_press = (float)baro_bmp280.pressure;
94 baro_temp = ((float)baro_bmp280.temperature) / 100;
95 }
