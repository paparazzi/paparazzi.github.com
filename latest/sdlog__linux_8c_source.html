132static void sdlog_write(FileDes fd, const uint8_t *data, size_t len)
133{
134 FILE *f = sdlog_file(fd);
135 if (f == NULL) {
136 return;
137 }
138 if (fwrite(data, 1, len, f) != len) {
139 sdlog_error = sdlog_errno_error(errno);
140 sdlog_status = SDLOG_ERROR;
141 return;
142 }
143 sdlog_bytes += len;
144
145 float now = get_sys_time_float();
146 if (now - sdlog_last_flush > SDLOG_AUTO_FLUSH_PERIOD || now < sdlog_last_flush) {
147 sdlog_last_flush = now;
148 sdLogFlushAllLogs();
149 }
150}

153SdioError sdLogWriteLog(const FileDes fd, const char *fmt, ...)
154{
155 char buf[SDLOG_MAX_MSG_LEN];
156 va_list ap;
157 va_start(ap, fmt);
158 int len = vsnprintf(buf, sizeof(buf), fmt, ap);
159 va_end(ap);
160 if (len < 0) {
161 return SDLOG_INTERNAL_ERROR;
162 }
163 if ((size_t)len >= sizeof(buf)) {
164 len = sizeof(buf) - 1;
165 }
166 sdlog_write(fd, (uint8_t *)buf, len);
167 return sdlog_error;
168}

260static int sdlog_mkdir(const char *path)
261{
262 char tmp[256];
263 snprintf(tmp, sizeof(tmp), "%s", path);
264 for (char *c = tmp + 1; *c; c++) {
265 if (*c == '/') {
266 *c = '\0';
267 if (mkdir(tmp, 0755) != 0 && errno != EEXIST) {
268 return -1;
269 }
270 *c = '/';
271 }
272 }
273 if (mkdir(tmp, 0755) != 0 && errno != EEXIST) {
274 return -1;
275 }
276 return 0;
277}

280static int sdlog_next_number(const char *dir, const char *prefix)
281{
282 int next = 1;
283 size_t plen = strlen(prefix);
284 DIR *d = opendir(dir);
285 if (d == NULL) {
286 return next;
287 }
288 struct dirent *e;
289 while ((e = readdir(d)) != NULL) {
290 int num;
291 char ext[8];
292 if (strncmp(e->d_name, prefix, plen) != 0 ||
293 sscanf(e->d_name + plen, "%d.%7s", &num, ext) != 2 || strcmp(ext, "LOG") != 0) {
294 continue;
295 }
296 char path[512];
297 struct stat st;
298 snprintf(path, sizeof(path), "%s/%s", dir, e->d_name);
299 if (stat(path, &st) == 0 && st.st_size == 0) {
300 remove(path);
301 } else if (num >= next) {
302 next = num + 1;
303 }
304 }
305 closedir(d);
306 return next;
307}

310static FileDes sdlog_open(const char *base, const char *dir, const char *prefix, char *name, size_t name_len)
311{
312 FileDes fd = -1;
313 for (int i = 0; i < SDLOG_MAX_FILES; i++) {
314 if (sdlog_files[i] == NULL) {
315 fd = i;
316 break;
317 }
318 }
319 if (fd < 0) {
320 sdlog_error = SDLOG_FDFULL;
321 return -1;
322 }
323
324 char path[256];
325 // a truncated path would silently log into another folder
326 if (snprintf(path, sizeof(path), "%s/%s", base, dir) >= (int)sizeof(path) || sdlog_mkdir(path) != 0) {
327 sdlog_error = SDLOG_NOCARD;
328 return -1;
329 }
330
331 int num = sdlog_next_number(path, prefix);
332 if (num > 9999) {
333 sdlog_error = SDLOG_LOGNUM_ERROR;
334 return -1;
335 }
336 snprintf(name, name_len, "%s%04d.LOG", prefix, num);
337
338 char filename[512];
339 snprintf(filename, sizeof(filename), "%s/%s", path, name);
340 sdlog_files[fd] = fopen(filename, "wb");
341 if (sdlog_files[fd] == NULL) {
342 sdlog_error = SDLOG_FATFS_ERROR;
343 return -1;
344 }
345 printf("[sdlog] Logging to %s\n", filename);
346 return fd;
347}

371 err = errno;

411static void sdlog_report_status(struct transport_tx *trans, struct link_device *dev)
412{
413 static int reported = SDLOG_STOPPED;
414 int status = sdlog_status;
415 // only on a change: the start (stopped to running or error), and every time it goes into error
416 if (status == reported || status == SDLOG_STOPPED || (status == SDLOG_RUNNING && reported == SDLOG_ERROR)) {
417 reported = status;
418 return;
419 }
420 reported = status;
421 char msg[120];
422 int len;
423 if (status == SDLOG_ERROR) {
424 len = snprintf(msg, sizeof(msg), "SD log error: %s", sdlog_error_string(sdlog_error));
425 } else if (sdlog_conf_error != SDLOG_OK) {
426 len = snprintf(msg, sizeof(msg), "SD log: configuration not written: %s", sdlog_error_string(sdlog_conf_error));
427 } else {
428 return; // a normal start is not reported
429 }
430 if (len > 0) {
431 len = Min(len, (int)sizeof(msg) - 1);
432 pprz_msg_send_INFO_MSG(trans, dev, AC_ID, len, msg);
433 }
434}

465
466 char base[256];

468 if (dir == NULL) {
469 const char *home = getenv("PAPARAZZI_HOME");
470 snprintf(base, sizeof(base), "%s/var/logs/sdlog/%s", home ? home : ".", AIRFRAME_NAME);
471 } else {
472 snprintf(base, sizeof(base), "%s", dir);
473 }
474
475 char name[32];
476 sdlog_error = SDLOG_OK;
477 pprzLogFile = sdlog_open(base, PPRZ_LOG_DIR, PPRZ_LOG_NAME, name, sizeof(name));
478 snprintf(sdlog_filenames, sizeof(sdlog_filenames), "%s", pprzLogFile != -1 ? name : "");
479#if SDLOG_CONF
480 if (pprzLogFile != -1) {
481 sdlog_conf_error = sdlog_write_conf(base, PPRZ_LOG_DIR);
482 }
483#endif
484#if FLIGHTRECORDER_SDLOG
485 flightRecorderLogFile = sdlog_open(base, FR_LOG_DIR, FLIGHTRECORDER_LOG_NAME, name, sizeof(name));
486 if (flightRecorderLogFile != -1) {
487 size_t l = strlen(sdlog_filenames);
488 snprintf(sdlog_filenames + l, sizeof(sdlog_filenames) - l, "%s%s", l > 0 ? "," : "", name);
489 }
490#endif
491
492 if (sdlog_error == SDLOG_OK) {
493 sdlog_status = SDLOG_RUNNING;
494 } else {
495 printf("[sdlog] Could not open the log files in %s (error %d)\n", base, sdlog_error);
496 sdlog_status = SDLOG_ERROR;
497 }
498 sdlog_last_flush = get_sys_time_float();
499 atexit(sdlog_atexit);
500}

502void sdlog_chibios_finish(const bool flush)
503{
504 pprzLogFile = -1;
505#if FLIGHTRECORDER_SDLOG
506 flightRecorderLogFile = -1;
507#endif
508 for (int i = 0; i < SDLOG_MAX_FILES; i++) {
509 if (sdlog_files[i] != NULL) {
510 if (flush) {
511 fflush(sdlog_files[i]);
512 }
513 fclose(sdlog_files[i]);
514 sdlog_files[i] = NULL;
515 }
516 }
517 sdlog_status = SDLOG_STOPPED;
518}
