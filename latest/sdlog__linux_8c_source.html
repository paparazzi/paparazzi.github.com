110static void sdlog_write(FileDes fd, const uint8_t *data, size_t len)
111{
112 FILE *f = sdlog_file(fd);
113 if (f == NULL) {
114 return;
115 }
116 if (fwrite(data, 1, len, f) != len) {
117 sdlog_error = SDLOG_FSFULL;
118 sdlog_status = SDLOG_ERROR;
119 return;
120 }
121 sdlog_bytes += len;
122
123 float now = get_sys_time_float();
124 if (now - sdlog_last_flush > SDLOG_AUTO_FLUSH_PERIOD || now < sdlog_last_flush) {
125 sdlog_last_flush = now;
126 sdLogFlushAllLogs();
127 }
128}

131SdioError sdLogWriteLog(const FileDes fd, const char *fmt, ...)
132{
133 char buf[SDLOG_MAX_MSG_LEN];
134 va_list ap;
135 va_start(ap, fmt);
136 int len = vsnprintf(buf, sizeof(buf), fmt, ap);
137 va_end(ap);
138 if (len < 0) {
139 return SDLOG_INTERNAL_ERROR;
140 }
141 if ((size_t)len >= sizeof(buf)) {
142 len = sizeof(buf) - 1;
143 }
144 sdlog_write(fd, (uint8_t *)buf, len);
145 return sdlog_error;
146}

238static int sdlog_mkdir(const char *path)
239{
240 char tmp[256];
241 snprintf(tmp, sizeof(tmp), "%s", path);
242 for (char *c = tmp + 1; *c; c++) {
243 if (*c == '/') {
244 *c = '\0';
245 if (mkdir(tmp, 0755) != 0 && errno != EEXIST) {
246 return -1;
247 }
248 *c = '/';
249 }
250 }
251 if (mkdir(tmp, 0755) != 0 && errno != EEXIST) {
252 return -1;
253 }
254 return 0;
255}

258static int sdlog_next_number(const char *dir, const char *prefix)
259{
260 int next = 1;
261 size_t plen = strlen(prefix);
262 DIR *d = opendir(dir);
263 if (d == NULL) {
264 return next;
265 }
266 struct dirent *e;
267 while ((e = readdir(d)) != NULL) {
268 int num;
269 char ext[8];
270 if (strncmp(e->d_name, prefix, plen) != 0 ||
271 sscanf(e->d_name + plen, "%d.%7s", &num, ext) != 2 || strcmp(ext, "LOG") != 0) {
272 continue;
273 }
274 char path[512];
275 struct stat st;
276 snprintf(path, sizeof(path), "%s/%s", dir, e->d_name);
277 if (stat(path, &st) == 0 && st.st_size == 0) {
278 remove(path);
279 } else if (num >= next) {
280 next = num + 1;
281 }
282 }
283 closedir(d);
284 return next;
285}

288static FileDes sdlog_open(const char *base, const char *dir, const char *prefix, char *name, size_t name_len)
289{
290 FileDes fd = -1;
291 for (int i = 0; i < SDLOG_MAX_FILES; i++) {
292 if (sdlog_files[i] == NULL) {
293 fd = i;
294 break;
295 }
296 }
297 if (fd < 0) {
298 sdlog_error = SDLOG_FDFULL;
299 return -1;
300 }
301
302 char path[256];
303 // a truncated path would silently log into another folder
304 if (snprintf(path, sizeof(path), "%s/%s", base, dir) >= (int)sizeof(path) || sdlog_mkdir(path) != 0) {
305 sdlog_error = SDLOG_NOCARD;
306 return -1;
307 }
308
309 int num = sdlog_next_number(path, prefix);
310 if (num > 9999) {
311 sdlog_error = SDLOG_LOGNUM_ERROR;
312 return -1;
313 }
314 snprintf(name, name_len, "%s%04d.LOG", prefix, num);
315
316 char filename[512];
317 snprintf(filename, sizeof(filename), "%s/%s", path, name);
318 sdlog_files[fd] = fopen(filename, "wb");
319 if (sdlog_files[fd] == NULL) {
320 sdlog_error = SDLOG_FATFS_ERROR;
321 return -1;
322 }
323 printf("[sdlog] Logging to %s\n", filename);
324 return fd;
325}

378 if (dir == NULL) {
379 const char *home = getenv("PAPARAZZI_HOME");
380 snprintf(base, sizeof(base), "%s/var/logs/sdlog/%s", home ? home : ".", AIRFRAME_NAME);
381 } else {
382 snprintf(base, sizeof(base), "%s", dir);
383 }
384
385 char name[32];
386 sdlog_error = SDLOG_OK;
387 pprzLogFile = sdlog_open(base, PPRZ_LOG_DIR, PPRZ_LOG_NAME, name, sizeof(name));
388 snprintf(sdlog_filenames, sizeof(sdlog_filenames), "%s", pprzLogFile != -1 ? name : "");
389#if FLIGHTRECORDER_SDLOG
390 flightRecorderLogFile = sdlog_open(base, FR_LOG_DIR, FLIGHTRECORDER_LOG_NAME, name, sizeof(name));
391 if (flightRecorderLogFile != -1) {
392 size_t l = strlen(sdlog_filenames);
393 snprintf(sdlog_filenames + l, sizeof(sdlog_filenames) - l, "%s%s", l > 0 ? "," : "", name);
394 }
395#endif
396
397 if (sdlog_error == SDLOG_OK) {
398 sdlog_status = SDLOG_RUNNING;
399 } else {
400 printf("[sdlog] Could not open the log files in %s (error %d)\n", base, sdlog_error);
401 sdlog_status = SDLOG_ERROR;
402 }
403 sdlog_last_flush = get_sys_time_float();
404 atexit(sdlog_atexit);
405}

407void sdlog_chibios_finish(const bool flush)
408{
409 pprzLogFile = -1;
410#if FLIGHTRECORDER_SDLOG
411 flightRecorderLogFile = -1;
412#endif
413 for (int i = 0; i < SDLOG_MAX_FILES; i++) {
414 if (sdlog_files[i] != NULL) {
415 if (flush) {
416 fflush(sdlog_files[i]);
417 }
418 fclose(sdlog_files[i]);
419 sdlog_files[i] = NULL;
420 }
421 }
422 sdlog_status = SDLOG_STOPPED;
423}
