49void buzzer_arch_init(void)

57void buzzer_arch_tone(uint16_t freq)
58{
59 if (freq == 0) {
60 pwmDisableChannel(&BUZZER_PWMD, BUZZER_PWM_CHANNEL);
61 return;
62 }
63 pwmcnt_t period = BUZZER_PWM_FREQUENCY / freq;
64 pwmChangePeriod(&BUZZER_PWMD, period);
65 pwmEnableChannel(&BUZZER_PWMD, BUZZER_PWM_CHANNEL, period / 2);
66}
