174 static void ac_char_update(float throttle, float pitch, float climb, float accelerate)
175 {
176  if ((accelerate > -0.02) && (accelerate < 0.02)) {
177  if (throttle >= 1.0f) {
178  ac_char_climb_count++;
179  ac_char_average(&ac_char_climb_pitch, pitch * 57.6f, ac_char_climb_count);
180  ac_char_average(&ac_char_climb_max , stateGetSpeedEnu_f()->z, ac_char_climb_count);
181  } else if (throttle <= 0.0f) {
182  ac_char_descend_count++;
183  ac_char_average(&ac_char_descend_pitch, pitch * 57.6f , ac_char_descend_count);
184  ac_char_average(&ac_char_descend_max , stateGetSpeedEnu_f()->z , ac_char_descend_count);
185  } else if ((climb > -0.125) && (climb < 0.125)) {
186  ac_char_cruise_count++;
187  ac_char_average(&ac_char_cruise_throttle , throttle , ac_char_cruise_count);
188  ac_char_average(&ac_char_cruise_pitch , pitch * 57.6f , ac_char_cruise_count);
189  }
190  }
191 }

269 void v_ctl_altitude_loop(void)
270 {
271  // Airspeed Command Saturation
272  if (v_ctl_auto_airspeed_setpoint <= STALL_AIRSPEED * 1.23) { v_ctl_auto_airspeed_setpoint = STALL_AIRSPEED * 1.23; }
273 
274  // Altitude Controller
275  v_ctl_altitude_error = v_ctl_altitude_setpoint - stateGetPositionUtm_f()->alt;
276  float sp = v_ctl_altitude_pgain * v_ctl_altitude_error + v_ctl_altitude_pre_climb ;
277 
278  // Vertical Speed Limiter
279  BoundAbs(sp, v_ctl_max_climb);
280 
281  // Vertical Acceleration Limiter
282  float incr = sp - v_ctl_climb_setpoint;
283  BoundAbs(incr, 2 * dt_navigation);
284  v_ctl_climb_setpoint += incr;
285 }

291 static float low_pass_vdot(float v);
292 static float low_pass_vdot(float v)
293 {
294  lp_vdot[4] += (v - lp_vdot[4]) / 3;
295  lp_vdot[3] += (lp_vdot[4] - lp_vdot[3]) / 3;
296  lp_vdot[2] += (lp_vdot[3] - lp_vdot[2]) / 3;
297  lp_vdot[1] += (lp_vdot[2] - lp_vdot[1]) / 3;
298  lp_vdot[0] += (lp_vdot[1] - lp_vdot[0]) / 3;
299 
300  return lp_vdot[0];
301 }

307 void v_ctl_climb_loop(void)
308 {
309  // Airspeed setpoint rate limiter:
310  // AIRSPEED_SETPOINT_SLEW in m/s/s - a change from 15m/s to 18m/s takes 3s with the default value of 1
311  float airspeed_incr = v_ctl_auto_airspeed_setpoint - v_ctl_auto_airspeed_setpoint_slew;
312  BoundAbs(airspeed_incr, AIRSPEED_SETPOINT_SLEW * dt_attidude);
313  v_ctl_auto_airspeed_setpoint_slew += airspeed_incr;
314 
315 #ifdef V_CTL_AUTO_GROUNDSPEED_SETPOINT
316 // Ground speed control loop (input: groundspeed error, output: airspeed controlled)
317  float err_groundspeed = (v_ctl_auto_groundspeed_setpoint - (*stateGetHorizontalSpeedNorm_f()));
318  v_ctl_auto_groundspeed_sum_err += err_groundspeed;
319  BoundAbs(v_ctl_auto_groundspeed_sum_err, V_CTL_AUTO_GROUNDSPEED_MAX_SUM_ERR);
320  v_ctl_auto_airspeed_controlled = (err_groundspeed + v_ctl_auto_groundspeed_sum_err * v_ctl_auto_groundspeed_igain) *
321  v_ctl_auto_groundspeed_pgain;
322 
323  // Do not allow controlled airspeed below the setpoint
324  if (v_ctl_auto_airspeed_controlled < v_ctl_auto_airspeed_setpoint_slew) {
325  v_ctl_auto_airspeed_controlled = v_ctl_auto_airspeed_setpoint_slew;
326  // reset integrator of ground speed loop
327  v_ctl_auto_groundspeed_sum_err = v_ctl_auto_airspeed_controlled / (v_ctl_auto_groundspeed_pgain *
328  v_ctl_auto_groundspeed_igain);
329  }
330 #else
331  v_ctl_auto_airspeed_controlled = v_ctl_auto_airspeed_setpoint_slew;
332 #endif
333 
334  // Airspeed outerloop: positive means we need to accelerate
335  float speed_error = v_ctl_auto_airspeed_controlled - (*stateGetAirspeed_f());
336 
337  // Speed Controller to PseudoControl: gain 1 -> 5m/s error = 0.5g acceleration
338  v_ctl_desired_acceleration = speed_error * v_ctl_airspeed_pgain / 9.81f;
339  BoundAbs(v_ctl_desired_acceleration, v_ctl_max_acceleration);
340 
341  // Actual Acceleration from IMU: attempt to reconstruct the actual kinematic acceleration
342 #ifndef SITL
343  struct Int32Vect3 accel_meas_body;
344  struct Int32RMat *body_to_imu_rmat = orientationGetRMat_i(&imu.body_to_imu);
345  int32_rmat_transp_vmult(&accel_meas_body, body_to_imu_rmat, &imu.accel);
346  float vdot = ACCEL_FLOAT_OF_BFP(accel_meas_body.x) / 9.81f - sinf(stateGetNedToBodyEulers_f()->theta);
347 #else
348  float vdot = 0;
349 #endif
350 
351  // Acceleration Error: positive means UAV needs to accelerate: needs extra energy
352  float vdot_err = low_pass_vdot(v_ctl_desired_acceleration - vdot);
353 
354  // Flight Path Outerloop: positive means needs to climb more: needs extra energy
355  float gamma_err = (v_ctl_climb_setpoint - stateGetSpeedEnu_f()->z) / v_ctl_auto_airspeed_controlled;
356 
357  // Total Energy Error: positive means energy should be added
358  float en_tot_err = gamma_err + vdot_err;
359 
360  // Energy Distribution Error: positive means energy should go from overspeed to altitude = pitch up
361  float en_dis_err = gamma_err - vdot_err;
362 
363  // Auto Cruise Throttle
364  if (launch && (v_ctl_mode >= V_CTL_MODE_AUTO_CLIMB)) {
365  v_ctl_auto_throttle_nominal_cruise_throttle +=
366  v_ctl_auto_throttle_of_airspeed_igain * speed_error * dt_attidude
367  + en_tot_err * v_ctl_energy_total_igain * dt_attidude;
368  Bound(v_ctl_auto_throttle_nominal_cruise_throttle, 0.0f, 1.0f);
369  }
370 
371  // Total Controller
372  float controlled_throttle = v_ctl_auto_throttle_nominal_cruise_throttle
373  + v_ctl_auto_throttle_climb_throttle_increment * v_ctl_climb_setpoint
374  + v_ctl_auto_throttle_of_airspeed_pgain * speed_error
375  + v_ctl_energy_total_pgain * en_tot_err;
376 
377  if ((controlled_throttle >= 1.0f) || (controlled_throttle <= 0.0f) || (kill_throttle == 1)) {
378  // If your energy supply is not sufficient, then neglect the climb requirement
379  en_dis_err = -vdot_err;
380 
381  // adjust climb_setpoint to maintain airspeed in case of an engine failure or an unrestriced climb
382  if (v_ctl_climb_setpoint > 0) { v_ctl_climb_setpoint += - 30. * dt_attidude; }
383  if (v_ctl_climb_setpoint < 0) { v_ctl_climb_setpoint += 30. * dt_attidude; }
384  }
385 
386 
387  /* pitch pre-command */
388  if (launch && (v_ctl_mode >= V_CTL_MODE_AUTO_CLIMB)) {
389  v_ctl_auto_throttle_nominal_cruise_pitch += v_ctl_auto_pitch_of_airspeed_igain * (-speed_error) * dt_attidude
390  + v_ctl_energy_diff_igain * en_dis_err * dt_attidude;
391  Bound(v_ctl_auto_throttle_nominal_cruise_pitch, H_CTL_PITCH_MIN_SETPOINT, H_CTL_PITCH_MAX_SETPOINT);
392  }
393  float v_ctl_pitch_of_vz =
394  + (v_ctl_climb_setpoint /*+ d_err * v_ctl_auto_throttle_pitch_of_vz_dgain*/) * v_ctl_auto_throttle_pitch_of_vz_pgain
395  - v_ctl_auto_pitch_of_airspeed_pgain * speed_error
396  + v_ctl_auto_pitch_of_airspeed_dgain * vdot
397  + v_ctl_energy_diff_pgain * en_dis_err
398  + v_ctl_auto_throttle_nominal_cruise_pitch;
399  if (kill_throttle) { v_ctl_pitch_of_vz = v_ctl_pitch_of_vz - 1 / V_CTL_GLIDE_RATIO; }
400 
401  v_ctl_pitch_setpoint = v_ctl_pitch_of_vz + nav_pitch;
402  Bound(v_ctl_pitch_setpoint, H_CTL_PITCH_MIN_SETPOINT, H_CTL_PITCH_MAX_SETPOINT)
403 
404  ac_char_update(controlled_throttle, v_ctl_pitch_of_vz, v_ctl_climb_setpoint, v_ctl_desired_acceleration);
405 
406  v_ctl_throttle_setpoint = TRIM_UPPRZ(controlled_throttle * MAX_PPRZ);
407 }
