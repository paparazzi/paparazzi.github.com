152void gps_ubx_ucenter_init(void)
153{
154 // Start UCenter
155 gps_ubx_ucenter.status = GPS_UBX_UCENTER_STATUS_AUTOBAUD;
156 gps_ubx_ucenter.reply = GPS_UBX_UCENTER_REPLY_NONE;
157 gps_ubx_ucenter.cnt = 0;
158
159 gps_ubx_ucenter.baud_init = 0;
160 gps_ubx_ucenter.baud_run = 0;
161 gps_ubx_ucenter.baud_target = UBX_GPS_BAUD;
162
163 gps_ubx_ucenter.sw_ver_h = 0;
164 gps_ubx_ucenter.sw_ver_l = 0;
165 gps_ubx_ucenter.hw_ver_h = 0;
166 gps_ubx_ucenter.hw_ver_l = 0;
167 gps_ubx_ucenter.sbas_in_use = 0;
168 gps_ubx_ucenter.gnss_in_use = 0;
169
170 gps_ubx_ucenter.cfg_prt_reply = false;
171 gps_ubx_ucenter.version_reply = false;
172 gps_ubx_ucenter.protver = 0;
173 gps_ubx_ucenter.valset = false;
174
175 for (int i = 0; i < GPS_UBX_UCENTER_CONFIG_STEPS; i++) {
176 gps_ubx_ucenter.replies[i] = 0;
177 }
178
179 gps_ubx_ucenter.dev = &(UBX_GPS_PORT).device;
180
181#if PERIODIC_TELEMETRY
182 // The init also runs again when the configuration is restarted from the settings
183 static bool telemetry_registered = false;
184 if (!telemetry_registered) {
185 register_periodic_telemetry(DefaultPeriodic, PPRZ_MSG_ID_UBLOX_INFO, send_ublox_info);
186 telemetry_registered = true;
187 }
188#endif
189}

205void gps_ubx_ucenter_periodic(void)
206{
207 switch (gps_ubx_ucenter.status) {
208 // Save processing time inflight
209 case GPS_UBX_UCENTER_STATUS_STOPPED:
210 return;
211 // Automatically Determine Current Baudrate
212 case GPS_UBX_UCENTER_STATUS_AUTOBAUD:
213#ifdef GPS_I2C
214 gps_ubx_ucenter.cnt = 0;
215 gps_ubx_ucenter.status = GPS_UBX_UCENTER_STATUS_CONFIG;
216#else
217 if (!gps_ubx_ucenter_autobaud(gps_ubx_ucenter.cnt)) {
218 gps_ubx_ucenter.status = GPS_UBX_UCENTER_STATUS_CONFIG;
219 gps_ubx_ucenter.cnt = 0;
220 } else {
221 gps_ubx_ucenter.cnt++;
222 }
223#endif /* GPS_I2C */
224 break;
225 // Send Configuration
226 case GPS_UBX_UCENTER_STATUS_CONFIG:
227 if (!gps_ubx_ucenter_configure_step(gps_ubx_ucenter.cnt)) {
228 gps_ubx_ucenter.status = GPS_UBX_UCENTER_STATUS_STOPPED;
229#ifdef GPS_I2C
230 gps_i2c_begin();
231#endif
232 gps_ubx_ucenter.cnt = 0;
233 } else {
234 gps_ubx_ucenter.cnt++;
235#ifdef GPS_I2C
236 gps_ubx_ucenter.status = GPS_UBX_UCENTER_STATUS_WAITING;
237#endif
238 }
239 break;
240#ifdef GPS_I2C
241 case GPS_UBX_UCENTER_STATUS_WAITING:
242 if (gps_i2c_tx_is_ready()) {
243 gps_ubx_ucenter.status = GPS_UBX_UCENTER_STATUS_CONFIG;
244 }
245 break;
246#endif /* GPS_I2C */
247 default:
248 break;
249 }
250}

257static void gps_ubx_ucenter_parse_sw_version(uint8_t *payload)
258{
259 // swVersion: 30 characters, the version is the first number
260 for (uint8_t i = 0; i < 26; i++) {
261 uint8_t c = UBX_MON_VER_c(payload, i);
262 if (c >= '0' && c <= '9' && UBX_MON_VER_c(payload, i + 1) == '.') {
263 gps_ubx_ucenter.sw_ver_h = c - '0';
264 gps_ubx_ucenter.sw_ver_l = 10 * (UBX_MON_VER_c(payload, i + 2) - '0') + (UBX_MON_VER_c(payload, i + 3) - '0');
265 return;
266 }
267 }
268}
269
272static uint8_t gps_ubx_ucenter_parse_protver(uint8_t *payload, uint16_t len)
273{
274 static const char protver[] = "PROTVER";
275 for (uint16_t ext = 40; ext + 30 <= len; ext += 30) {
276 uint8_t i = 0;
277 while (i < 7 && UBX_MON_VER_c(payload, ext + i) == protver[i]) {
278 i++;
279 }
280 if (i == 7) {
281 // "PROTVER=" or "PROTVER " followed by the major
282 uint8_t major = 0;
283 for (i = 8; i < 11; i++) {
284 uint8_t c = UBX_MON_VER_c(payload, ext + i);
285 if (c < '0' || c > '9') {
286 break;
287 }
288 major = 10 * major + (c - '0');
289 }
290 return major;
291 }
292 }
293 return 0;
294}

297static uint8_t gps_ubx_ucenter_hex(char c)
298{
299 if (c >= '0' && c <= '9') {
300 return c - '0';
301 } else if (c >= 'A' && c <= 'F') {
302 return c - 'A' + 10;
303 } else if (c >= 'a' && c <= 'f') {
304 return c - 'a' + 10;
305 }
306 return 0;
307}

309void gps_ubx_ucenter_event(void)
310{
311 // Save processing time inflight
312 if (gps_ubx_ucenter.status == GPS_UBX_UCENTER_STATUS_STOPPED) {
313 return;
314 }
315
316 // Read Configuration Reply's
317 switch (gps_ubx[0].msg_class) {
318 case UBX_ACK_ID:
319 if (gps_ubx[0].msg_id == UBX_ACK_ACK_ID) {
320 gps_ubx_ucenter.reply = GPS_UBX_UCENTER_REPLY_ACK;
321 } else if (gps_ubx[0].msg_id == UBX_ACK_NAK_ID) {
322 gps_ubx_ucenter.reply = GPS_UBX_UCENTER_REPLY_NACK;
323 }
324 break;
325 case UBX_MON_ID:
326 if (gps_ubx[0].msg_id == UBX_MON_VER_ID) {
327 gps_ubx_ucenter.reply = GPS_UBX_UCENTER_REPLY_VERSION;
328 gps_ubx_ucenter.version_reply = true;
329 gps_ubx_ucenter_parse_sw_version(gps_ubx[0].msg_buf);
330 gps_ubx_ucenter.protver = gps_ubx_ucenter_parse_protver(gps_ubx[0].msg_buf, gps_ubx[0].len);
331 // hwVersion: 10 hexadecimal characters at 30, e.g. "00190000" (generation 9), "000B0000" (X20)
332 gps_ubx_ucenter.hw_ver_h = 16 * gps_ubx_ucenter_hex(UBX_MON_VER_c(gps_ubx[0].msg_buf, 32)) +
333 gps_ubx_ucenter_hex(UBX_MON_VER_c(gps_ubx[0].msg_buf, 33));
334 gps_ubx_ucenter.hw_ver_l = 16 * gps_ubx_ucenter_hex(UBX_MON_VER_c(gps_ubx[0].msg_buf, 36)) +
335 gps_ubx_ucenter_hex(UBX_MON_VER_c(gps_ubx[0].msg_buf, 37));
336 } else if (gps_ubx[0].msg_id == UBX_MON_GNSS_ID) {
337 gps_ubx_ucenter.gnss_in_use = UBX_MON_GNSS_enabled(gps_ubx[0].msg_buf);
338 } else if (gps_ubx[0].msg_id == UBX_MON_COMMS_ID) {
339 // Port this reply is output on (X20, 0 when not reported): 1 I2C, 2 UART1, 3 UART2, 4 USB, 5 SPI
340 uint8_t output_port = (UBX_MON_COMMS_txErrors(gps_ubx[0].msg_buf) >> 2) & 0x07;
341 if (output_port != 0) {
342 gps_ubx_ucenter.port_id = output_port - 1;
343 }
344 }
345 break;
346 case UBX_CFG_ID:
347 if (gps_ubx[0].msg_id == UBX_CFG_PRT_ID) {
348 gps_ubx_ucenter.reply = GPS_UBX_UCENTER_REPLY_CFG_PRT;
349 gps_ubx_ucenter.cfg_prt_reply = true;
350 gps_ubx_ucenter.port_id = UBX_CFG_PRT_PortId(gps_ubx[0].msg_buf, 0);
351 gps_ubx_ucenter.baud_run = UBX_CFG_PRT_Baudrate(gps_ubx[0].msg_buf, 0);
352 }
353 break;
354 default:
355 break;
356 }
357}

398static bool gps_ubx_ucenter_autobaud(uint8_t nr)
399{
400 // Very important for some modules: give the GPS some time to boot (up to 0.75 second)
401 if (nr < 2) {
402 return true;
403 }
404
405 // The polls of the previous step were answered at that baudrate
406 if (nr > 2 && (gps_ubx_ucenter.cfg_prt_reply || gps_ubx_ucenter.version_reply)) {
407 gps_ubx_ucenter.baud_init = gps_ubx_ucenter.baud_run;
408 // The legacy UBX-CFG messages are gone from protocol version 40 (e.g. X20). The configuration
409 // interface exists from 27 (generation 9, F9 HPG 1.00): also used then when the CFG-PRT poll isn't answered.
410 gps_ubx_ucenter.valset = gps_ubx_ucenter.protver >= 40 ||
411 (gps_ubx_ucenter.protver >= 27 && !gps_ubx_ucenter.cfg_prt_reply);
412 // Not over I2C: when the receiver didn't report the port (still the I2C/DDC default), assume UART1
413 if (gps_ubx_ucenter.port_id == GPS_PORT_DDC) {
414 gps_ubx_ucenter.port_id = GPS_PORT_UART1;
415 }
416 return false;
417 }
418
419 // Try the target baudrate first (often already configured), then the others
420 uint8_t i = nr - 2;
421 if (i <= GPS_UBX_UCENTER_NB_BAUDS) {
422 uint32_t baud = (i == 0) ? gps_ubx_ucenter.baud_target : gps_ubx_ucenter_bauds[i - 1];
423 gps_ubx_ucenter.baud_run = UART_SPEED(baud); // the CFG-PRT reply overwrites it with the reported one
424 uart_periph_set_baudrate(&(UBX_GPS_PORT), baud);
425 gps_ubx_ucenter_autobaud_poll();
426 return true;
427 }
428
429 // Autoconfig failed... set the failsafe baudrate
430 gps_ubx_ucenter.baud_init = 0; // Set as zero to indicate that we couldn't verify the baudrate
431 uart_periph_set_baudrate(&(UBX_GPS_PORT), B9600);
432 return false;
433}
434#endif /* GPS_I2C */

501// UART mode: 8N1 with reserved1 set for compatability with A4
502#define UBX_UART_MODE_MASK 0x000008D0
503
504#define UBX_PROTO_MASK 0x0001
505#define NMEA_PROTO_MASK 0x0002
506#define RTCM_PROTO_MASK 0x0004
507#define RTCM3_PROTO_MASK 0x0020
508
509static inline void gps_ubx_ucenter_config_port(void)
510{
511 switch (gps_ubx_ucenter.port_id) {
512 // I2C Interface
513 case GPS_PORT_DDC:
514#ifdef GPS_I2C
515 UbxSend_CFG_PRT(gps_ubx_ucenter.dev, gps_ubx_ucenter.port_id, 0x0, 0x0, (0x42 << 1), 0x0, UBX_PROTO_MASK,
516 UBX_PROTO_MASK, 0x0, 0x0);
517#endif
518 break;
519 // UART Interface
520 case GPS_PORT_UART1:
521 case GPS_PORT_UART2:
522 UbxSend_CFG_PRT(gps_ubx_ucenter.dev,
523 gps_ubx_ucenter.port_id, RESERVED, RESERVED,
524 UBX_UART_MODE_MASK, UART_SPEED(gps_ubx_ucenter.baud_target), UBX_PROTO_MASK | NMEA_PROTO_MASK | RTCM3_PROTO_MASK,
525 UBX_PROTO_MASK | (NMEA_PROTO_MASK & GPS_UBX_ENABLE_NMEA_DATA_MASK), 0x0, 0x0);
526 break;
527 // USB Interface
528 case GPS_PORT_USB:
529 UbxSend_CFG_PRT(gps_ubx_ucenter.dev,
530 gps_ubx_ucenter.port_id, 0x0, 0x0, 0x0, 0x0,
531 UBX_PROTO_MASK | NMEA_PROTO_MASK | RTCM3_PROTO_MASK, UBX_PROTO_MASK | (NMEA_PROTO_MASK & GPS_UBX_ENABLE_NMEA_DATA_MASK), 0x0, 0x0);
532 break;
533 default:
534 // SPI is not supported
535 break;
536 }

558static bool gps_ubx_ucenter_configure(uint8_t nr)
559{
560 // Store the reply of the last configuration step and reset
561 if (nr < GPS_UBX_UCENTER_CONFIG_STEPS) {
562 gps_ubx_ucenter.replies[nr] = gps_ubx_ucenter.reply;
563 }
564
565 switch (nr) {
566 case 0:
567 // Use old baudrate to issue a baudrate change command
568 gps_ubx_ucenter_config_port();
569 break;
570 case 1:
571#ifndef GPS_I2C
572 // Now the GPS baudrate should have changed, unless the receiver refused it
573 if (gps_ubx_ucenter.reply != GPS_UBX_UCENTER_REPLY_NACK) {
574 uart_periph_set_baudrate(&(UBX_GPS_PORT), gps_ubx_ucenter.baud_target);
575 gps_ubx_ucenter.baud_run = UART_SPEED(gps_ubx_ucenter.baud_target);
576 }
577#endif /*GPS_I2C*/
578 UbxSend_MON_GET_VER(gps_ubx_ucenter.dev);
579 break;
580 case 2:
581 case 3:
582 case 4:
583 case 5:
584 // UBX_G5010 takes 0.7 seconds to answer a firmware request

727#define CFG_UART2OUTPROT_NMEA 0x10760002
728/* Message output rate keys (per navigation solution) for the I2C port, the key for the other ports is
729 * the port number (CFG-PRT) higher: UART1 +1, UART2 +2, USB +3, SPI +4 */
730#define CFG_MSGOUT_NAV_PVT 0x20910006
731#define CFG_MSGOUT_NAV_POSLLH 0x20910029
732#define CFG_MSGOUT_NAV_VELNED 0x20910042
733#define CFG_MSGOUT_NAV_STATUS 0x2091001a
734#define CFG_MSGOUT_NAV_SAT 0x20910015
735#define CFG_MSGOUT_NAV_RELPOSNED 0x2091008d
736#define CFG_MSGOUT_NAV_DAHEADING 0x209103df // X20D

738#if USE_GPS_UBX_RTCM

745#define VALSET_RECEIVER 0 // the receiver

777static uint32_t gps_ubx_ucenter_valset_key(uint8_t type, uint32_t key, uint32_t key_uart2)
778{
779 switch (type) {
780 case VALSET_UART:
781 if (gps_ubx_ucenter.port_id == GPS_PORT_UART1) {
782 return key;
783 } else if (gps_ubx_ucenter.port_id == GPS_PORT_UART2) {
784 return key_uart2;
785 }
786 return 0;
787 case VALSET_MSGOUT:
788 return key + gps_ubx_ucenter.port_id;
789 default:
790 return key;
791 }
792}

795static void gps_ubx_ucenter_valset(uint32_t key, uint32_t value)
796{
797 switch ((key >> 28) & 0x07) {
798 case 0: // no key
799 break;
800 case 1: // one bit
801 case 2: // one byte
802 UbxSend_CFG_VALSET_U1(gps_ubx_ucenter.dev, 0, VALSET_LAYER_RAM, RESERVED, key, value);
803 break;
804 case 3: // two bytes
805 UbxSend_CFG_VALSET_U2(gps_ubx_ucenter.dev, 0, VALSET_LAYER_RAM, RESERVED, key, value);
806 break;
807 default: // four bytes
808 UbxSend_CFG_VALSET_U4(gps_ubx_ucenter.dev, 0, VALSET_LAYER_RAM, RESERVED, key, value);
809 break;
810 }
811}

814static bool gps_ubx_ucenter_configure_valset(uint8_t nr)
815{
816 // Store the reply of the last configuration step and reset
817 if (nr < GPS_UBX_UCENTER_CONFIG_STEPS) {
818 gps_ubx_ucenter.replies[nr] = gps_ubx_ucenter.reply;
819 }
820
821 if (nr == 0) {
822 // Use the found baudrate to change it to the target one
823 gps_ubx_ucenter_valset(gps_ubx_ucenter_valset_key(VALSET_UART, CFG_UART1_BAUDRATE, CFG_UART2_BAUDRATE),
824 UART_SPEED(gps_ubx_ucenter.baud_target));
825 } else if (nr == 1) {
826#ifndef GPS_I2C
827 // Now the receiver baudrate should have changed, unless the receiver refused it
828 if (gps_ubx_ucenter.reply != GPS_UBX_UCENTER_REPLY_NACK) {
829 uart_periph_set_baudrate(&(UBX_GPS_PORT), gps_ubx_ucenter.baud_target);
830 gps_ubx_ucenter.baud_run = UART_SPEED(gps_ubx_ucenter.baud_target);
831 }
832#endif
833 } else if (nr < 2 + VALSET_CFG_NB) {
834 uint8_t i = nr - 2;
835 // A slower measurement rate when it is refused (NAK, e.g. in moving base heading mode)
836 if (i > 0 && gps_ubx_ucenter_valset_cfg[i - 1].key == CFG_RATE_MEAS &&
837 gps_ubx_ucenter.reply == GPS_UBX_UCENTER_REPLY_NACK &&
838 gps_ubx_ucenter_valset_cfg[i - 1].value < GPS_UBX_UCENTER_RATE_VALSET_FALLBACK) {
839 gps_ubx_ucenter_valset(CFG_RATE_MEAS, GPS_UBX_UCENTER_RATE_VALSET_FALLBACK);
840 }
841 gps_ubx_ucenter_valset(gps_ubx_ucenter_valset_key(gps_ubx_ucenter_valset_cfg[i].type,
842 gps_ubx_ucenter_valset_cfg[i].key, gps_ubx_ucenter_valset_cfg[i].key_uart2),
843 gps_ubx_ucenter_valset_cfg[i].value);
844 } else if (nr == 2 + VALSET_CFG_NB) {
845#if GPS_UBX_UCENTER_SAVE
846 // Save the configuration in RAM to battery backed RAM and flash
847 UbxSend_CFG_CFG(gps_ubx_ucenter.dev, 0x00000000, 0xffffffff, 0x00000000);
848#endif
849 } else if (nr == 3 + VALSET_CFG_NB) {
850 UbxSend_MON_GET_GNSS(gps_ubx_ucenter.dev);
851 } else if (nr == 4 + VALSET_CFG_NB) {
852 // Wait for the GNSS reply
853 } else {
854 gps_ubx_ucenter_send_info(0);
855 return false;
856 }
857
858 gps_ubx_ucenter.reply = GPS_UBX_UCENTER_REPLY_NONE;
859 return true; // Continue, except for the last step
860}
