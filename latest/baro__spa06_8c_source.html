127void baro_spa06_event(void)
128{
129 spa06_event(&baro_spa06);
130
131 if (baro_spa06.data_available) {
132 uint32_t now_ts = get_sys_time_usec();
133 // send ABI message
134 AbiSendMsgBARO_ABS(BARO_SPA_SENDER_ID, now_ts, baro_spa06.pressure);
135 AbiSendMsgTEMPERATURE(BARO_SPA_SENDER_ID, baro_spa06.temperature);
136 baro_spa06.data_available = false;
137 baro_spa06_alt = pprz_isa_altitude_of_pressure(baro_spa06.pressure);
138 baro_spa06_alt_valid = true;
139 baro_spa06_press = baro_spa06.pressure; // Pa
140 baro_spa06_temp = baro_spa06.temperature; // deg Celcius
141
142#if defined(SPA06_SYNC_SEND)
143 int32_t up = (int32_t)(baro_spa06.raw_pressure);
144 int32_t ut = (int32_t)(baro_spa06.raw_temperature);
145 int32_t p = (int32_t) baro_spa06.pressure;
146 int32_t t = (int32_t)(baro_spa06.temperature * 10.0f); // 0.1 deg Celcius
147 DOWNLINK_SEND_BMP_STATUS(DefaultChannel, DefaultDevice, &up, &ut, &p, &t);
148#endif
149 }
