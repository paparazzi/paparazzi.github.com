94void baro_spa06_event(void)
95{
96 spa06_event(&baro_spa06);
97
98 if (baro_spa06.data_available) {
99 uint32_t now_ts = get_sys_time_usec();
100 // send ABI message
101 AbiSendMsgBARO_ABS_RAW(BARO_SPA_SENDER_ID, now_ts, baro_spa06.pressure);
102 AbiSendMsgTEMPERATURE_RAW(BARO_SPA_SENDER_ID, now_ts, baro_spa06.temperature, TEMPERATURE_TYPE_BARO);
103 baro_spa06.data_available = false;
104 baro_spa06_alt = pprz_isa_altitude_of_pressure(baro_spa06.pressure);
105 baro_spa06_alt_valid = true;
106 }
